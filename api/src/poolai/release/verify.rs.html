30#[derive(Debug, Deserialize)]
31struct SignatureEnvelope {
32    algorithm: String,
33    key_id: String,
34    signature_hex: String,
35}

37pub fn verify_release(
38    opts: VerifyReleaseOptions,
39) -> Result<VerifyReleaseReport, VerifyReleaseError> {
40    match verify_release_inner(opts) {
41        Ok(report) => {
42            crate::grid::galaxy_governance_metrics::record_release_verify_success();
43            crate::grid::galaxy_update_policy::tick_update_notify_from_env();
44            Ok(report)
45        }
46        Err(err) => {
47            crate::grid::galaxy_governance_metrics::record_release_verify_fail();
48            Err(err)
49        }
50    }
51}
52
53fn verify_release_inner(
54    opts: VerifyReleaseOptions,
55) -> Result<VerifyReleaseReport, VerifyReleaseError> {
56    let manifest_path = &opts.manifest_path;
57    if !manifest_path.exists() {
58        return Err(VerifyReleaseError::ManifestNotFound(manifest_path.clone()));
59    }
60    let sig_path = &opts.signature_path;
61    if !sig_path.exists() {
62        return Err(VerifyReleaseError::SignatureNotFound(sig_path.clone()));
63    }
64
65    let manifest_bytes = read_file(manifest_path)?;
66    let sig_bytes = read_file(sig_path)?;
67    let envelope: SignatureEnvelope = serde_json::from_slice(&sig_bytes)
68        .map_err(|e| VerifyReleaseError::InvalidSignatureJson(e.to_string()))?;
69
70    if envelope.algorithm != "ed25519" {
71        return Err(VerifyReleaseError::UnsupportedAlgorithm(
72            envelope.algorithm.clone(),
73        ));
74    }
75
76    let pk_hex = resolve_public_key_hex(
77        &envelope.key_id,
78        opts.trust_root_path.as_deref(),
79        opts.public_key_hex.as_deref(),
80    )?;
81    verify_ed25519(&manifest_bytes, &envelope.signature_hex, &pk_hex)?;
82
83    let manifest = ReleaseManifest::parse_json(&manifest_bytes)
84        .map_err(|e| VerifyReleaseError::InvalidManifestJson(e.to_string()))?;
85
86    let mut artifacts_verified = Vec::new();
87    if let Some(artifact_path) = &opts.artifact_path {
88        let name = opts
89            .artifact_name
90            .clone()
91            .or_else(|| {
92                artifact_path
93                    .file_name()
94                    .map(|s| s.to_string_lossy().into_owned())
95            })
96            .unwrap_or_else(|| "artifact".to_string());
97        verify_artifact_sha256(&manifest, &name, artifact_path)?;
98        artifacts_verified.push(name);
99    }
100
101    Ok(VerifyReleaseReport {
102        manifest_version: manifest.version,
103        git_tag: manifest.git_tag,
104        protocol_min: manifest.protocol_min,
105        protocol_max: manifest.protocol_max,
106        signature_key_id: envelope.key_id,
107        artifacts_verified,
108    })
109}

111fn read_file(path: &Path) -> Result<Vec<u8>, VerifyReleaseError> {
112    std::fs::read(path).map_err(|e| VerifyReleaseError::IoRead {
113        path: path.to_path_buf(),
114        source: e,
115    })
116}

118fn resolve_public_key_hex(
119    key_id: &str,
120    trust_root_path: Option<&Path>,
121    public_key_hex: Option<&str>,
122) -> Result<String, VerifyReleaseError> {
123    if let Some(hex) = public_key_hex {
124        return Ok(hex.trim().to_string());
125    }
126    let Some(path) = trust_root_path else {
127        return Err(VerifyReleaseError::MissingPublicKey);
128    };
129    let root = TrustRoot::load(path)?;
130    let map = root.key_map();
131    map.get(key_id)
132        .map(|h| (*h).to_string())
133        .ok_or_else(|| VerifyReleaseError::UnknownKeyId(key_id.to_string()))
134}

136fn verify_ed25519(
137    message: &[u8],
138    signature_hex: &str,
139    public_key_hex: &str,
140) -> Result<(), VerifyReleaseError> {
141    let pk_bytes = hex::decode(public_key_hex.trim())
142        .map_err(|e| VerifyReleaseError::InvalidPublicKeyHex(e.to_string()))?;
143    let vk = VerifyingKey::from_bytes(
144        pk_bytes
145            .as_slice()
146            .try_into()
147            .map_err(|_| VerifyReleaseError::InvalidPublicKeyHex("expected 32 bytes".into()))?,
148    )
149    .map_err(|e| VerifyReleaseError::InvalidPublicKeyHex(e.to_string()))?;
150
151    let sig_bytes = hex::decode(signature_hex.trim())
152        .map_err(|e| VerifyReleaseError::InvalidSignatureHex(e.to_string()))?;
153    let signature = Signature::from_bytes(
154        sig_bytes
155            .as_slice()
156            .try_into()
157            .map_err(|_| VerifyReleaseError::InvalidSignatureHex("expected 64 bytes".into()))?,
158    );
159
160    vk.verify_strict(message, &signature)
161        .map_err(|_| VerifyReleaseError::BadSignature)
162}

164fn verify_artifact_sha256(
165    manifest: &ReleaseManifest,
166    name: &str,
167    artifact_path: &Path,
168) -> Result<(), VerifyReleaseError> {
169    if !artifact_path.exists() {
170        return Err(VerifyReleaseError::ArtifactNotFound(
171            artifact_path.to_path_buf(),
172        ));
173    }
174    let entry = manifest
175        .find_artifact(name)
176        .ok_or_else(|| VerifyReleaseError::ArtifactNotInManifest(name.to_string()))?;
177    let bytes = read_file(artifact_path)?;
178    let digest = Sha256::digest(&bytes);
179    let actual = hex::encode(digest);
180    let expected = entry.sha256.trim().to_ascii_lowercase();
181    if actual != expected {
182        return Err(VerifyReleaseError::ArtifactSha256Mismatch {
183            name: name.to_string(),
184            expected,
185            actual,
186        });
187    }
188    Ok(())
189}

207    fn write_signature(dir: &tempfile::TempDir, manifest_bytes: &[u8], key_id: &str) -> PathBuf {
208        let sig = test_signing_key().sign(manifest_bytes);
209        let envelope = serde_json::json!({
210            "algorithm": "ed25519",
211            "key_id": key_id,
212            "signature_hex": hex::encode(sig.to_bytes()),
213        });
214        let path = dir.path().join("manifest.json.sig");
215        std::fs::write(&path, serde_json::to_vec_pretty(&envelope).unwrap()).unwrap();
216        path
217    }

233    fn verify_release_ok() {
234        let dir = tempfile::tempdir().unwrap();
235        let artifact_path = dir.path().join("poolai.exe");
236        std::fs::write(&artifact_path, b"poolai-test-artifact-v1").unwrap();
237        let hash = hex::encode(Sha256::digest(b"poolai-test-artifact-v1"));
238
239        let mut manifest = sample_manifest_json();
240        let manifest_str = String::from_utf8(manifest.clone()).unwrap();
241        let manifest_str = manifest_str.replace("PLACEHOLDER", &hash);
242        manifest = manifest_str.into_bytes();
243
244        let manifest_path = dir.path().join("manifest.json");
245        std::fs::write(&manifest_path, &manifest).unwrap();
246
247        let sig_path = write_signature(&dir, &manifest, "poolai-dev");
248        let trust_path = dir.path().join("trust_root.json");
249        std::fs::write(
250            &trust_path,
251            format!(
252                r#"{{"maintainer_keys":[{{"key_id":"poolai-dev","public_key_hex":"{}"}}]}}"#,
253                test_public_hex()
254            ),
255        )
256        .unwrap();
257
258        let report = verify_release(VerifyReleaseOptions {
259            manifest_path: manifest_path.clone(),
260            signature_path: sig_path,
261            trust_root_path: Some(trust_path),
262            public_key_hex: None,
263            artifact_path: Some(artifact_path),
264            artifact_name: Some("poolai".into()),
265        })
266        .unwrap();
267
268        assert_eq!(report.manifest_version, "0.2.2");
269        assert_eq!(report.artifacts_verified, vec!["poolai"]);
