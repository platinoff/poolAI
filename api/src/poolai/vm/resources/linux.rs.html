1//! Linux-specific resource limiting using cgroups
2//!
3//! Provides:
4//! - CPU limits via cgroup v2 (cpu.max) or v1 (cpu.cfs_quota_us, cpu.cfs_period_us)
5//! - Memory limits via cgroup v2 (memory.max) or v1 (memory.limit_in_bytes)
6//! - Resource usage monitoring from cgroup stats
7
8use crate::core::error::AppError;

10use std::path::{Path, PathBuf};
11use tokio::fs;

15/// Linux cgroup manager for resource limiting
16#[allow(dead_code)] // Used in resources.rs
17pub struct LinuxCgroupLimiter {
18    /// Base path for cgroup filesystem
19    cgroup_root: PathBuf,
20    /// Cgroup version (v1 or v2)
21    cgroup_version: CgroupVersion,
22}
23
24#[derive(Debug, Clone, Copy)]
25enum CgroupVersion {
26    V1,
27    V2,
28}
29
30impl LinuxCgroupLimiter {
31    /// Create a new Linux cgroup limiter
32    pub fn new() -> Result<Self, AppError> {
33        // Detect cgroup version
34        let cgroup_root = if Path::new("/sys/fs/cgroup/unified").exists() {
35            // cgroup v2 (unified hierarchy)
36            PathBuf::from("/sys/fs/cgroup")
37        } else if Path::new("/sys/fs/cgroup/cpu").exists() {
38            // cgroup v1 (separate controllers)
39            PathBuf::from("/sys/fs/cgroup")
40        } else {
41            return Err(AppError::ConfigError(
42                "cgroup filesystem not found".to_string(),
43            ));
44        };
45
46        let cgroup_version = if Path::new("/sys/fs/cgroup/cgroup.controllers").exists() {
47            CgroupVersion::V2
48        } else {
49            CgroupVersion::V1
50        };
51
52        Ok(Self {
53            cgroup_root,
54            cgroup_version,
55        })
56    }
57
58    /// Placeholder when cgroup fs is unavailable (post-spawn becomes no-op).
59    pub fn disabled() -> Self {
60        Self {
61            cgroup_root: PathBuf::new(),
62            cgroup_version: CgroupVersion::V2,
63        }
64    }

66    pub fn is_available(&self) -> bool {
67        !self.cgroup_root.as_os_str().is_empty() && self.cgroup_root.exists()
68    }

70    /// Get cgroup path for a process
71    fn get_cgroup_path(&self, process_id: Uuid, controller: &str) -> PathBuf {
72        let cgroup_name = format!("poolai-{}", process_id);
73        match self.cgroup_version {
74            CgroupVersion::V2 => {
75                // cgroup v2: single unified hierarchy
76                self.cgroup_root.join(&cgroup_name)
77            }
78            CgroupVersion::V1 => {
79                // cgroup v1: separate controller directories
80                self.cgroup_root.join(controller).join(&cgroup_name)
81            }
82        }
83    }
84
85    /// Create cgroup for a process
86    async fn create_cgroup(&self, process_id: Uuid, controller: &str) -> Result<PathBuf, AppError> {
87        let cgroup_path = self.get_cgroup_path(process_id, controller);
88
89        // Create cgroup directory
90        fs::create_dir_all(&cgroup_path).await.map_err(|e| {
91            AppError::ConfigError(format!("Failed to create cgroup directory: {}", e))
92        })?;
93
94        Ok(cgroup_path)
95    }

97    /// Apply CPU limits using cgroup
98    async fn apply_cpu_limits(&self, process_id: Uuid, cpu_cores: u16) -> Result<(), AppError> {
99        match self.cgroup_version {
100            CgroupVersion::V2 => {
101                // cgroup v2: use cpu.max (format: "quota period" or "max" for unlimited)
102                let cgroup_path = self.create_cgroup(process_id, "cpu").await?;
103                let cpu_max_path = cgroup_path.join("cpu.max");
104
105                // Calculate quota: cpu_cores * 100000 (period is 100000 microseconds = 0.1s)
106                let period = 100000u64;
107                let quota = (cpu_cores as u64) * period;
108                let cpu_max = format!("{} {}", quota, period);
109
110                fs::write(&cpu_max_path, cpu_max.as_bytes())
111                    .await
112                    .map_err(|e| {
113                        AppError::ConfigError(format!("Failed to write cpu.max: {}", e))
114                    })?;
115            }
116            CgroupVersion::V1 => {
117                // cgroup v1: use cpu.cfs_quota_us and cpu.cfs_period_us
118                let cgroup_path = self.create_cgroup(process_id, "cpu").await?;
119
120                // Set period (100ms = 100000 microseconds)
121                let period_path = cgroup_path.join("cpu.cfs_period_us");
122                fs::write(&period_path, b"100000").await.map_err(|e| {
123                    AppError::ConfigError(format!("Failed to write cpu.cfs_period_us: {}", e))
124                })?;
125
126                // Set quota (cpu_cores * period)
127                let quota = (cpu_cores as u64) * 100000;
128                let quota_path = cgroup_path.join("cpu.cfs_quota_us");
129                fs::write(&quota_path, quota.to_string().as_bytes())
130                    .await
131                    .map_err(|e| {
132                        AppError::ConfigError(format!("Failed to write cpu.cfs_quota_us: {}", e))
133                    })?;
134            }
135        }
136
137        Ok(())
138    }

140    /// Apply memory limits using cgroup
141    async fn apply_memory_limits(&self, process_id: Uuid, memory_mb: u32) -> Result<(), AppError> {
142        let memory_bytes = (memory_mb as u64) * 1024 * 1024;
143        let cgroup_path = self.create_cgroup(process_id, "memory").await?;
144
145        match self.cgroup_version {
146            CgroupVersion::V2 => {
147                // cgroup v2: use memory.max
148                let memory_max_path = cgroup_path.join("memory.max");
149                fs::write(&memory_max_path, memory_bytes.to_string().as_bytes())
150                    .await
151                    .map_err(|e| {
152                        AppError::ConfigError(format!("Failed to write memory.max: {}", e))
153                    })?;
154            }
155            CgroupVersion::V1 => {
156                // cgroup v1: use memory.limit_in_bytes
157                let memory_limit_path = cgroup_path.join("memory.limit_in_bytes");
158                fs::write(&memory_limit_path, memory_bytes.to_string().as_bytes())
159                    .await
160                    .map_err(|e| {
161                        AppError::ConfigError(format!(
162                            "Failed to write memory.limit_in_bytes: {}",
163                            e
164                        ))
165                    })?;
166            }
167        }
168
169        Ok(())
170    }

172    /// Add process to cgroup
173    async fn add_process_to_cgroup(&self, process_id: Uuid, pid: u32) -> Result<(), AppError> {
174        let cgroup_path = self.get_cgroup_path(process_id, "cpu");
175
176        match self.cgroup_version {
177            CgroupVersion::V2 => {
178                // cgroup v2: use cgroup.procs
179                let procs_path = cgroup_path.join("cgroup.procs");
180                fs::write(&procs_path, pid.to_string().as_bytes())
181                    .await
182                    .map_err(|e| {
183                        AppError::ConfigError(format!("Failed to write cgroup.procs: {}", e))
184                    })?;
185            }
186            CgroupVersion::V1 => {
187                // cgroup v1: use tasks (or cgroup.procs for thread groups)
188                let tasks_path = cgroup_path.join("tasks");
189                fs::write(&tasks_path, pid.to_string().as_bytes())
190                    .await
191                    .map_err(|e| AppError::ConfigError(format!("Failed to write tasks: {}", e)))?;
192            }
193        }
194
195        Ok(())
196    }

234    /// Get memory usage from cgroup
235    async fn get_memory_usage(&self, process_id: Uuid) -> Result<u32, AppError> {
236        let cgroup_path = self.get_cgroup_path(process_id, "memory");
237
238        match self.cgroup_version {
239            CgroupVersion::V2 => {
240                // cgroup v2: read memory.current
241                let memory_current_path = cgroup_path.join("memory.current");
242                let usage_bytes = fs::read_to_string(&memory_current_path)
243                    .await
244                    .map_err(|e| {
245                        AppError::ConfigError(format!("Failed to read memory.current: {}", e))
246                    })?
247                    .trim()
248                    .parse::<u64>()
249                    .map_err(|e| {
250                        AppError::ConfigError(format!("Failed to parse memory.current: {}", e))
251                    })?;
252
253                Ok((usage_bytes / 1024 / 1024) as u32)
254            }
255            CgroupVersion::V1 => {
256                // cgroup v1: read memory.usage_in_bytes
257                let memory_usage_path = cgroup_path.join("memory.usage_in_bytes");
258                let usage_bytes = fs::read_to_string(&memory_usage_path)
259                    .await
260                    .map_err(|e| {
261                        AppError::ConfigError(format!(
262                            "Failed to read memory.usage_in_bytes: {}",
263                            e
264                        ))
265                    })?
266                    .trim()
267                    .parse::<u64>()
268                    .map_err(|e| {
269                        AppError::ConfigError(format!(
270                            "Failed to parse memory.usage_in_bytes: {}",
271                            e
272                        ))
273                    })?;
274
275                Ok((usage_bytes / 1024 / 1024) as u32)
276            }
277        }
278    }

280    /// Apply resource limits to a process
281    pub async fn apply_limits(
282        &self,
283        process_id: Uuid,
284        pid: u32,
285        limits: &ResourceLimits,
286    ) -> Result<(), AppError> {
287        // Apply CPU limits
288        if limits.cpu_cores > 0 {
289            self.apply_cpu_limits(process_id, limits.cpu_cores).await?;
290        }
291
292        // Apply memory limits
293        if limits.memory_mb > 0 {
294            self.apply_memory_limits(process_id, limits.memory_mb)
295                .await?;
296        }
297
298        // Add process to cgroup
299        self.add_process_to_cgroup(process_id, pid).await?;
300
301        Ok(())
302    }

354/// Pre-spawn hints (cgroups need PID — applied in `apply_limits_post_spawn`).
355pub async fn apply_linux_pre_spawn(
356    command: &mut Command,
357    limits: &ResourceLimits,
358) -> Result<(), AppError> {
359    validate_resource_limits(limits)?;
360    if limits.cpu_cores > 0 {
361        command.env("POOLAI_CPU_LIMIT", limits.cpu_cores.to_string());
362    }
363    if limits.memory_mb > 0 {
364        command.env("POOLAI_MEMORY_LIMIT_MB", limits.memory_mb.to_string());
365    }
366    Ok(())
