3use crate::core::error::AppError;
4use crate::security::jwt_secrets;
5use serde::{Deserialize, Serialize};

11/// Supported rotatable secret kinds.
12#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
13#[serde(rename_all = "snake_case")]
14pub enum SecretKind {
15    Jwt,
16    TlsCertificate,
17    TelegramWebhook,
18}
19
20impl SecretKind {
21    pub fn as_str(self) -> &'static str {
22        match self {
23            SecretKind::Jwt => "jwt",
24            SecretKind::TlsCertificate => "tls_certificate",
25            SecretKind::TelegramWebhook => "telegram_webhook",
26        }
27    }
28
29    pub fn from_str(s: &str) -> Option<Self> {
30        match s {
31            "jwt" => Some(SecretKind::Jwt),
32            "tls_certificate" | "tls" => Some(SecretKind::TlsCertificate),
33            "telegram_webhook" | "telegram" => Some(SecretKind::TelegramWebhook),
34            _ => None,
35        }
36    }
37}
38
39type RotationHook = Arc<dyn Fn() -> Result<(), String> + Send + Sync>;
40
41#[derive(Debug, Default)]
42struct KindMeta {
43    last_rotated_unix: Option<u64>,
44    rotation_count: u64,
45}
46
47struct Registry {
48    hooks: parking_lot::RwLock<HashMap<SecretKind, Vec<RotationHook>>>,
49    meta: parking_lot::RwLock<HashMap<SecretKind, KindMeta>>,
50}
51
52impl Registry {
53    fn new() -> Self {
54        Self {
55            hooks: parking_lot::RwLock::new(HashMap::new()),
56            meta: parking_lot::RwLock::new(HashMap::new()),
57        }
58    }
59
60    fn register(&self, kind: SecretKind, hook: RotationHook) {
61        self.hooks.write().entry(kind).or_default().push(hook);
62    }
63
64    fn run(&self, kind: SecretKind) -> Result<RotationRunReport, AppError> {
65        let hooks = self.hooks.read().get(&kind).cloned().unwrap_or_default();
66        if hooks.is_empty() {
67            return Err(AppError::ConfigError(format!(
68                "no rotation hooks registered for {}",
69                kind.as_str()
70            )));
71        }
72        let mut hook_results = Vec::new();
73        for (i, hook) in hooks.iter().enumerate() {
74            match hook() {
75                Ok(()) => hook_results.push(HookResult {
76                    index: i,
77                    ok: true,
78                    message: None,
79                }),
80                Err(e) => {
81                    hook_results.push(HookResult {
82                        index: i,
83                        ok: false,
84                        message: Some(e),
85                    });
86                }
87            }
88        }
89        let all_ok = hook_results.iter().all(|r| r.ok);
90        #[cfg(feature = "prometheus")]
91        crate::observability::record_secret_rotation(kind.as_str(), all_ok);
92        if all_ok {
93            let now = unix_now();
94            let mut meta = self.meta.write();
95            let entry = meta.entry(kind).or_default();
96            entry.last_rotated_unix = Some(now);
97            entry.rotation_count = entry.rotation_count.saturating_add(1);
98            info!(kind = kind.as_str(), "secret rotation hooks completed");
99        } else {
100            warn!(kind = kind.as_str(), "secret rotation hooks had failures");
101        }
102        Ok(RotationRunReport {
103            kind,
104            hooks: hook_results,
105            success: all_ok,
106        })
107    }
108
109    fn status(&self) -> Vec<RotationStatusEntry> {
110        let hooks = self.hooks.read();
111        let meta = self.meta.read();
112        let kinds = [
113            SecretKind::Jwt,
114            SecretKind::TlsCertificate,
115            SecretKind::TelegramWebhook,
116        ];
117        kinds
118            .into_iter()
119            .map(|kind| {
120                let m = meta.get(&kind);
121                RotationStatusEntry {
122                    kind,
123                    configured: is_kind_configured(kind),
124                    last_rotated_unix: m.and_then(|x| x.last_rotated_unix),
125                    rotation_count: m.map(|x| x.rotation_count).unwrap_or(0),
126                    grace_active: kind == SecretKind::Jwt
127                        && jwt_secrets::jwt_store().read().grace_active(),
128                    hook_count: hooks.get(&kind).map(|v| v.len()).unwrap_or(0),
129                }
130            })
131            .collect()
132    }
133}
134
135fn unix_now() -> u64 {
136    SystemTime::now()
137        .duration_since(UNIX_EPOCH)
138        .map(|d| d.as_secs())
139        .unwrap_or(0)
140}
141
142fn is_kind_configured(kind: SecretKind) -> bool {
143    match kind {
144        SecretKind::Jwt => std::env::var("POOLAI_JWT_SECRET").is_ok(),
145        SecretKind::TlsCertificate => {
146            std::env::var("HTTPS_CERT_PATH").is_ok() || std::env::var("HTTPS_KEY_PATH").is_ok()
147        }
148        SecretKind::TelegramWebhook => std::env::var("POOLAI_TELEGRAM_WEBHOOK_SECRET").is_ok(),
149    }
150}
151
152static REGISTRY: OnceLock<Arc<Registry>> = OnceLock::new();
153static DEFAULT_HOOKS: OnceLock<()> = OnceLock::new();
154
155fn registry() -> &'static Arc<Registry> {
156    REGISTRY.get_or_init(|| Arc::new(Registry::new()))
157}
158
159/// Register a custom rotation hook for a secret kind.
160pub fn register_rotation_hook<F>(kind: SecretKind, hook: F)
161where
162    F: Fn() -> Result<(), String> + Send + Sync + 'static,
163{
164    registry().register(kind, Arc::new(hook));
165}
166
167/// Register TLS certificate reload (called from HTTPS server startup).
168pub fn register_tls_reload_hook<F>(hook: F)
169where
170    F: Fn() -> Result<(), String> + Send + Sync + 'static,
171{
172    registry().register(SecretKind::TlsCertificate, Arc::new(hook));
173}
174
175fn hook_reload_jwt() -> Result<(), String> {
176    jwt_secrets::reload_jwt_from_env()
177}
178
179fn hook_reload_telegram_webhook() -> Result<(), String> {
180    if std::env::var("POOLAI_TELEGRAM_WEBHOOK_SECRET").is_err() {
181        return Err("POOLAI_TELEGRAM_WEBHOOK_SECRET not set".into());
182    }
183    Ok(())
184}
185
186/// Register built-in hooks (JWT env reload, Telegram env presence, TLS if registered).
187pub fn init_default_rotation_hooks() {
188    DEFAULT_HOOKS.get_or_init(|| {
189        register_rotation_hook(SecretKind::Jwt, hook_reload_jwt);
190        register_rotation_hook(SecretKind::TelegramWebhook, hook_reload_telegram_webhook);
191    });
192}
193
194/// Run all hooks for a secret kind.
195pub fn run_rotation(kind: SecretKind) -> Result<RotationRunReport, AppError> {
196    init_default_rotation_hooks();
197    registry().run(kind)
198}
199
200/// Status for admin/ops (no secret values).
201pub fn rotation_status() -> Vec<RotationStatusEntry> {
202    init_default_rotation_hooks();
203    registry().status()
204}

206#[derive(Debug, Clone, Serialize)]
207pub struct RotationStatusEntry {
208    pub kind: SecretKind,
209    pub configured: bool,
210    pub last_rotated_unix: Option<u64>,
211    pub rotation_count: u64,
212    pub grace_active: bool,

216#[derive(Debug, Clone, Serialize)]
217pub struct RotationRunReport {
218    pub kind: SecretKind,
219    pub success: bool,
220    pub hooks: Vec<HookResult>,

223#[derive(Debug, Clone, Serialize)]
224pub struct HookResult {

231/// Background poll: reload JWT secrets when `POOLAI_SECRET_ROTATION_POLL_SECS` is set.
232pub fn spawn_jwt_env_poll_if_configured() {
233    let Some(secs) = std::env::var("POOLAI_SECRET_ROTATION_POLL_SECS")
234        .ok()
235        .and_then(|s| s.parse::<u64>().ok())
236        .filter(|&n| n > 0)
237    else {
238        return;
239    };
240    init_default_rotation_hooks();
241    info!(
242        interval_secs = secs,
243        "JWT secret env poll enabled (POOLAI_SECRET_ROTATION_POLL_SECS)"
244    );
245    tokio::spawn(async move {
246        let mut interval = tokio::time::interval(std::time::Duration::from_secs(secs));
247        interval.tick().await;
248        loop {
249            interval.tick().await;
250            if let Err(e) = hook_reload_jwt() {
251                warn!(error = %e, "JWT secret env poll reload failed");
252            }
253        }
254    });
255}

261    #[test]
262    fn secret_kind_parse_roundtrip() {
263        assert_eq!(SecretKind::from_str("jwt"), Some(SecretKind::Jwt));
264        assert_eq!(
265            SecretKind::from_str("tls_certificate"),
266            Some(SecretKind::TlsCertificate)

270    #[test]
271    fn jwt_rotation_hook_updates_store() {
272        init_default_rotation_hooks();
273        let before = jwt_secrets::jwt_store().read().loaded_at_unix;
274        run_rotation(SecretKind::Jwt).expect("jwt rotation");
275        let after = jwt_secrets::jwt_store().read().loaded_at_unix;
276        assert!(after >= before);
277    }
