1//! Build distributed RAID PutArtifact probe messages for virtual-node workers (FM-016+++).
2
3use crate::raid::protocol::{ArtifactMetadata, ProtocolMessage, PutArtifactPayload, SyncMode};
4use base64::{engine::general_purpose::STANDARD as B64, Engine};
5use chrono::Utc;
6use serde_json::Value;
7use uuid::Uuid;
8
9const DEFAULT_PROBE_BYTES: &[u8] = b"poolai-vn-artifact-probe";
10
11/// Logical name for a probe task payload.
12pub fn probe_logical_name(task_payload: &Value) -> &str {
13    task_payload
14        .get("name")
15        .and_then(|v| v.as_str())
16        .unwrap_or("vn-probe")
17}
18
19/// Bytes to upload for a probe (from payload or default).
20pub fn probe_bytes(task_payload: &Value) -> Result<Vec<u8>, String> {
21    task_payload
22        .get("data_b64")
23        .and_then(|v| v.as_str())
24        .map(|s| B64.decode(s).map_err(|e| e.to_string()))
25        .transpose()
26        .map(|opt| opt.unwrap_or_else(|| DEFAULT_PROBE_BYTES.to_vec()))
27}
28
29/// Small PutArtifact wire message for coordinator RAID replicate endpoint.
30pub fn build_probe_message(
31    worker_id: &str,
32    task_payload: &Value,
33) -> Result<ProtocolMessage, String> {
34    let logical_name = probe_logical_name(task_payload);
35    let bytes = probe_bytes(task_payload)?;
36
37    let metadata = ArtifactMetadata {
38        name: logical_name.to_string(),
39        version: "probe-1".to_string(),
40        size_bytes: bytes.len() as u64,
41        checksum: format!("probe-{}", bytes.len()),
42        created_at: Utc::now(),
43        content_type: Some("application/octet-stream".to_string()),
44        tags: Some(vec!["virtual_node".to_string(), "probe".to_string()]),
45    };
46    let payload = PutArtifactPayload {
47        artifact_id: format!("vn-probe-{}", Uuid::new_v4()),
48        source_node: worker_id.to_string(),
49        data: Some(B64.encode(&bytes)),
50        metadata,
51        replication_factor: 1,
52        sync_mode: SyncMode::Async,
53    };
54    ProtocolMessage::put_artifact(worker_id.to_string(), payload).map_err(|e| e.to_string())
55}

63        let msg = build_probe_message("vn-a", &Value::Null).expect("message");
