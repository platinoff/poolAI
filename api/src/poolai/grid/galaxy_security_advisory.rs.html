1//! Security advisory acknowledge audit wire (PH-S573, Galaxy §9.6).
2//! Read-only advisory list stub (PH-S586).

4use serde::{Deserialize, Serialize};
5use std::collections::HashSet;
6use std::sync::atomic::{AtomicU64, Ordering};
7use std::sync::{LazyLock, Mutex};
8
9/// Metric: operator security advisory acknowledgements (PH-S573).
10pub const METRIC_ADVISORY_ACKNOWLEDGED_TOTAL: &str = "poolai_advisory_acknowledged_total";
11
12/// Operator-facing advisory row (PH-S586 list stub).
13#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
14pub struct SecurityAdvisoryEntry {
15    pub id: String,
16    pub severity: String,
17    pub summary: String,
18    pub acknowledged: bool,
19}
20
21static ADVISORY_ACK_TOTAL: AtomicU64 = AtomicU64::new(0);
22
23static ACKNOWLEDGED_IDS: LazyLock<Mutex<HashSet<String>>> =
24    LazyLock::new(|| Mutex::new(HashSet::new()));
25
26/// Canonical stub advisories for ops UI (Galaxy §9.6); acknowledgement state is live.
27pub fn list_security_advisories() -> Vec<SecurityAdvisoryEntry> {
28    [
29        (
30            "CVE-2026-0001",
31            "medium",
32            "Signed release manifest rotation advisory (Galaxy §9.2)",
33        ),
34        (
35            "key_transition-2026-q2",
36            "low",
37            "Maintainer trust-root key transition window (Galaxy §9.3)",
38        ),
39        (
40            "protocol_sunset-1.0",
41            "high",
42            "Protocol 1.0 sunset — upgrade workers to 1.2.x (Galaxy §9.3)",
43        ),
44    ]
45    .into_iter()
46    .map(|(id, severity, summary)| SecurityAdvisoryEntry {
47        id: id.to_string(),
48        severity: severity.to_string(),
49        summary: summary.to_string(),
50        acknowledged: is_advisory_acknowledged(id),
51    })
52    .collect()
53}
54
55pub fn advisory_acknowledged_total() -> u64 {
56    ADVISORY_ACK_TOTAL.load(Ordering::Relaxed)
57}
58
59/// Record advisory acknowledgement; returns false when already acknowledged.
60pub fn acknowledge_security_advisory(advisory_id: &str) -> bool {
61    let id = advisory_id.trim();
62    if id.is_empty() {
63        return false;
64    }
65    let mut guard = ACKNOWLEDGED_IDS.lock().unwrap_or_else(|e| e.into_inner());
66    if !guard.insert(id.to_string()) {
67        return false;
68    }
69    ADVISORY_ACK_TOTAL.fetch_add(1, Ordering::Relaxed);
70    true
71}
72
73pub fn is_advisory_acknowledged(advisory_id: &str) -> bool {
74    ACKNOWLEDGED_IDS
75        .lock()
76        .ok()
77        .is_some_and(|g| g.contains(advisory_id.trim()))
78}

80#[cfg(any(test, feature = "test-utils"))]
81pub fn reset_security_advisory_for_test() {
82    ADVISORY_ACK_TOTAL.store(0, Ordering::Relaxed);
83    if let Ok(mut g) = ACKNOWLEDGED_IDS.lock() {
84        g.clear();
85    }
86}

88#[cfg(test)]
89mod tests {
90    use super::*;
91
92    #[test]
93    fn advisory_list_ph_s586() {
94        reset_security_advisory_for_test();
95        let rows = list_security_advisories();
96        assert_eq!(rows.len(), 3);
97        assert!(rows.iter().all(|r| !r.acknowledged));
98        assert!(acknowledge_security_advisory("CVE-2026-0001"));
99        let rows = list_security_advisories();
100        assert!(rows
101            .iter()
102            .find(|r| r.id == "CVE-2026-0001")
103            .is_some_and(|r| r.acknowledged));
104        reset_security_advisory_for_test();
105    }
106
107    #[test]
108    fn advisory_ack_ph_s573() {
109        reset_security_advisory_for_test();
110        assert!(acknowledge_security_advisory("CVE-2026-0001"));
111        assert_eq!(advisory_acknowledged_total(), 1);
112        assert!(!acknowledge_security_advisory("CVE-2026-0001"));
113        assert!(is_advisory_acknowledged("CVE-2026-0001"));
114        reset_security_advisory_for_test();
115    }
