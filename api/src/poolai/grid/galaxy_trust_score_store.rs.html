14fn trust_map() -> &'static Mutex<HashMap<String, TrustScore>> {
15    static TRUST_BY_PEER: OnceLock<Mutex<HashMap<String, TrustScore>>> = OnceLock::new();
16    TRUST_BY_PEER.get_or_init(|| Mutex::new(HashMap::new()))
17}
18
19static DISK_LOADED: OnceLock<Mutex<bool>> = OnceLock::new();
20
21fn disk_loaded_flag() -> &'static Mutex<bool> {
22    DISK_LOADED.get_or_init(|| Mutex::new(false))
23}

25/// Trust score persistence backend (PH-S910).
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub enum TrustStoreBackend {
28    Ephemeral,
29    Json,
30    Sqlite,
31}
32
33/// Env: optional JSON file path for trust score persistence (legacy single-file).
34pub const ENV_TRUST_SCORE_STORE_PATH: &str = "POOLAI_TRUST_SCORE_STORE_PATH";
35
36fn json_store_path() -> Option<PathBuf> {
37    std::env::var(ENV_TRUST_SCORE_STORE_PATH)
38        .ok()
39        .map(|v| v.trim().to_string())
40        .filter(|v| !v.is_empty())
41        .map(PathBuf::from)
42}
43
44/// Active trust store backend from env (PH-S910).
45pub fn current_trust_store_backend() -> TrustStoreBackend {
46    if galaxy_trust_score_store_sqlite::sqlite_enabled()
47        && galaxy_trust_score_store_sqlite::data_dir_from_env().is_some()
48    {
49        TrustStoreBackend::Sqlite
50    } else if json_store_path().is_some() {
51        TrustStoreBackend::Json
52    } else {
53        TrustStoreBackend::Ephemeral
54    }
55}

57/// Wire label for admin / trust-metrics depth fields (PH-S913).
58pub fn trust_store_backend_wire_label(backend: TrustStoreBackend) -> &'static str {
59    match backend {
60        TrustStoreBackend::Ephemeral => "ephemeral",
61        TrustStoreBackend::Json => "json",
62        TrustStoreBackend::Sqlite => "sqlite",
63    }
64}

66/// Count of peers with stored trust scores in the in-process map (PH-S914).
67pub fn persisted_trust_peer_count() -> u32 {
68    trust_map()
69        .lock()
70        .ok()
71        .map(|map| map.len() as u32)
72        .unwrap_or(0)
73}

75fn flush_to_disk(map: &HashMap<String, TrustScore>) {
76    match current_trust_store_backend() {
77        TrustStoreBackend::Ephemeral => {}
78        TrustStoreBackend::Json => {
79            let Some(path) = json_store_path() else {
80                return;
81            };
82            if let Some(parent) = path.parent() {
83                let _ = fs::create_dir_all(parent);
84            }
85            if let Ok(json) = serde_json::to_string_pretty(map) {
86                let _ = fs::write(path, json);
87            }
88        }
89        TrustStoreBackend::Sqlite => {
90            if let Some(dir) = galaxy_trust_score_store_sqlite::data_dir_from_env() {
91                let _ = galaxy_trust_score_store_sqlite::persist(&dir, map);
92            }
93        }
94    }
95}
96
97fn load_from_disk_once() {
98    let mut loaded = disk_loaded_flag().lock().unwrap_or_else(|e| e.into_inner());
99    if *loaded {
100        return;
101    }
102    match current_trust_store_backend() {
103        TrustStoreBackend::Ephemeral => {}
104        TrustStoreBackend::Json => {
105            if let Some(path) = json_store_path() {
106                if Path::new(&path).exists() {
107                    if let Ok(text) = fs::read_to_string(path) {
108                        if let Ok(map) = serde_json::from_str::<HashMap<String, TrustScore>>(&text)
109                        {
110                            if let Ok(mut guard) = trust_map().lock() {
111                                *guard = map;
112                            }
113                        }
114                    }
115                }
116            }
117        }
118        TrustStoreBackend::Sqlite => {
119            if let Some(dir) = galaxy_trust_score_store_sqlite::data_dir_from_env() {
120                if let Some(legacy) = json_store_path() {
121                    let _ = migrate_legacy_json_file(&dir, &legacy);
122                }
123                if let Ok(map) = galaxy_trust_score_store_sqlite::load(&dir) {
124                    if let Ok(mut guard) = trust_map().lock() {
125                        *guard = map;
126                    }
127                }
128            }
129        }
130    }
131    *loaded = true;
132}

134/// Persist peer trust score after grid result ingest.
135pub fn persist_peer_trust_score(peer_id: &str, score: TrustScore) {
136    if peer_id.trim().is_empty() {
137        return;
138    }
139    load_from_disk_once();
140    if let Ok(mut map) = trust_map().lock() {
141        map.insert(peer_id.trim().to_string(), score);
142        flush_to_disk(&map);
143    }
144}

146/// Hydrate register-remote metadata with stored trust score when absent.
147pub fn hydrate_register_metadata_trust_score(
148    peer_id: &str,
149    metadata: &mut HashMap<String, String>,
150) {
151    load_from_disk_once();
152    if metadata.contains_key("trust_score") {
153        return;
154    }
155    if let Ok(map) = trust_map().lock() {
156        if let Some(score) = map.get(peer_id) {
157            metadata.insert("trust_score".to_string(), score.to_string());
158        }
159    }
160}

162/// Apply trust delta to stored peer score (or default) and persist (PH-S610 / PH-S611).
163pub fn apply_peer_trust_delta(peer_id: &str, delta: i16) -> TrustScore {
164    let peer = peer_id.trim();
165    if peer.is_empty() {
166        return DEFAULT_TRUST_SCORE;
167    }
168    load_from_disk_once();
169    let current = lookup_peer_trust_score(peer).unwrap_or(DEFAULT_TRUST_SCORE);
170    let adjusted = apply_trust_delta(current, delta);
171    persist_peer_trust_score(peer, adjusted);
172    adjusted
173}

175/// Trust delta on stale-epoch grid result reject (PH-S610).
176pub fn apply_lease_epoch_rejected_trust_delta(peer_id: &str) -> TrustScore {
177    apply_peer_trust_delta(peer_id, TRUST_DELTA_LEASE_EPOCH_REJECTED)
178}
179
180/// Trust delta when worker newly marked unhealthy (PH-S611).
181pub fn apply_worker_unhealthy_trust_delta(peer_id: &str) -> TrustScore {
182    apply_peer_trust_delta(peer_id, TRUST_DELTA_WORKER_UNHEALTHY)
183}
184
185/// Lookup stored trust score for a peer (tests / admin / payout gate PH-S911).
186pub fn lookup_peer_trust_score(peer_id: &str) -> Option<TrustScore> {
187    load_from_disk_once();
188    trust_map()
189        .lock()
190        .ok()
191        .and_then(|map| map.get(peer_id).copied())
192}

194#[cfg(any(test, feature = "test-utils"))]
195pub fn reset_trust_score_store_for_test() {
196    if let Ok(mut map) = trust_map().lock() {
197        map.clear();
198    }
199    if let Ok(mut loaded) = disk_loaded_flag().lock() {
200        *loaded = false;
201    }
202}

210    fn trust_score_persist_roundtrip_ph_s552() {
211        static ENV_LOCK: std::sync::Mutex<()> = std::sync::Mutex::new(());
212        let _guard = ENV_LOCK.lock().unwrap_or_else(|e| e.into_inner());
213        reset_trust_score_store_for_test();
214        std::env::remove_var(galaxy_trust_score_store_sqlite::ENV_TRUST_SCORE_STORE);
215        std::env::remove_var(galaxy_trust_score_store_sqlite::ENV_TRUST_SCORE_DATA_DIR);
216        let tmp = TempDir::new().expect("tempdir");
217        let path = tmp.path().join("trust_scores.json");
218        std::env::set_var(ENV_TRUST_SCORE_STORE_PATH, path.to_string_lossy().as_ref());
219        persist_peer_trust_score("tg-edge-1", 72);
220        reset_trust_score_store_for_test();
221        let mut meta = HashMap::new();
222        hydrate_register_metadata_trust_score("tg-edge-1", &mut meta);
223        assert_eq!(meta.get("trust_score").map(String::as_str), Some("72"));
224        std::env::remove_var(ENV_TRUST_SCORE_STORE_PATH);
225        reset_trust_score_store_for_test();
226    }
