6/// Pricing oracle wire depth (Galaxy §4.2).
7#[derive(Debug, Clone, Copy, PartialEq, Eq)]
8pub enum PricingDepth {
9    None,
10    L2Fallback,
11    L1Cache,
12    LiveFetch,
13    FullProduction,
14}
15
16/// Classify pricing depth from metrics snapshot + configured timeout (PH-S904).
17pub fn pricing_depth_stub(
18    snapshot: Option<&PricingMetricsSnapshot>,
19    provider_timeout_ms: u64,
20) -> PricingDepth {
21    let Some(s) = snapshot else {
22        return PricingDepth::None;
23    };
24    let has_l2 = s.forced_fallback_total > 0;
25    let has_l1 = s.fresh_served_total > 0 || s.stale_served_total > 0;
26    let has_live = s.provider_catalog_lookups_total > 0 || s.provider_errors_total > 0;
27    let timeout_configured =
28        provider_timeout_ms >= crate::grid::galaxy_pricing_oracle::MIN_PROVIDER_HTTP_TIMEOUT_MS;
29
30    if has_l2 && has_l1 && has_live && timeout_configured {
31        PricingDepth::FullProduction
32    } else if has_live {
33        PricingDepth::LiveFetch
34    } else if has_l1 {
35        PricingDepth::L1Cache
36    } else if has_l2 {
37        PricingDepth::L2Fallback
38    } else {
39        PricingDepth::None
40    }
41}
42
43/// Wire label for pricing-metrics / stand smoke (PH-S904).
44pub fn pricing_depth_wire_label(depth: PricingDepth) -> &'static str {
45    match depth {
46        PricingDepth::None => "none",
47        PricingDepth::L2Fallback => "l2_fallback",
48        PricingDepth::L1Cache => "l1_cache",
49        PricingDepth::LiveFetch => "live_fetch",
50        PricingDepth::FullProduction => "full_production",
51    }
52}
53
54/// Runtime pricing depth from in-process counters.
55pub fn current_pricing_depth() -> PricingDepth {
56    pricing_depth_stub(
57        Some(&crate::grid::galaxy_pricing_metrics::pricing_metrics_snapshot()),
58        provider_http_timeout_ms_from_env(),
59    )
60}

62#[cfg(test)]
63mod tests {
64    use super::*;
65    use crate::grid::galaxy_pricing_metrics::pricing_metrics_snapshot;
66    use crate::grid::galaxy_pricing_oracle::{
67        bump_forced_fallback_for_test, bump_fresh_served_for_test,
68        reset_forced_fallback_total_for_test, reset_fresh_served_total_for_test,
69    };
70    use crate::grid::galaxy_pricing_provider_metrics::{
71        record_provider_catalog_lookup, reset_provider_catalog_metrics_for_test,
72    };
73
74    #[test]
75    fn pricing_depth_stub_ph_s904() {
76        reset_fresh_served_total_for_test();
77        reset_forced_fallback_total_for_test();
78        reset_provider_catalog_metrics_for_test();
79
80        let empty = pricing_metrics_snapshot();
81        assert_eq!(pricing_depth_stub(None, 1500), PricingDepth::None);
82        assert_eq!(pricing_depth_stub(Some(&empty), 1500), PricingDepth::None);
83
84        bump_forced_fallback_for_test();
85        let l2 = pricing_metrics_snapshot();
86        assert_eq!(
87            pricing_depth_stub(Some(&l2), 1500),
88            PricingDepth::L2Fallback
89        );
90        reset_forced_fallback_total_for_test();
91
92        bump_fresh_served_for_test();
93        let l1 = pricing_metrics_snapshot();
94        assert_eq!(pricing_depth_stub(Some(&l1), 1500), PricingDepth::L1Cache);
95        reset_fresh_served_total_for_test();
96
97        record_provider_catalog_lookup(1);
98        let live = pricing_metrics_snapshot();
99        assert_eq!(
100            pricing_depth_stub(Some(&live), 1500),
101            PricingDepth::LiveFetch
102        );
103        reset_provider_catalog_metrics_for_test();
104
105        bump_forced_fallback_for_test();
106        bump_fresh_served_for_test();
107        record_provider_catalog_lookup(1);
108        let full = pricing_metrics_snapshot();
109        assert_eq!(
110            pricing_depth_stub(Some(&full), 1500),
111            PricingDepth::FullProduction
112        );
113        assert_eq!(
114            pricing_depth_wire_label(PricingDepth::FullProduction),
115            "full_production"
116        );
117
118        reset_fresh_served_total_for_test();
119        reset_forced_fallback_total_for_test();
120        reset_provider_catalog_metrics_for_test();
121    }
