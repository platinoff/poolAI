1//! Galaxy governance ops Prometheus stubs (PH-S528, Galaxy §9.8).
2
3use std::sync::atomic::{AtomicU64, Ordering};
4
5/// Successful `poolai-verify-release` runs (PH-S528).
6pub const METRIC_RELEASE_VERIFY_TOTAL: &str = "poolai_release_verify_total";
7
8/// Failed `poolai-verify-release` runs (PH-S528).
9pub const METRIC_RELEASE_VERIFY_FAIL_TOTAL: &str = "poolai_release_verify_fail_total";
10
11/// Pending opt-in update notifications (Galaxy §9.8 stub gauge).
12pub const METRIC_UPDATE_NOTIFY_PENDING: &str = "poolai_update_notify_pending";
13
14static RELEASE_VERIFY_TOTAL: AtomicU64 = AtomicU64::new(0);
15static RELEASE_VERIFY_FAIL_TOTAL: AtomicU64 = AtomicU64::new(0);
16static UPDATE_NOTIFY_PENDING: AtomicU64 = AtomicU64::new(0);
17
18pub fn release_verify_total() -> u64 {
19    RELEASE_VERIFY_TOTAL.load(Ordering::Relaxed)
20}
21
22pub fn release_verify_fail_total() -> u64 {
23    RELEASE_VERIFY_FAIL_TOTAL.load(Ordering::Relaxed)
24}
25
26pub fn update_notify_pending() -> u64 {
27    UPDATE_NOTIFY_PENDING.load(Ordering::Relaxed)
28}
29
30pub fn record_release_verify_success() {
31    RELEASE_VERIFY_TOTAL.fetch_add(1, Ordering::Relaxed);
32}
33
34pub fn record_release_verify_fail() {
35    RELEASE_VERIFY_FAIL_TOTAL.fetch_add(1, Ordering::Relaxed);
36}
37
38pub fn set_update_notify_pending(count: u64) {
39    UPDATE_NOTIFY_PENDING.store(count, Ordering::Relaxed);
40}
41
42/// Read-only governance ops snapshot for `GET /api/v1/grid/governance-metrics` (PH-S791).
43#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize)]
44pub struct GovernanceMetricsSnapshot {
45    pub release_verify_total: u64,
46    pub release_verify_fail_total: u64,
47    pub update_notify_pending: u64,
48    pub advisory_acknowledged_total: u64,
49}
50
51/// Coordinator governance metrics snapshot (PH-S791).
52pub fn governance_metrics_snapshot() -> GovernanceMetricsSnapshot {
53    GovernanceMetricsSnapshot {
54        release_verify_total: release_verify_total(),
55        release_verify_fail_total: release_verify_fail_total(),
56        update_notify_pending: update_notify_pending(),
57        advisory_acknowledged_total:
58            crate::grid::galaxy_security_advisory::advisory_acknowledged_total(),
59    }
60}

62#[cfg(any(test, feature = "test-utils"))]
63pub fn reset_governance_metrics_for_test() {
64    RELEASE_VERIFY_TOTAL.store(0, Ordering::Relaxed);
65    RELEASE_VERIFY_FAIL_TOTAL.store(0, Ordering::Relaxed);
66    UPDATE_NOTIFY_PENDING.store(0, Ordering::Relaxed);
67}
