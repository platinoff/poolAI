1//! Shadow verification checker job submit (PH-S534, Galaxy §6.2).
2//!
3//! On `SampleScheduled`, enqueue a trusted `local_srv` shadow job in [`JobStore`].
4
5use chrono::Utc;
6
7use crate::core::error::AppError;
8use crate::grid::galaxy_verification_metrics::{
9    enqueue_verification_checker_task, record_verification_checker_enqueue,
10};
11use crate::job::{JobId, JobKind, JobRecord, JobSpec, JobStatus, JobStore};
12
13/// Trusted local worker id for shadow verification checks (Galaxy §6.2).
14pub const SHADOW_CHECKER_WORKER_ID: &str = "local_srv";
15
16/// Verification policy marker on shadow checker jobs.
17pub const SHADOW_CHECKER_POLICY: &str = "shadow_checker";
18
19/// Metric: shadow checker jobs submitted to JobStore (PH-S534).
20pub const METRIC_VERIFICATION_CHECKER_JOB_SUBMIT_TOTAL: &str =
21    "galaxy_verification_checker_job_submit_total";
22
23static CHECKER_JOB_SUBMIT_TOTAL: std::sync::atomic::AtomicU64 =
24    std::sync::atomic::AtomicU64::new(0);
25
26pub fn verification_checker_job_submit_total() -> u64 {
27    CHECKER_JOB_SUBMIT_TOTAL.load(std::sync::atomic::Ordering::Relaxed)
28}
29
30#[cfg(any(test, feature = "test-utils"))]
31pub fn reset_verification_checker_job_submit_for_test() {
32    CHECKER_JOB_SUBMIT_TOTAL.store(0, std::sync::atomic::Ordering::Relaxed);
33}
34
35/// Shadow checker job id derived from the primary edge job.
36pub fn shadow_checker_job_id(primary_job_id: &str) -> String {
37    format!("{primary_job_id}-shadow-check")
38}
39
40/// Submit shadow verification checker job to JobStore when sample is scheduled.
41pub fn submit_shadow_verification_checker_job(
42    jobs: &JobStore,
43    primary_job_id: &str,
44) -> Result<String, AppError> {
45    let shadow_id = shadow_checker_job_id(primary_job_id);
46    if jobs.get(&shadow_id)?.is_some() {
47        return Ok(shadow_id);
48    }
49    let record = JobRecord {
50        spec: JobSpec {
51            id: JobId::new(shadow_id.clone()),
52            kind: JobKind::System,
53            resources: Default::default(),
54            priority: 0,
55            max_duration_secs: None,
56            input_artifact_ids: vec![],
57            verification_policy: Some(SHADOW_CHECKER_POLICY.into()),
58            deadline: None,
59        },
60        status: JobStatus::Scheduled,
61        created_at: Utc::now(),
62        worker_id: Some(SHADOW_CHECKER_WORKER_ID.into()),
63        vm_id: None,
64        lease_owner: None,
65        lease_epoch: None,
66        lease_expires_at: None,
67        migration_count: None,
68        fail_reason: None,
69        leased_at: None,
70    };
71    jobs.push(record)?;
72    enqueue_verification_checker_task(primary_job_id);
73    record_verification_checker_enqueue();
74    CHECKER_JOB_SUBMIT_TOTAL.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
75    Ok(shadow_id)
76}

78#[cfg(test)]
79mod tests {
80    use super::*;
81    use crate::job::JobStore;
82
83    #[test]
84    fn submit_shadow_checker_job_ph_s534() {
85        reset_verification_checker_job_submit_for_test();
86        let jobs = JobStore::open_for_test(None);
87        let id = submit_shadow_verification_checker_job(&jobs, "job-edge-1").expect("submit");
88        assert_eq!(id, "job-edge-1-shadow-check");
89        let row = jobs.get(&id).expect("get").expect("row");
90        assert_eq!(row.status, JobStatus::Scheduled);
91        assert_eq!(row.worker_id.as_deref(), Some(SHADOW_CHECKER_WORKER_ID));
92        assert_eq!(
93            row.spec.verification_policy.as_deref(),
94            Some(SHADOW_CHECKER_POLICY)
95        );
96        assert_eq!(verification_checker_job_submit_total(), 1);
97        reset_verification_checker_job_submit_for_test();
98    }
99}
