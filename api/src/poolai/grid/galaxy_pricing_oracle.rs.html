368impl GalaxyPricingProviderCatalog {
369    pub fn is_empty(&self) -> bool {
370        self.providers.is_empty()
371    }

373    /// Enabled US-region rows matching task/model filters (empty filter = wildcard).
374    pub fn matching_entries(
375        &self,
376        task_profile: &str,
377        model_profile: &str,
378    ) -> Vec<&GalaxyPricingProviderEntry> {
379        let matched: Vec<_> = self
380            .providers
381            .iter()
382            .filter(|p| {
383                p.enabled
384                    && p.region.eq_ignore_ascii_case("us")
385                    && provider_matches_profiles(p, task_profile, model_profile)
386            })
387            .collect();
388        crate::grid::galaxy_pricing_provider_metrics::record_provider_catalog_lookup(matched.len());
389        matched
390    }
391}

393fn provider_matches_profiles(
394    entry: &GalaxyPricingProviderEntry,
395    task_profile: &str,
396    model_profile: &str,
397) -> bool {
398    let task_ok =
399        entry.task_profiles.is_empty() || entry.task_profiles.iter().any(|t| t == task_profile);
400    let model_ok = entry
401        .model_profile
402        .as_deref()
403        .is_none_or(|m| m.is_empty() || m == model_profile);
404    task_ok && model_ok
405}

417fn sanitize_provider_entry(
418    mut entry: GalaxyPricingProviderEntry,
419) -> Option<GalaxyPricingProviderEntry> {
420    if !entry.enabled || entry.provider_id.trim().is_empty() {
421        return None;
422    }
423    entry.provider_id = entry.provider_id.trim().to_string();
424    entry.region = entry.region.trim().to_string();
425    if entry.region.is_empty() {
426        entry.region = default_provider_region();
427    }
428    entry.units = normalize_provider_units(entry.units);
429    if entry.units.is_empty() {
430        return None;
431    }
432    if let Some(endpoint) = entry.endpoint.as_mut() {
433        let trimmed = endpoint.trim();
434        if trimmed.is_empty() {
435            entry.endpoint = None;
436        } else {
437            *endpoint = trimmed.to_string();
438        }
439    }
440    Some(entry)
441}

443/// Parse `POOLAI_GALAXY_PRICING_PROVIDERS` JSON (array or `{"providers":[...]}`).
444pub fn parse_pricing_providers_json(raw: &str) -> GalaxyPricingProviderCatalog {
445    let trimmed = raw.trim();
446    if trimmed.is_empty() {
447        return GalaxyPricingProviderCatalog::default();
448    }
449    if let Ok(rows) = serde_json::from_str::<Vec<GalaxyPricingProviderEntry>>(trimmed) {
450        return catalog_from_rows(rows);
451    }
452    if let Ok(wrapped) = serde_json::from_str::<GalaxyPricingProviderCatalog>(trimmed) {
453        return catalog_from_rows(wrapped.providers);
454    }
455    GalaxyPricingProviderCatalog::default()
456}

458fn catalog_from_rows(rows: Vec<GalaxyPricingProviderEntry>) -> GalaxyPricingProviderCatalog {
459    let providers = rows
460        .into_iter()
461        .filter_map(sanitize_provider_entry)
462        .collect();
463    GalaxyPricingProviderCatalog { providers }

552/// PH-S102: live provider HTTP fetch for a single unit key.
553pub async fn fetch_live_provider_quotes(
554    catalog: &GalaxyPricingProviderCatalog,
555    task_profile: &str,
556    model_profile: &str,
557    unit_key: GalaxyPriceUnitKey,
558    timeout_ms: u64,
559) -> Vec<MockProviderQuote> {
560    let timeout = Duration::from_millis(timeout_ms);
561    let connect_timeout = Duration::from_millis(timeout_ms.min(5_000));
562    let Ok(client) = reqwest::Client::builder()
563        .connect_timeout(connect_timeout)
564        .timeout(timeout)
565        .build()
566    else {
567        return Vec::new();
568    };
569    let mut out = Vec::new();
570    for entry in catalog.matching_entries(task_profile, model_profile) {
571        let Some(endpoint) = entry.endpoint.as_deref() else {
572            continue;
573        };
574        let url = format!(
575            "{endpoint}?task_profile={task_profile}&model_profile={model_profile}&unit_key={}",
576            unit_key.as_str()
577        );
578        let resp = match client.get(url).send().await {
579            Ok(r) => r,
580            Err(e) => {
581                if e.is_timeout() {
582                    crate::grid::galaxy_pricing_provider_metrics::record_provider_fetch_timeout();
583                }
584                crate::grid::galaxy_pricing_provider_metrics::record_provider_fetch_error();
585                continue;
586            }
587        };
588        if !resp.status().is_success() {
589            crate::grid::galaxy_pricing_provider_metrics::record_provider_fetch_error();
590            continue;
591        }
592        let body = match resp.bytes().await {
593            Ok(bytes) => bytes,
594            Err(_) => {
595                crate::grid::galaxy_pricing_provider_metrics::record_provider_fetch_error();
596                continue;
597            }
598        };
599        let Some(units) = parse_live_units_map_bytes(body.as_ref()) else {
600            crate::grid::galaxy_pricing_provider_metrics::record_provider_fetch_error();
601            continue;
602        };
603        let Some(usd_micro) = units.get(unit_key.as_str()).copied() else {
604            crate::grid::galaxy_pricing_provider_metrics::record_provider_fetch_error();
605            continue;
606        };
607        if usd_micro == 0 {
608            crate::grid::galaxy_pricing_provider_metrics::record_provider_fetch_error();
609            continue;
610        }
611        out.push(MockProviderQuote {
612            provider_id: entry.provider_id.clone(),
613            unit_key,
614            usd_micro,
615            healthy: true,
616        });
617    }
618    out
619}

621fn parse_fallback_json(raw: &str) -> HashMap<GalaxyPriceUnitKey, u64> {
622    let parsed = serde_json::from_str::<HashMap<String, u64>>(raw);
623    let mut out = HashMap::new();
624    if let Ok(map) = parsed {
625        for (k, v) in map {
626            if let Ok(unit_key) = GalaxyPriceUnitKey::from_str(&k) {
627                if v > 0 {
628                    out.insert(unit_key, v);
629                }
630            }
631        }
632    }
633    out

704/// L1 cache TTL metadata for API/ops (§4.2.3, PH-S89).

708    pub cache_age_secs: u64,

717/// Build TTL metadata for an L1 cache hit (pairs with [`cache_freshness`]).
718pub fn cache_metadata(
719    now_secs: u64,
720    cached_at_secs: u64,

724    GalaxyPricingCacheMetadata {
725        cache_age_secs,
726        cache_ttl_secs: config.cache_ttl_secs,
727        max_stale_secs: config.max_stale_secs,
728        cache_fresh_until_secs: cached_at_secs.saturating_add(config.cache_ttl_secs),
729        cache_stale_until_secs: cached_at_secs.saturating_add(config.max_stale_secs),
730    }

733/// Classify cache entry age (pure; inject `now_secs` in tests).
734pub fn cache_freshness(
735    now_secs: u64,
736    cached_at_secs: u64,
737    cache_ttl_secs: u64,
738    max_stale_secs: u64,
739) -> CacheFreshness {
740    let age = now_secs.saturating_sub(cached_at_secs);
741    if age <= cache_ttl_secs {
742        CacheFreshness::Fresh
743    } else if age <= max_stale_secs {
744        CacheFreshness::Stale
745    } else {
746        CacheFreshness::Expired
747    }
748}

872    /// Lookup fresh/stale cache, provider refresh, or L2 fallback (§4.2.3–4.2.4).
873    pub fn try_quote(
874        &mut self,
875        now_secs: u64,
876        key: GalaxyPricingCacheKey,
877        providers: &[MockProviderQuote],
878    ) -> Result<GalaxyPricingQuote, GalaxyPricingUnavailable> {
879        if self.config.force_fallback {
880            let quote = self
881                .l2_fallback_quote(now_secs, key)
882                .ok_or(GalaxyPricingUnavailable)?;
883            record_forced_fallback(quote.unit_key);
884            observe_served_quote_metrics(&quote);
885            return Ok(quote);
886        }
887        if let Some((entry, freshness)) = self.lookup(now_secs, &key) {
888            if freshness == CacheFreshness::Fresh || freshness == CacheFreshness::Stale {
889                observe_l1_cache_age_secs(now_secs.saturating_sub(entry.quote.cached_at_secs));
890                let quote = serve_l1_cache_quote(entry, freshness);
891                observe_served_quote_metrics(&quote);
892                return Ok(quote);
893            }
894        }
895        let quote = self
896            .refresh_from_providers(now_secs, key.clone(), providers)
897            .or_else(|| self.l2_fallback_quote(now_secs, key))
898            .ok_or(GalaxyPricingUnavailable)?;
899        observe_served_quote_metrics(&quote);
900        Ok(quote)
901    }
