10/// Wire DTO for signed capability documents (Galaxy §6.6 roadmap stub).
11#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
12pub struct GalaxyCapabilityDocument {
13    pub peer_id: String,
14    #[serde(default)]
15    pub capabilities: Vec<String>,
16    #[serde(skip_serializing_if = "Option::is_none")]
17    pub signature: Option<String>,
18    #[serde(skip_serializing_if = "Option::is_none")]
19    pub expires_at: Option<String>,
20    /// TEE attestation blob (Galaxy §6.6 roadmap, PH-S572).
21    #[serde(skip_serializing_if = "Option::is_none")]
22    pub tee_attestation: Option<String>,
23}
24
25/// Parse/validation failure for capability documents.
26#[derive(Debug, Clone, PartialEq, Eq)]
27pub struct CapabilityDocParseError {
28    pub message: String,
29}
30
31impl CapabilityDocParseError {
32    fn new(message: impl Into<String>) -> Self {
33        Self {
34            message: message.into(),
35        }
36    }
37}
38
39impl std::fmt::Display for CapabilityDocParseError {
40    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
41        write!(f, "{}", self.message)
42    }
43}
44
45/// Parse capability document from JSON value.
46pub fn parse_capability_document(
47    value: &serde_json::Value,
48) -> Result<GalaxyCapabilityDocument, CapabilityDocParseError> {
49    serde_json::from_value(value.clone())
50        .map_err(|e| CapabilityDocParseError::new(format!("capability document parse failed: {e}")))
51}

66/// Returns true when TEE attestation is required for telegram_edge GPU workers (PH-S572).
67pub fn tee_attestation_required() -> bool {
68    matches!(
69        std::env::var(ENV_TEE_ATTEST_REQUIRED)
70            .ok()
71            .map(|v| v.trim().to_ascii_lowercase())
72            .as_deref(),
73        Some("1") | Some("true") | Some("yes")
74    )
75}

77/// Reject telegram_edge `inference:gpu` without `tee_attestation` when env requires (PH-S572).
78pub fn check_tee_attestation_required(
79    doc: &GalaxyCapabilityDocument,
80) -> Result<(), CapabilityDocParseError> {
81    if !tee_attestation_required() {
82        return Ok(());
83    }
84    let needs_tee = doc.capabilities.iter().any(|c| {
85        let lower = c.trim().to_ascii_lowercase();
86        lower == "inference:gpu" || lower.contains("gpu")
87    });
88    if !needs_tee {
89        return Ok(());
90    }
91    let has_attest = doc
92        .tee_attestation
93        .as_ref()
94        .map(|s| !s.trim().is_empty())
95        .unwrap_or(false);
96    if has_attest {
97        Ok(())
98    } else {
99        Err(CapabilityDocParseError::new(
100            "tee_attestation required for inference:gpu when POOLAI_TEE_ATTEST_REQUIRED=1",
101        ))
102    }
103}

105/// Returns true when production capability verify key is configured (PH-S561).
106pub fn capability_verify_production_mode() -> bool {
107    std::env::var(ENV_CAPABILITY_VERIFY_KEY)
108        .ok()
109        .filter(|v| !v.trim().is_empty())
110        .is_some()
111        || std::env::var(ENV_CAPABILITY_VERIFY_PK_HEX)
112            .ok()
113            .filter(|v| !v.trim().is_empty())
114            .is_some()
115}

117fn capability_verify_pk_hex() -> String {
118    std::env::var(ENV_CAPABILITY_VERIFY_KEY)
119        .ok()
120        .filter(|v| !v.trim().is_empty())
121        .or_else(|| std::env::var(ENV_CAPABILITY_VERIFY_PK_HEX).ok())
122        .filter(|v| !v.trim().is_empty())
123        .unwrap_or_else(|| DEV_CAPABILITY_VERIFY_PK_HEX.to_string())
124}

126/// Canonical signing message for capability documents (PH-S466 stub).
127pub fn capability_signing_message(doc: &GalaxyCapabilityDocument) -> String {
128    format!("{}:{}", doc.peer_id.trim(), doc.capabilities.join(","))
129}

131/// Ed25519 signature verify stub for dev fixtures (PH-S466).
132pub fn verify_capability_signature_stub(
133    doc: &GalaxyCapabilityDocument,
134) -> Result<(), CapabilityDocParseError> {
135    let Some(sig_hex) = doc.signature.as_ref() else {
136        return Ok(());
137    };
138    if sig_hex.trim().is_empty() {
139        return Err(CapabilityDocParseError::new("signature must not be empty"));
140    }
141    let pk_hex = capability_verify_pk_hex();
142    let pk_bytes = hex::decode(pk_hex.trim())
143        .map_err(|e| CapabilityDocParseError::new(format!("capability verify pk decode: {e}")))?;
144    let pk_array: [u8; 32] = pk_bytes
145        .try_into()
146        .map_err(|_| CapabilityDocParseError::new("dev capability verify pk length"))?;
147    let verifying_key = VerifyingKey::from_bytes(&pk_array)
148        .map_err(|e| CapabilityDocParseError::new(format!("capability verify pk: {e}")))?;
149    let sig_bytes = hex::decode(sig_hex.trim()).map_err(|e| {
150        CapabilityDocParseError::new(format!("capability signature hex decode: {e}"))
151    })?;
152    let sig_array: [u8; 64] = sig_bytes
153        .try_into()
154        .map_err(|_| CapabilityDocParseError::new("capability signature length"))?;
155    let signature = Signature::from_bytes(&sig_array);
156    verifying_key
157        .verify_strict(capability_signing_message(doc).as_bytes(), &signature)
158        .map_err(|_| CapabilityDocParseError::new("capability signature invalid"))
159}

161/// Env: override validation clock for tests (RFC3339, PH-S527).
162pub const ENV_CAPABILITY_VALIDATION_NOW: &str = "POOLAI_CAPABILITY_VALIDATION_NOW";
163
164fn capability_validation_now() -> DateTime<Utc> {
165    std::env::var(ENV_CAPABILITY_VALIDATION_NOW)
166        .ok()
167        .and_then(|s| DateTime::parse_from_rfc3339(s.trim()).ok())
168        .map(|dt| dt.with_timezone(&Utc))
169        .unwrap_or_else(Utc::now)
170}
171
172fn check_capability_expires_at(
173    doc: &GalaxyCapabilityDocument,
174    now: DateTime<Utc>,
175    require_expiry: bool,
176) -> Result<(), CapabilityDocParseError> {
177    let Some(raw) = doc
178        .expires_at
179        .as_ref()
180        .map(|s| s.trim())
181        .filter(|s| !s.is_empty())
182    else {
183        if require_expiry {
184            return Err(CapabilityDocParseError::new(
185                "expires_at required for signed capability documents",
186            ));
187        }
188        return Ok(());
189    };
190    let expires = DateTime::parse_from_rfc3339(raw)
191        .map_err(|_| CapabilityDocParseError::new("expires_at must be RFC3339"))?
192        .with_timezone(&Utc);
193    if now >= expires {
194        return Err(CapabilityDocParseError::new("capability document expired"));
195    }
196    Ok(())
197}

199/// Validate required fields and optional dev signature (PH-S466).
200pub fn validate_capability_document(
201    doc: &GalaxyCapabilityDocument,
202) -> Result<(), CapabilityDocParseError> {
203    validate_capability_document_at(doc, capability_validation_now(), false)
204}
205
206/// Validate with explicit clock (PH-S527 tests).
207pub fn validate_capability_document_at(
208    doc: &GalaxyCapabilityDocument,
209    now: DateTime<Utc>,
210    require_expiry: bool,
211) -> Result<(), CapabilityDocParseError> {
212    if doc.peer_id.trim().is_empty() {
213        return Err(CapabilityDocParseError::new("peer_id required"));
214    }
215    verify_capability_signature_stub(doc)?;
216    check_capability_expires_at(doc, now, require_expiry)
217}

219/// `telegram_edge` register-remote requires signed capability document (PH-S504, Galaxy §6.6).
220pub fn validate_telegram_edge_capability(
221    is_telegram_edge: bool,
222    doc: Option<&GalaxyCapabilityDocument>,
223) -> Result<(), CapabilityDocParseError> {
224    if !is_telegram_edge {
225        return Ok(());
226    }
227    let doc = doc.ok_or_else(|| {
228        CapabilityDocParseError::new("capability_document required for telegram_edge origin")
229    })?;
230    if doc
231        .signature
232        .as_ref()
233        .map(|s| s.trim().is_empty())
234        .unwrap_or(true)
235    {
236        return Err(CapabilityDocParseError::new(
237            "signed capability_document required for telegram_edge origin",
238        ));
239    }
240    validate_capability_document_at(doc, capability_validation_now(), true)?;
241    check_tee_attestation_required(doc)
242}
