16/// Settlement gate verdict (stub — no fee-split or payout wire).
17#[derive(Debug, Clone, Copy, PartialEq, Eq)]
18pub enum SettlementGateVerdict {
19    /// Trusted origin or `trust_score` ≥ threshold; payout stub may proceed.
20    PayoutEligible,
21    /// `telegram_edge` below threshold — hold payout (`pending_verification`).
22    PayoutHeld,
23    /// Gate not applied (non-edge origin).
24    NotApplicable,
25}

90/// Gate configuration (env-backed stub).
91#[derive(Debug, Clone, Copy, PartialEq, Eq)]
92pub struct TrustScoreGateConfig {
93    pub min_trust_for_payout: TrustScore,
94}
95
96impl TrustScoreGateConfig {
97    pub const fn default_stub() -> Self {
98        Self {
99            min_trust_for_payout: DEFAULT_MIN_TRUST_FOR_PAYOUT,
100        }
101    }
102
103    /// Parse [`ENV_MIN_TRUST_PAYOUT`] (0..=100); invalid/missing → default.
104    pub fn from_env() -> Self {
105        match std::env::var(ENV_MIN_TRUST_PAYOUT) {
106            Ok(raw) => match raw.trim().parse::<u16>() {
107                Ok(v) if v <= 100 => Self {
108                    min_trust_for_payout: v as TrustScore,
109                },
110                _ => Self::default_stub(),
111            },
112            Err(_) => Self::default_stub(),
113        }
114    }
115}

117/// Clamp arbitrary input to 0..=100.
118#[inline]
119pub fn clamp_trust_score(score: u16) -> TrustScore {
120    score.min(100) as TrustScore
121}

123/// Infer worker origin from peer id prefix (stub until unified worker DTO wire).
124#[inline]
125pub fn infer_worker_origin(peer_id: Option<&str>) -> WorkerOrigin {
126    let Some(id) = peer_id else {
127        return WorkerOrigin::LocalSrv;
128    };
129    if id.starts_with("tg-") || id.starts_with("telegram-") {
130        WorkerOrigin::TelegramEdge
131    } else {
132        WorkerOrigin::LocalSrv
133    }
134}

136/// Evaluate settlement gate for a grid result producer (no payout wire).
137pub fn evaluate_settlement_gate(
138    origin: WorkerOrigin,
139    trust_score: TrustScore,
140    config: &TrustScoreGateConfig,
141) -> SettlementGateVerdict {
142    match origin {
143        WorkerOrigin::LocalSrv => SettlementGateVerdict::NotApplicable,
144        WorkerOrigin::TelegramEdge => {
145            if trust_score >= config.min_trust_for_payout {
146                SettlementGateVerdict::PayoutEligible
147            } else {
148                SettlementGateVerdict::PayoutHeld
149            }
150        }
151    }
152}
153
154/// Grid result path helper: infer origin from `source_peer_id`, apply gate with optional score.
155pub fn evaluate_result_settlement_gate(
156    source_peer_id: Option<&str>,
157    trust_score: Option<TrustScore>,
158    config: &TrustScoreGateConfig,
159) -> SettlementGateVerdict {
160    GATE_EVALUATIONS_TOTAL.fetch_add(1, Ordering::Relaxed);
161    let origin = infer_worker_origin(source_peer_id);
162    let score = match trust_score {
163        Some(s) => {
164            EXPLICIT_SCORE_TOTAL.fetch_add(1, Ordering::Relaxed);
165            s
166        }
167        None => {
168            DEFAULT_SCORE_APPLIED_TOTAL.fetch_add(1, Ordering::Relaxed);
169            DEFAULT_TRUST_SCORE
170        }
171    };
172    observe_last_trust_score(score);
173    let verdict = evaluate_settlement_gate(origin, score, config);
174    record_settlement_gate_verdict(verdict);
175    verdict
176}

178/// Apply signed trust delta and record counter (PH-S456 / PH-S610 / PH-S611).
179pub fn apply_trust_delta(current: TrustScore, delta: i16) -> TrustScore {
180    TRUST_SCORE_DELTA_TOTAL.fetch_add(1, Ordering::Relaxed);
181    let adjusted = i16::from(current).saturating_add(delta);
182    clamp_trust_score(adjusted.max(0) as u16)
183}
184
185/// Apply verification verdict trust delta and record counter (PH-S456 stub).
186pub fn apply_verification_trust_delta(verdict: &str, current: TrustScore) -> TrustScore {
187    let delta = if verdict.eq_ignore_ascii_case("match") {
188        TRUST_DELTA_MATCH
189    } else if verdict.eq_ignore_ascii_case("mismatch") {
190        TRUST_DELTA_MISMATCH
191    } else {
192        return current;
193    };
194    apply_trust_delta(current, delta)
195}

216/// Record settlement gate verdict for Prometheus scrape (grid result path only).
217pub fn record_settlement_gate_verdict(verdict: SettlementGateVerdict) {
218    match verdict {
219        SettlementGateVerdict::PayoutEligible => {
220            PAYOUT_ELIGIBLE_TOTAL.fetch_add(1, Ordering::Relaxed);
221        }
222        SettlementGateVerdict::PayoutHeld => {
223            PAYOUT_HELD_TOTAL.fetch_add(1, Ordering::Relaxed);
224        }
225        SettlementGateVerdict::NotApplicable => {
226            PAYOUT_NOT_APPLICABLE_TOTAL.fetch_add(1, Ordering::Relaxed);
227        }
228    }
229}
