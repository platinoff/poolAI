1//! SQLite persistence for per-peer trust scores (PH-S910, Galaxy §6.5).
2
3use std::collections::HashMap;
4use std::fs;
5use std::path::{Path, PathBuf};
6
7use rusqlite::{params, Connection};

9use crate::grid::galaxy_trust_score::TrustScore;

11const DB_FILE: &str = "trust_scores.db";
12const JSON_LEGACY: &str = "trust_scores.json";
13const MIGRATED_SUFFIX: &str = "json.migrated";
14
15/// Env: data directory for SQLite trust store (`trust_scores.db`).
16pub const ENV_TRUST_SCORE_DATA_DIR: &str = "POOLAI_TRUST_SCORE_DATA_DIR";
17
18/// Env: `sqlite` enables SQLite backend when [`ENV_TRUST_SCORE_DATA_DIR`] is set.
19pub const ENV_TRUST_SCORE_STORE: &str = "POOLAI_TRUST_SCORE_STORE";
20
21pub fn data_dir_from_env() -> Option<PathBuf> {
22    std::env::var(ENV_TRUST_SCORE_DATA_DIR)
23        .ok()
24        .map(|s| s.trim().to_string())
25        .filter(|s| !s.is_empty())
26        .map(PathBuf::from)
27}
28
29pub fn sqlite_enabled() -> bool {
30    std::env::var(ENV_TRUST_SCORE_STORE)
31        .map(|v| v.trim().eq_ignore_ascii_case("sqlite"))
32        .unwrap_or(false)
33}
34
35pub fn load(dir: &Path) -> Result<HashMap<String, TrustScore>, String> {
36    fs::create_dir_all(dir).map_err(|e| format!("mkdir {}: {e}", dir.display()))?;
37    let db_path = dir.join(DB_FILE);
38    let mut conn =
39        Connection::open(&db_path).map_err(|e| format!("open {}: {e}", db_path.display()))?;
40    init_schema(&conn)?;
41    let mut map = read_all(&conn)?;
42    let json_path = dir.join(JSON_LEGACY);
43    if map.is_empty() && json_path.exists() {
44        let text = fs::read_to_string(&json_path)
45            .map_err(|e| format!("read {}: {e}", json_path.display()))?;
46        let from_json: HashMap<String, TrustScore> = serde_json::from_str(&text)
47            .map_err(|e| format!("parse {}: {e}", json_path.display()))?;
48        if !from_json.is_empty() {
49            write_all(&mut conn, &from_json)?;
50            let migrated = json_path.with_extension(MIGRATED_SUFFIX);
51            fs::rename(&json_path, &migrated).map_err(|e| {
52                format!(
53                    "rename {} → {}: {e}",
54                    json_path.display(),
55                    migrated.display()
56                )
57            })?;
58            map = from_json;
59        }
60    }
61    Ok(map)
62}
63
64pub fn persist(dir: &Path, map: &HashMap<String, TrustScore>) -> Result<(), String> {
65    fs::create_dir_all(dir).map_err(|e| format!("mkdir {}: {e}", dir.display()))?;
66    let db_path = dir.join(DB_FILE);
67    let mut conn =
68        Connection::open(&db_path).map_err(|e| format!("open {}: {e}", db_path.display()))?;
69    init_schema(&conn)?;
70    write_all(&mut conn, map)
71}

73/// Import legacy single-file JSON path into SQLite dir on first open (PH-S910).
74pub fn migrate_legacy_json_file(dir: &Path, legacy_path: &Path) -> Result<(), String> {
75    if !legacy_path.exists() {
76        return Ok(());
77    }
78    let db_path = dir.join(DB_FILE);
79    let mut conn =
80        Connection::open(&db_path).map_err(|e| format!("open {}: {e}", db_path.display()))?;
81    init_schema(&conn)?;
82    if !read_all(&conn)?.is_empty() {
83        return Ok(());
84    }
85    let text = fs::read_to_string(legacy_path)
86        .map_err(|e| format!("read {}: {e}", legacy_path.display()))?;
87    let from_json: HashMap<String, TrustScore> =
88        serde_json::from_str(&text).map_err(|e| format!("parse {}: {e}", legacy_path.display()))?;
89    if from_json.is_empty() {
90        return Ok(());
91    }
92    write_all(&mut conn, &from_json)?;
93    let migrated = legacy_path.with_extension(MIGRATED_SUFFIX);
94    fs::rename(legacy_path, &migrated).map_err(|e| {
95        format!(
96            "rename {} → {}: {e}",
97            legacy_path.display(),
98            migrated.display()
99        )
100    })
101}

103fn init_schema(conn: &Connection) -> Result<(), String> {
104    conn.execute_batch(
105        "CREATE TABLE IF NOT EXISTS peer_trust (
106            peer_id TEXT PRIMARY KEY NOT NULL,
107            score INTEGER NOT NULL
108        );",
109    )
110    .map_err(|e| format!("init schema: {e}"))
111}
112
113fn read_all(conn: &Connection) -> Result<HashMap<String, TrustScore>, String> {
114    let mut stmt = conn
115        .prepare("SELECT peer_id, score FROM peer_trust")
116        .map_err(|e| format!("prepare select: {e}"))?;
117    let rows = stmt
118        .query_map([], |row| {
119            let peer_id: String = row.get(0)?;
120            let score: i64 = row.get(1)?;
121            Ok((peer_id, score.clamp(0, 100) as TrustScore))
122        })
123        .map_err(|e| format!("query peer_trust: {e}"))?;
124    let mut map = HashMap::new();
125    for row in rows {
126        let (peer_id, score) = row.map_err(|e| format!("read row: {e}"))?;
127        map.insert(peer_id, score);
128    }
129    Ok(map)
130}
131
132fn write_all(conn: &mut Connection, map: &HashMap<String, TrustScore>) -> Result<(), String> {
133    let tx = conn
134        .transaction()
135        .map_err(|e| format!("begin transaction: {e}"))?;
136    tx.execute("DELETE FROM peer_trust", [])
137        .map_err(|e| format!("clear peer_trust: {e}"))?;
138    for (peer_id, score) in map {
139        tx.execute(
140            "INSERT INTO peer_trust (peer_id, score) VALUES (?1, ?2)",
141            params![peer_id, i64::from(*score)],
142        )
143        .map_err(|e| format!("insert peer_trust {peer_id}: {e}"))?;
144    }
145    tx.commit().map_err(|e| format!("commit peer_trust: {e}"))
146}

148#[cfg(test)]
149mod tests {
150    use super::*;
151    use tempfile::TempDir;
152
153    #[test]
154    fn trust_score_sqlite_roundtrip_ph_s910() {
155        let tmp = TempDir::new().expect("tempdir");
156        let dir = tmp.path();
157        let mut map = HashMap::new();
158        map.insert("tg-edge-1".to_string(), 72u8);
159        map.insert("peer-local".to_string(), 50u8);
160        persist(dir, &map).expect("persist");
161        let loaded = load(dir).expect("load");
162        assert_eq!(loaded.get("tg-edge-1"), Some(&72));
163        assert_eq!(loaded.get("peer-local"), Some(&50));
164    }
165
166    #[test]
167    fn trust_score_sqlite_migrates_json_ph_s910() {
168        let tmp = TempDir::new().expect("tempdir");
169        let dir = tmp.path();
170        let json_path = dir.join(JSON_LEGACY);
171        fs::write(&json_path, r#"{"tg-migrate":88}"#).expect("write json");
172        let loaded = load(dir).expect("load");
173        assert_eq!(loaded.get("tg-migrate"), Some(&88));
174        assert!(!json_path.exists());
175        assert!(dir.join(format!("trust_scores.{MIGRATED_SUFFIX}")).exists());
176    }
177}
