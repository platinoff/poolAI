1//! Galaxy fraud-proof horizon stub (PH-S571, §6.6 TBD).
2//!
3//! When `POOLAI_GALAXY_FRAUD_PROOF=1`, verification mismatch increments
4//! `galaxy_fraud_proof_pending_total` and holds settlement for proof review.
5
6use std::sync::atomic::{AtomicU64, Ordering};
7
8/// Env: enable fraud-proof pending hold on verification mismatch (PH-S571).
9pub const ENV_FRAUD_PROOF: &str = "POOLAI_GALAXY_FRAUD_PROOF";
10
11/// Metric: settlement rows held for fraud-proof review (PH-S571).
12pub const METRIC_FRAUD_PROOF_PENDING_TOTAL: &str = "galaxy_fraud_proof_pending_total";
13
14static FRAUD_PROOF_PENDING_TOTAL: AtomicU64 = AtomicU64::new(0);
15
16pub fn fraud_proof_enabled() -> bool {
17    matches!(
18        std::env::var(ENV_FRAUD_PROOF)
19            .ok()
20            .map(|v| v.trim().to_ascii_lowercase())
21            .as_deref(),
22        Some("1") | Some("true") | Some("yes")
23    )
24}
25
26pub fn fraud_proof_pending_total() -> u64 {
27    FRAUD_PROOF_PENDING_TOTAL.load(Ordering::Relaxed)
28}
29
30pub fn record_fraud_proof_pending() {
31    FRAUD_PROOF_PENDING_TOTAL.fetch_add(1, Ordering::Relaxed);
32}
33
34/// Returns true when mismatch should enter fraud-proof hold (PH-S571).
35pub fn evaluate_fraud_proof_hold(is_mismatch: bool) -> bool {
36    is_mismatch && fraud_proof_enabled()
37}
38
39#[cfg(any(test, feature = "test-utils"))]
40pub fn reset_fraud_proof_metrics_for_test() {
41    FRAUD_PROOF_PENDING_TOTAL.store(0, Ordering::Relaxed);
42}
43
44#[cfg(test)]
45mod tests {
46    use super::*;
47
48    #[test]
49    fn fraud_proof_hold_ph_s571() {
50        static ENV_LOCK: std::sync::Mutex<()> = std::sync::Mutex::new(());
51        let _guard = ENV_LOCK.lock().unwrap_or_else(|e| e.into_inner());
52        reset_fraud_proof_metrics_for_test();
53        std::env::remove_var(ENV_FRAUD_PROOF);
54        assert!(!evaluate_fraud_proof_hold(true));
55        std::env::set_var(ENV_FRAUD_PROOF, "1");
56        assert!(evaluate_fraud_proof_hold(true));
57        record_fraud_proof_pending();
58        assert_eq!(fraud_proof_pending_total(), 1);
59        std::env::remove_var(ENV_FRAUD_PROOF);
60        reset_fraud_proof_metrics_for_test();
61    }
62}
