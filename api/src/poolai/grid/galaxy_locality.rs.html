153/// Fraction of `required_shard_ids` present in worker inventory (0..=1).
154#[inline]
155pub fn shard_local_hit(inventory_shard_ids: &[String], required_shard_ids: &[String]) -> f64 {
156    if required_shard_ids.is_empty() {
157        return 1.0;
158    }
159    let hits = required_shard_ids
160        .iter()
161        .filter(|id| inventory_shard_ids.contains(id))
162        .count();
163    hits as f64 / required_shard_ids.len() as f64
164}

166/// `1 / (1 + latency_ms_p50/100)` from `network_profile` (concept §5.2).
167#[inline]
168pub fn latency_factor(latency_ms_p50: u32) -> f64 {
169    1.0 / (1.0 + f64::from(latency_ms_p50) / 100.0)
170}

172/// Penalty when `latency_ms_p95` exceeds `ratio_threshold × p50` (Galaxy §8.1, PH-S603).
173#[inline]
174pub fn tail_latency_penalty(
175    latency_ms_p50: u32,
176    latency_ms_p95: Option<u32>,
177    ratio_threshold: f64,
178    max_penalty: f64,
179) -> f64 {
180    let Some(p95) = latency_ms_p95 else {
181        return 0.0;
182    };
183    if latency_ms_p50 == 0 || p95 <= latency_ms_p50 {
184        return 0.0;
185    }
186    let ratio = f64::from(p95) / f64::from(latency_ms_p50);
187    if ratio <= ratio_threshold {
188        return 0.0;
189    }
190    let excess = (ratio - ratio_threshold) / ratio_threshold;
191    excess.mul_add(max_penalty, 0.0).min(max_penalty)
192}

194pub fn record_tail_latency_penalty_if_applicable(latency_ms_p50: u32, latency_ms_p95: Option<u32>) {
195    let penalty = tail_latency_penalty(
196        latency_ms_p50,
197        latency_ms_p95,
198        DEFAULT_TAIL_LATENCY_RATIO_THRESHOLD,
199        DEFAULT_TAIL_LATENCY_MAX_PENALTY,
200    );
201    if penalty > 0.0 {
202        TAIL_LATENCY_PENALTY_TOTAL.fetch_add(1, Ordering::Relaxed);
203    }
204}

215/// Hot tier effectiveness: blend shard presence and task profile match (0..=1).
216pub fn hot_tier_hit_ratio(inventory: &LocalitySeedInventory, task: &LocalityTask) -> f64 {
217    let shard = shard_local_hit(&inventory.shard_ids, &task.required_shard_ids);
218    let profile_ok = inventory.hot_tier.profiles.is_empty()
219        || inventory.hot_tier.profiles.contains(&task.task_profile);
220    let profile = if profile_ok { 1.0 } else { 0.0 };
221    (shard + profile) / 2.0
222}

224/// Cross-region egress MB applied to the score (0 when local replica or same region).
225pub fn effective_cross_region_egress_mb(worker: &LocalityWorker, task: &LocalityTask) -> f64 {
226    if shard_local_hit(&worker.seed_inventory.shard_ids, &task.required_shard_ids) >= 1.0 {
227        return 0.0;
228    }
229    let Some(source) = task.source_region.as_deref() else {
230        return 0.0;
231    };
232    if worker.network_profile.region == source {
233        return 0.0;
234    }
235    if worker
236        .seed_inventory
237        .local_replica_regions
238        .iter()
239        .any(|r| r == source)
240    {
241        return 0.0;
242    }
243    task.estimated_cross_region_egress_mb.max(0.0)
244}

246/// Penalty subtracted from `locality_score` when `profile_age_secs` is missing or exceeds `max_age_secs`
247/// (Galaxy §8.1 stale profile stub, PH-S169).
248#[inline]
249pub fn stale_network_profile_penalty(
250    profile_age_secs: Option<u64>,
251    max_age_secs: u64,
252    max_penalty: f64,
253) -> f64 {
254    match profile_age_secs {
255        None => max_penalty,
256        Some(age) if age > max_age_secs => max_penalty,
257        Some(_) => 0.0,
258    }
259}

261/// Placement score (concept §5.2):
262/// `w_shard×shard_local_hit + w_lat×latency_factor + w_hot×hot_tier_hit − w_egress×egress_mb − stale_penalty`.
263pub fn locality_score(
264    worker: &LocalityWorker,
265    task: &LocalityTask,
266    weights: &LocalityWeights,
267) -> f64 {
268    let shard = shard_local_hit(&worker.seed_inventory.shard_ids, &task.required_shard_ids);
269    let lat = latency_factor(worker.network_profile.latency_ms_p50);
270    let hot = hot_tier_hit_ratio(&worker.seed_inventory, task);
271    let egress = effective_cross_region_egress_mb(worker, task);
272    let stale = stale_network_profile_penalty(
273        worker.network_profile.profile_age_secs,
274        DEFAULT_STALE_PROFILE_MAX_AGE_SECS,
275        DEFAULT_STALE_PROFILE_PENALTY,
276    );
277    let tail = tail_latency_penalty(
278        worker.network_profile.latency_ms_p50,
279        worker.network_profile.latency_ms_p95,
280        DEFAULT_TAIL_LATENCY_RATIO_THRESHOLD,
281        DEFAULT_TAIL_LATENCY_MAX_PENALTY,
282    );
283
284    weights.w_shard * shard + weights.w_lat * lat + weights.w_hot * hot
285        - weights.w_egress * egress
286        - stale
287        - tail
288}

298/// Same as [`rank_workers_by_locality`] with explicit weights.
299pub fn rank_workers_by_locality_with_weights(
300    workers: &[LocalityWorker],
301    task: &LocalityTask,
302    weights: &LocalityWeights,
303) -> Vec<LocalityRankedWorker> {
304    let mut ranked: Vec<LocalityRankedWorker> = workers
305        .iter()
306        .map(|w| {
307            record_network_profile_stale_if_applicable(
308                w.network_profile.profile_age_secs,
309                DEFAULT_STALE_PROFILE_MAX_AGE_SECS,
310            );
311            record_tail_latency_penalty_if_applicable(
312                w.network_profile.latency_ms_p50,
313                w.network_profile.latency_ms_p95,
314            );
315            LocalityRankedWorker {
316                worker_id: w.worker_id.clone(),
317                score: locality_score(w, task, weights),
318            }
319        })
320        .collect();
321    ranked.sort_by(|a, b| {
322        b.score
323            .partial_cmp(&a.score)
324            .unwrap_or(std::cmp::Ordering::Equal)
325            .then_with(|| {
326                worker_queue_depth(workers, &a.worker_id)
327                    .cmp(&worker_queue_depth(workers, &b.worker_id))
328            })
329            .then_with(|| {
330                worker_pricing_usd_micro(workers, &a.worker_id)
331                    .cmp(&worker_pricing_usd_micro(workers, &b.worker_id))
332            })
333            .then_with(|| a.worker_id.cmp(&b.worker_id))
334    });
335    if let Some(best) = ranked.first() {
336        if let Some(worker) = workers.iter().find(|w| w.worker_id == best.worker_id) {
337            observe_last_shard_local_hit_ratio(shard_local_hit(
338                &worker.seed_inventory.shard_ids,
339                &task.required_shard_ids,
340            ));
341            observe_last_hot_tier_hit_ratio(hot_tier_hit_ratio(&worker.seed_inventory, task));
342            observe_last_cross_region_egress_mb(effective_cross_region_egress_mb(worker, task));
343        }
344    }
345    ranked
346}

491}

514/// Prefer workers with hot tier hit ratio above gate when cold peers exist (PH-S612).
515pub fn pick_best_worker_by_locality_with_hot_tier_gate<'a>(
516    workers: &'a [LocalityWorker],
517    task: &LocalityTask,
518) -> Option<&'a LocalityWorker> {
519    if task.required_shard_ids.is_empty() || workers.len() < 2 {
520        return pick_best_worker_by_locality(workers, task);
521    }
522    let has_cold = workers
523        .iter()
524        .any(|w| hot_tier_hit_ratio(&w.seed_inventory, task) <= f64::EPSILON);
525    if !has_cold {
526        return pick_best_worker_by_locality(workers, task);
527    }
528    let hot_workers: Vec<LocalityWorker> = workers
529        .iter()
530        .filter(|w| hot_tier_hit_ratio(&w.seed_inventory, task) > HOT_TIER_SCHEDULE_GATE_MIN)
531        .cloned()
532        .collect();
533    if hot_workers.is_empty() || hot_workers.len() >= workers.len() {
534        return pick_best_worker_by_locality(workers, task);
535    }
536    record_hot_tier_gate_applied();
537    let best = pick_best_worker_by_locality(&hot_workers, task)?;
538    workers.iter().find(|w| w.worker_id == best.worker_id)
539}
