1//! File-backed job store (post-S38). Set `POOLAI_JOB_DATA_DIR` (e.g. `data/jobs`) to persist
2//! across coordinator restarts; when unset, in-memory only (same as S38 stub).
3
4use std::fs;
5use std::path::{Path, PathBuf};
6use std::sync::{LazyLock, Mutex};
7
8use crate::core::error::AppError;
9use crate::job::{JobRecord, JobStatus};
10
11const JOBS_FILE: &str = "jobs.json";
12
13#[derive(Debug, serde::Serialize, serde::Deserialize, Default)]
14struct JobsFile {
15    jobs: Vec<JobRecord>,
16}
17
18/// In-process job registry with optional JSON persistence.
19pub struct JobStore {
20    jobs: Mutex<Vec<JobRecord>>,
21    data_dir: Option<PathBuf>,
22}
23
24impl JobStore {
25    /// Shared store for HTTP handlers (respects `POOLAI_JOB_DATA_DIR` at first use).
26    pub fn global() -> &'static JobStore {
27        static STORE: LazyLock<JobStore> = LazyLock::new(|| JobStore::open(data_dir_from_env()));
28        &STORE
29    }
30
31    #[cfg(test)]
32    pub fn open_for_test(data_dir: Option<PathBuf>) -> Self {
33        Self::open(data_dir)
34    }
35
36    fn open(data_dir: Option<PathBuf>) -> Self {
37        let jobs = data_dir
38            .as_ref()
39            .map(|d| d.join(JOBS_FILE))
40            .and_then(|p| load_jobs_file(&p).ok())
41            .unwrap_or_default();
42        Self {
43            jobs: Mutex::new(jobs),
44            data_dir,
45        }
46    }
47
48    pub fn list(&self) -> Result<Vec<JobRecord>, AppError> {
49        let guard = self
50            .jobs
51            .lock()
52            .map_err(|_| AppError::InternalError("job store lock poisoned".into()))?;
53        Ok(guard.clone())
54    }
55
56    pub fn get(&self, id: &str) -> Result<Option<JobRecord>, AppError> {
57        let guard = self
58            .jobs
59            .lock()
60            .map_err(|_| AppError::InternalError("job store lock poisoned".into()))?;
61        Ok(guard.iter().find(|r| r.spec.id.0 == id).cloned())
62    }
63
64    pub fn push(&self, record: JobRecord) -> Result<(), AppError> {
65        {
66            let mut guard = self
67                .jobs
68                .lock()
69                .map_err(|_| AppError::InternalError("job store lock poisoned".into()))?;
70            guard.push(record);
71        }
72        self.persist()
73    }
74
75    /// FM-020: transition all `Submitted` rows to `Scheduled` (priority desc), then persist once.
76    pub fn promote_submitted_to_scheduled(&self) -> Result<usize, AppError> {
77        let scheduled = {
78            let mut guard = self
79                .jobs
80                .lock()
81                .map_err(|_| AppError::InternalError("job store lock poisoned".into()))?;
82            let mut indices: Vec<usize> = guard
83                .iter()
84                .enumerate()
85                .filter(|(_, r)| r.status == JobStatus::Submitted)
86                .map(|(i, _)| i)
87                .collect();
88            indices.sort_by(|&a, &b| guard[b].spec.priority.cmp(&guard[a].spec.priority));
89            let count = indices.len();
90            for i in indices {
91                guard[i].status = JobStatus::Scheduled;
92            }
93            count
94        };
95        if scheduled > 0 {
96            self.persist()?;
97        }
98        Ok(scheduled)
99    }
100
101    fn persist(&self) -> Result<(), AppError> {
102        let Some(dir) = self.data_dir.as_ref() else {
103            return Ok(());
104        };
105        let guard = self
106            .jobs
107            .lock()
108            .map_err(|_| AppError::InternalError("job store lock poisoned".into()))?;
109        let path = dir.join(JOBS_FILE);
110        let snapshot = JobsFile {
111            jobs: guard.clone(),
112        };
113        write_json_atomic(&path, &snapshot)
114            .map_err(|e| AppError::InternalError(format!("persist jobs: {e}")))
115    }
116}
117
118pub fn data_dir_from_env() -> Option<PathBuf> {
119    std::env::var("POOLAI_JOB_DATA_DIR")
120        .ok()
121        .filter(|s| !s.trim().is_empty())
122        .map(PathBuf::from)
123}
124
125fn load_jobs_file(path: &Path) -> Result<Vec<JobRecord>, String> {
126    if !path.exists() {
127        return Ok(Vec::new());
128    }
129    let bytes = fs::read(path).map_err(|e| format!("read {}: {e}", path.display()))?;
130    let file: JobsFile =
131        serde_json::from_slice(&bytes).map_err(|e| format!("parse {}: {e}", path.display()))?;
132    Ok(file.jobs)
133}
134
135fn write_json_atomic<T: serde::Serialize>(path: &Path, value: &T) -> Result<(), String> {
136    if let Some(parent) = path.parent() {
137        fs::create_dir_all(parent).map_err(|e| format!("mkdir {}: {e}", parent.display()))?;
138    }
139    let data = serde_json::to_vec_pretty(value)
140        .map_err(|e| format!("serialize {}: {e}", path.display()))?;
141    let tmp = path.with_extension("json.tmp");
142    fs::write(&tmp, &data).map_err(|e| format!("write {}: {e}", tmp.display()))?;
143    fs::rename(&tmp, path).map_err(|e| format!("rename {}: {e}", path.display()))
144}

146#[cfg(test)]
147mod tests {
148    use super::*;
149    use crate::job::{JobId, JobKind, JobSpec, JobStatus};
150    use chrono::Utc;
151    use std::sync::{Mutex, OnceLock};
152    use tempfile::TempDir;
153
154    static ENV_LOCK: OnceLock<Mutex<()>> = OnceLock::new();
155
156    fn env_lock() -> std::sync::MutexGuard<'static, ()> {
157        ENV_LOCK.get_or_init(|| Mutex::new(())).lock().unwrap()
158    }
159
160    #[test]
161    fn persist_and_reload_jobs() {
162        let _guard = env_lock();
163        let tmp = TempDir::new().expect("tempdir");
164        let dir = tmp.path().to_path_buf();
165
166        let record = JobRecord {
167            spec: JobSpec {
168                id: JobId::new("job-persist-1"),
169                kind: JobKind::Inference,
170                resources: Default::default(),
171                priority: 0,
172                max_duration_secs: None,
173                input_artifact_ids: vec![],
174                verification_policy: None,
175                deadline: None,
176            },
177            status: JobStatus::Submitted,
178            created_at: Utc::now(),
179        };
180
181        {
182            let store = JobStore::open_for_test(Some(dir.clone()));
183            store.push(record.clone()).expect("push");
184        }
185
186        let reloaded = JobStore::open_for_test(Some(dir));
187        let jobs = reloaded.list().expect("list");
188        assert_eq!(jobs.len(), 1);
189        assert_eq!(jobs[0].spec.id.0, "job-persist-1");
190    }
