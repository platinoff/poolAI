106    pub fn build_model_request(request: &ChatCompletionRequest) -> ModelRequest {
107        let prompt = request
108            .messages
109            .iter()
110            .map(|m| format!("{}: {}", m.role, m.content))
111            .collect::<Vec<_>>()
112            .join("\n");
113
114        let model_params = ModelParameters {
115            temperature: request.temperature,
116            max_tokens: request.max_tokens,
117            top_p: request.top_p,
118            frequency_penalty: request.frequency_penalty,
119            presence_penalty: request.presence_penalty,
120            stop_sequences: request.stop.clone().unwrap_or_default(),
121            seed: None,
122        };
123
124        ModelRequest {
125            input: prompt,
126            parameters: model_params,
127            session_id: None,
128            priority: 5,
129            timeout: Some(30),
130        }
131    }

133    /// Returns `(manager Arc, instance_id)` when an instance serves `model_id`.
134    pub async fn resolve_instance_for_model(
135        ctx: &ApiContext,
136        model_id: &str,
137    ) -> Option<(Arc<TokioRwLock<InstanceManager>>, String)> {
138        let mgr = ctx.instance_manager.get().cloned()?;
139        let instance_id = {
140            let m = mgr.read().await;
141            m.get_instance_by_model_id(model_id)
142                .await
143                .map(|i| i.instance_id.clone())
144        }?;
145        Some((mgr, instance_id))
146    }

148    pub async fn complete_via_instance(
149        instance_id: &str,
150        model: &str,
151        request: ModelRequest,
152        manager_arc: Arc<TokioRwLock<InstanceManager>>,
153    ) -> Result<ChatCompletionResponse, AppError> {
154        let manager = manager_arc.read().await;
155        let model_response = manager
156            .process_request_via_instance(instance_id, request.clone())
157            .await?;
158
159        Ok(ChatCompletionResponse {
160            id: format!("chatcmpl-{}", uuid::Uuid::new_v4()),
161            object: "chat.completion".to_string(),
162            created: chrono::Utc::now().timestamp(),
163            model: model.to_string(),
164            choices: vec![ChatCompletionChoice {
165                index: 0,
166                message: ChatMessage {
167                    role: "assistant".to_string(),
168                    content: model_response.output,
169                },
170                finish_reason: "stop".to_string(),
171            }],
172            usage: ChatUsage {
173                prompt_tokens: model_response.metrics.tokens_generated,
174                completion_tokens: model_response.metrics.tokens_generated,
175                total_tokens: model_response.metrics.tokens_generated * 2,
176            },
177        })
178    }

193                index: 0,
194                message: ChatMessage {
195                    role: "assistant".to_string(),

198                finish_reason: "stop".to_string(),
199            }],
200            usage: ChatUsage {
201                prompt_tokens: request.input.len() / 4,
202                completion_tokens: response_text.len() / 4,
203                total_tokens: (request.input.len() + response_text.len()) / 4,

208    pub fn stream_from_instance(
209        instance_id: &str,
210        model: &str,
211        request: ModelRequest,
212        manager_opt: Option<Arc<TokioRwLock<InstanceManager>>>,
213    ) -> impl Stream<Item = Result<Event, Infallible>> {
214        use tokio::sync::mpsc;
215
216        let instance_id = instance_id.to_string();
217        let model = model.to_string();
218        let response_id = format!("chatcmpl-{}", uuid::Uuid::new_v4());
219        let created = chrono::Utc::now().timestamp();
220
221        if let Some(manager_arc) = manager_opt {
222            let (tx, rx) = mpsc::unbounded_channel();
223            let manager_clone = manager_arc.clone();
224            let request_clone = request.clone();
225            let instance_id_clone = instance_id.clone();
226            let model_clone = model.clone();
227            let response_id_clone = response_id.clone();
228
229            let request_input = request_clone.input.clone();
230            tokio::spawn(async move {
231                let manager = manager_clone.read().await;
232                match manager
233                    .process_request_via_instance(&instance_id_clone, request_clone)
234                    .await
235                {
236                    Ok(model_response) => {
237                        stream_chunks_to_channel(
238                            &tx,
239                            &response_id_clone,
240                            created,
241                            &model_clone,
242                            &model_response.output,
243                        );
244                    }
245                    Err(_) => {
246                        let response_text = format!("Processed: {}", request_input);
247                        stream_chunks_to_channel(
248                            &tx,
249                            &response_id_clone,
250                            created,
251                            &model_clone,
252                            &response_text,
253                        );
254                    }
255                }
256            });
257
258            return tokio_stream::wrappers::UnboundedReceiverStream::new(rx);
259        }
260
261        let (tx, rx) = mpsc::unbounded_channel();
262        let model_clone = model.clone();
263        let request_input = request.input.clone();
264        let response_id_clone = response_id.clone();
265
266        tokio::spawn(async move {
267            tokio::time::sleep(Duration::from_millis(100)).await;
268            let response_text = format!("Processed: {}", request_input);
269            stream_chunks_to_channel(
270                &tx,
271                &response_id_clone,
272                created,
273                &model_clone,
274                &response_text,
275            );
276        });
277
278        tokio_stream::wrappers::UnboundedReceiverStream::new(rx)
279    }

281    pub fn stream_fallback(
282        model: &str,
283        request: ModelRequest,
284    ) -> impl Stream<Item = Result<Event, Infallible>> {
285        let model = model.to_string();
286        let response_id = format!("chatcmpl-{}", uuid::Uuid::new_v4());
287        let created = chrono::Utc::now().timestamp();
288        let response_text = format!("Processed: {}", request.input);
289        let chunks: Vec<String> = response_text
290            .chars()
291            .collect::<Vec<_>>()
292            .chunks(5)
293            .map(|chunk| chunk.iter().collect())
294            .collect();
295
296        let chunks_len = chunks.len();
297        let model_clone = model.clone();
298        let response_id_clone = response_id.clone();
299
300        stream::iter(chunks.into_iter().enumerate().map(move |(index, chunk)| {
301            chunk_event(
302                &response_id_clone,
303                created,
304                &model_clone,
305                index,
306                chunks_len,
307                chunk,
308            )
309        }))
310        .chain(stream::iter(vec![Ok(Event::default().data("[DONE]"))]))
311    }
312}
313
314fn stream_chunks_to_channel(
315    tx: &tokio::sync::mpsc::UnboundedSender<Result<Event, Infallible>>,
316    response_id: &str,
317    created: i64,
318    model: &str,
319    response_text: &str,
320) {
321    let chunks: Vec<String> = response_text
322        .chars()
323        .collect::<Vec<_>>()
324        .chunks(5)
325        .map(|chunk| chunk.iter().collect())
326        .collect();
327    let chunks_len = chunks.len();
328    for (index, chunk) in chunks.into_iter().enumerate() {
329        let _ = tx.send(chunk_event(
330            response_id,
331            created,
332            model,
333            index,
334            chunks_len,
335            chunk,
336        ));
337    }
338    let _ = tx.send(Ok(Event::default().data("[DONE]")));
339}
340
341fn chunk_event(
342    response_id: &str,
343    created: i64,
344    model: &str,
345    index: usize,
346    chunks_len: usize,
347    chunk: String,
348) -> Result<Event, Infallible> {
349    let is_first = index == 0;
350    let is_last = index == chunks_len - 1;
351    let chunk_data = ChatCompletionChunk {
352        id: response_id.to_string(),
353        object: "chat.completion.chunk".to_string(),
354        created,
355        model: model.to_string(),
356        choices: vec![ChatCompletionChunkChoice {
357            index: 0,
358            delta: ChatMessageDelta {
359                role: if is_first {
360                    Some("assistant".to_string())
361                } else {
362                    None
363                },
364                content: Some(chunk),
365            },
366            finish_reason: if is_last {
367                Some("stop".to_string())
368            } else {
369                None
370            },
371        }],
372    };
373    let json = serde_json::to_string(&chunk_data).unwrap_or_default();
374    Ok(Event::default().data(json))
