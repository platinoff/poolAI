1//! Telegram user/chat ↔ virtual-node (`peer_id`) bindings (FM-016+).
2//!
3//! Coordinator resolves Telegram updates to a registered worker via `telegram_user_id`.
4//! Optional persistence: set `POOLAI_VIRTUAL_NODE_DATA_DIR` (see `virtual_node_store`).
5
6use chrono::{DateTime, Utc};
7use serde::{Deserialize, Serialize};
8use std::collections::HashMap;
9use std::sync::{Mutex, OnceLock};
10
11use super::virtual_node_store;
12
13#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
14pub struct TelegramBinding {
15    pub telegram_user_id: String,
16    #[serde(skip_serializing_if = "Option::is_none")]
17    pub chat_id: Option<String>,
18    pub peer_id: String,
19    pub bound_at: DateTime<Utc>,
20}
21
22fn bindings() -> &'static Mutex<HashMap<String, TelegramBinding>> {
23    static B: OnceLock<Mutex<HashMap<String, TelegramBinding>>> = OnceLock::new();
24    B.get_or_init(|| {
25        let mut map = HashMap::new();
26        if let Ok(loaded) = virtual_node_store::load_json("telegram_bindings.json") {
27            if let Ok(entries) = serde_json::from_value::<Vec<TelegramBinding>>(loaded) {
28                for b in entries {
29                    map.insert(b.telegram_user_id.clone(), b);
30                }
31            }
32        }
33        Mutex::new(map)
34    })
35}
36
37fn persist_all(guard: &HashMap<String, TelegramBinding>) {
38    let entries: Vec<_> = guard.values().cloned().collect();
39    if let Err(e) = virtual_node_store::save_json("telegram_bindings.json", &entries) {
40        tracing::warn!("virtual node telegram bindings persist failed: {e}");
41    }
42}
43
44pub struct VirtualNodeTelegramBindingService;
45
46impl VirtualNodeTelegramBindingService {
47    pub fn bind(telegram_user_id: &str, chat_id: Option<String>, peer_id: &str) -> TelegramBinding {
48        let binding = TelegramBinding {
49            telegram_user_id: telegram_user_id.to_string(),
50            chat_id,
51            peer_id: peer_id.to_string(),
52            bound_at: Utc::now(),
53        };
54        let mut guard = bindings().lock().expect("telegram bindings lock");
55        guard.insert(telegram_user_id.to_string(), binding.clone());
56        persist_all(&guard);
57        binding
58    }
59
60    pub fn lookup(telegram_user_id: &str) -> Option<TelegramBinding> {
61        let guard = bindings().lock().expect("telegram bindings lock");
62        guard.get(telegram_user_id).cloned()
63    }
64
65    pub fn lookup_by_peer(peer_id: &str) -> Option<TelegramBinding> {
66        let guard = bindings().lock().expect("telegram bindings lock");
67        guard.values().find(|b| b.peer_id == peer_id).cloned()
68    }
69
70    pub fn list() -> Vec<TelegramBinding> {
71        let guard = bindings().lock().expect("telegram bindings lock");
72        let mut v: Vec<_> = guard.values().cloned().collect();
73        v.sort_by(|a, b| a.telegram_user_id.cmp(&b.telegram_user_id));
74        v
75    }
76
77    pub fn unbind(telegram_user_id: &str) -> bool {
78        let mut guard = bindings().lock().expect("telegram bindings lock");
79        let removed = guard.remove(telegram_user_id).is_some();
80        if removed {
81            persist_all(&guard);
82        }
83        removed
84    }
85
86    /// Test helper — reset in-memory bindings (does not delete on-disk store unless dir set).
87    pub fn clear_all() {
88        let mut guard = bindings().lock().expect("telegram bindings lock");
89        guard.clear();
90        persist_all(&guard);
91    }
92}

94#[cfg(test)]
95mod tests {
96    use super::*;
97
98    #[test]
99    fn bind_and_lookup_roundtrip() {
100        VirtualNodeTelegramBindingService::clear_all();
101        VirtualNodeTelegramBindingService::bind("tg-42", Some("chat-99".into()), "worker-a");
102        let b = VirtualNodeTelegramBindingService::lookup("tg-42").expect("binding");
103        assert_eq!(b.peer_id, "worker-a");
104        assert_eq!(b.chat_id.as_deref(), Some("chat-99"));
105        VirtualNodeTelegramBindingService::clear_all();
106    }
