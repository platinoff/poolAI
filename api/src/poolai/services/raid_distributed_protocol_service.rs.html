1//! Distributed RAID wire-protocol orchestration (Put/Get/Delete artifact, sync, health, cluster).
2//!
3//! HTTP handlers in `network::raid_distributed_handlers` delegate here.
4//!
5//! **LeaveCluster (FM-008):** when [`crate::services::raid_service::RaidService::list_nodes`] returns a
6//! non-empty list, the leaving `node_id` must match a registered member (otherwise `InvalidRequest`
7//! before replication). Empty membership keeps prior behaviour (delete may still return not-found).
8

28    pub async fn put_artifact(ctx: &ApiContext, message: ProtocolMessage) -> Response {
29        info!("Received PutArtifact message: {}", message.id);
30
31        let payload = match message.extract_put_artifact() {
32            Ok(p) => p,
33            Err(e) => {
34                error!("Failed to extract PutArtifact payload: {}", e);
35                return create_error_response(
36                    &message,
37                    ErrorCode::InvalidRequest,
38                    format!("Invalid payload: {}", e),
39                );
40            }
41        };
42
43        // Decode base64 data if present
44        let artifact_data = if let Some(data_base64) = payload.data {
45            match base64_decode(&data_base64) {
46                Ok(data) => Some(data),
47                Err(e) => {
48                    error!("Failed to decode base64 data: {}", e);
49                    return create_error_response(
50                        &message,
51                        ErrorCode::InvalidRequest,
52                        format!("Invalid base64 data: {}", e),
53                    );
54                }
55            }
56        } else {
57            None
58        };
59
60        match artifact_data {
61            Some(data) => match RaidService::put_artifact_versioned_name(
62                ctx,
63                &payload.metadata.name,
64                &payload.metadata.version,
65                &data,
66            )
67            .await
68            {
69                Ok(artifact_ref) => {
70                    info!("Artifact stored successfully: {}", artifact_ref.id);
71                    let response = PutArtifactResponse {
72                        status: OperationStatus::Success,
73                        artifact_id: payload.artifact_id,
74                        stored_at: Utc::now(),
75                        error: None,
76                    };
77                    create_success_response(&message, response)
78                }
79                Err(RaidServiceError::ManagerUnavailable) => create_error_response(
80                    &message,
81                    ErrorCode::InvalidRequest,
82                    "RAID manager not initialized".to_string(),
83                ),
84                Err(RaidServiceError::Operation(e)) => {
85                    error!("Failed to store artifact: {}", e);
86                    create_error_response(
87                        &message,
88                        ErrorCode::ReplicationFailed,
89                        format!("Storage failed: {}", e),
90                    )
91                }
92                Err(e) => {
93                    error!("Failed to store artifact: {:?}", e);
94                    create_error_response(
95                        &message,
96                        ErrorCode::ReplicationFailed,
97                        format!("Storage failed: {:?}", e),
98                    )
99                }
100            },
101            None => {
102                // Metadata-only replication (data will be fetched separately)
103                warn!("PutArtifact without data - metadata-only replication");
104                let response = PutArtifactResponse {
105                    status: OperationStatus::Success,
106                    artifact_id: payload.artifact_id,
107                    stored_at: Utc::now(),
108                    error: None,
109                };
110                create_success_response(&message, response)
111            }
112        }
113    }

208            None => {

211                    status: ArtifactStatus::NotFound,
212                    artifact_id: payload.artifact_id,
213                    metadata: None,
214                    data: None,
215                    error: Some("Artifact not found".to_string()),
216                };
217                create_success_response(&message, response)
218            }
219        }

511    pub async fn leave_cluster(ctx: &ApiContext, message: ProtocolMessage) -> Response {
512        info!("Received LeaveCluster message: {}", message.id);
513
514        let payload = match message.extract_leave_cluster() {
515            Ok(p) => p,
516            Err(e) => {
517                error!("Failed to extract LeaveCluster payload: {}", e);
518                return create_error_response(
519                    &message,
520                    ErrorCode::InvalidRequest,
521                    format!("Invalid payload: {}", e),
522                );
523            }
524        };
525
526        let node_id = match Uuid::parse_str(&message.node_id) {
527            Ok(id) => id,
528            Err(e) => {
529                error!("Invalid node_id in LeaveCluster: {}", e);
530                return create_error_response(
531                    &message,
532                    ErrorCode::InvalidRequest,
533                    format!("Invalid node_id: {}", e),
534                );
535            }
536        };
537
538        // FM-008: if membership is non-empty, only registered nodes may leave (avoid pointless replication).
539        let cluster_nodes = match RaidService::list_nodes(ctx).await {
540            Ok(ns) => ns,
541            Err(RaidServiceError::ManagerUnavailable) => {
542                return create_error_response(
543                    &message,
544                    ErrorCode::InvalidRequest,
545                    "RAID manager not initialized".to_string(),
546                );
547            }
548            Err(e) => {
549                error!("LeaveCluster failed to list cluster nodes: {:?}", e);
550                return create_error_response(
551                    &message,
552                    ErrorCode::ReplicationFailed,
553                    format!("Failed to read cluster membership: {:?}", e),
554                );
555            }
556        };
557        if !cluster_nodes.is_empty() && !cluster_nodes.iter().any(|n| n.id == node_id) {
558            return create_error_response(
559                &message,
560                ErrorCode::InvalidRequest,
561                format!(
562                    "Node {} is not a cluster member ({} nodes registered)",
563                    node_id,
564                    cluster_nodes.len()
565                ),
566            );
567        }
568
569        let (replication_complete, artifacts_moved) = if payload.graceful {
570            match RaidService::list_artifacts(ctx).await {
571                Ok(artifacts) => {
572                    let total = artifacts.len() as u32;
573                    let mut moved = 0u32;
574                    for a in artifacts {
575                        if RaidService::replicate_stored_artifact(ctx, a.id)
576                            .await
577                            .is_ok()
578                        {
579                            moved = moved.saturating_add(1);
580                        }
581                    }
582                    let complete = total == 0 || moved == total;
583                    (complete, moved)
584                }
585                Err(RaidServiceError::ManagerUnavailable) => {
586                    return create_error_response(
587                        &message,
588                        ErrorCode::InvalidRequest,
589                        "RAID manager not initialized".to_string(),
590                    );
591                }
592                Err(e) => {
593                    error!("Graceful leave failed listing artifacts: {:?}", e);
594                    return create_error_response(
595                        &message,
596                        ErrorCode::ReplicationFailed,
597                        format!("Failed to list artifacts for graceful leave: {:?}", e),
598                    );
599                }
600            }
601        } else {
602            (true, 0)
603        };
604
605        match RaidService::delete_worker(ctx, node_id).await {
606            Ok(()) => {}
607            Err(RaidServiceError::ManagerUnavailable) => {
608                return create_error_response(
609                    &message,
610                    ErrorCode::InvalidRequest,
611                    "RAID manager not initialized".to_string(),
612                );
613            }
614            Err(RaidServiceError::Operation(AppError::ResourceError(ref msg)))
615                if msg.contains("not found") =>
616            {
617                return create_error_response(
618                    &message,
619                    ErrorCode::InvalidRequest,
620                    format!("RAID node not found: {}", node_id),
621                );
622            }
623            Err(e) => {
624                error!("LeaveCluster delete_node failed: {:?}", e);
625                return create_error_response(
626                    &message,
627                    ErrorCode::ReplicationFailed,
628                    format!("Failed to remove node from membership: {:?}", e),
629                );
630            }
631        }
632
633        let response = LeaveClusterResponse {
634            status: OperationStatus::Success,
635            replication_complete,
636            artifacts_moved,
637        };
638
639        create_success_response(&message, response)
640    }

643/// Compare local artifact IDs with the peer catalog from [`SyncArtifactsPayload::artifact_ids`].
644///
645/// - **Pull** — IDs present locally but absent on the peer (peer should pull from this node).
646/// - **Push** — IDs the peer has that we lack (this node should receive them).
647/// - **Bidirectional** — symmetric difference.
648///
649/// If `remote_ids` is `None`, returns `(local.len(), [])` (no peer catalog to diff).
650fn diff_sync_catalog(
651    direction: SyncDirection,
652    local: &[ArtifactRef],
653    remote_ids: Option<&[String]>,
654) -> (u32, Vec<String>) {
655    let local_set: HashSet<String> = local.iter().map(|a| a.id.to_string()).collect();
656    let Some(remote_slice) = remote_ids else {
657        return (local.len() as u32, Vec::new());
658    };
659    let remote_set: HashSet<String> = remote_slice.iter().cloned().collect();
660    let synced = local_set.intersection(&remote_set).count() as u32;
661    let mut missing: Vec<String> = match direction {
662        SyncDirection::Pull => local_set.difference(&remote_set).cloned().collect(),
663        SyncDirection::Push => remote_set.difference(&local_set).cloned().collect(),
664        SyncDirection::Bidirectional => local_set
665            .symmetric_difference(&remote_set)
666            .cloned()
667            .collect(),
668    };
669    missing.sort();
670    (synced, missing)
671}

673fn create_success_response<T: serde::Serialize>(
674    original_message: &ProtocolMessage,
675    payload: T,
676) -> Response {
677    let response_message = ProtocolMessage {
678        message_type: format!("{}_response", original_message.message_type),
679        id: Uuid::new_v4().to_string(),
680        timestamp: Utc::now(),
681        // Future improvement: Use actual node ID instead of copying from request
682        // 1. Get node ID from RaidManager using manager.get_node_id().await
683        //    - This returns the actual node ID from Raft configuration
684        //    - More reliable than trusting client-provided node_id
685        // 2. Handle errors gracefully if node ID cannot be retrieved
686        //    - Fallback to original_message.node_id if get_node_id fails
687        // Example: node_id: manager.get_node_id().await.to_string()
688        // Note: For now, using original_message.node_id for compatibility
689        node_id: original_message.node_id.clone(),
690        payload: serde_json::to_value(payload).unwrap_or(serde_json::Value::Null),
691    };
692
693    (StatusCode::OK, Json(response_message)).into_response()
694}
695
696fn create_error_response(
697    original_message: &ProtocolMessage,
698    error_code: ErrorCode,
699    error_message: String,
700) -> Response {
701    let error = ProtocolError {
702        error_code,
703        error_message,
704        details: None,
705    };
706
707    let response_message = ProtocolMessage {
708        message_type: "error".to_string(),
709        id: Uuid::new_v4().to_string(),
710        timestamp: Utc::now(),
711        node_id: original_message.node_id.clone(),
712        payload: serde_json::to_value(error).unwrap_or(serde_json::Value::Null),
713    };
714
715    (StatusCode::BAD_REQUEST, Json(response_message)).into_response()
716}

718fn base64_decode(data: &str) -> Result<Vec<u8>, AppError> {
719    use base64::{engine::general_purpose, Engine as _};
720    general_purpose::STANDARD
721        .decode(data)
722        .map_err(|e| AppError::ValidationError(format!(
723            "Base64 decode error: {}. Context: Failed to decode base64-encoded artifact data. \
724            Suggestion: Verify the data is valid base64 encoding. Ensure no whitespace or invalid characters are present. \
725            Error details: {}, Data length: {}",
726            e, e, data.len()
727        )))
728}
