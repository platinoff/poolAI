11/// Env: max concurrent active `telegram_edge` workers (Galaxy §3.1 seat cap).
12pub const ENV_TELEGRAM_SEAT_LIMIT: &str = "POOLAI_TELEGRAM_SEAT_LIMIT";
13
14/// Env: `flat` (default) | `bound_wallet_session` seat policy (PH-S486).
15pub const ENV_TELEGRAM_SEAT_POLICY: &str = "POOLAI_TELEGRAM_SEAT_POLICY";
16
17static ACTIVE_TELEGRAM_EDGE: LazyLock<Mutex<HashSet<String>>> =
18    LazyLock::new(|| Mutex::new(HashSet::new()));
19
20/// Whether register-remote metadata marks a `telegram_edge` worker.
21pub fn is_telegram_edge_metadata(metadata: &std::collections::HashMap<String, String>) -> bool {
22    metadata
23        .get("origin")
24        .map(|v| v.trim().eq_ignore_ascii_case("telegram_edge"))
25        .unwrap_or(false)
26        || metadata
27            .get("role")
28            .map(|v| v.trim().eq_ignore_ascii_case("telegram_edge"))
29            .unwrap_or(false)
30}

32/// Configured seat limit from env; `None` when unset (no cap).
33pub fn telegram_seat_limit_from_env() -> Option<u32> {
34    std::env::var(ENV_TELEGRAM_SEAT_LIMIT)
35        .ok()
36        .and_then(|v| v.trim().parse().ok())
37        .filter(|&n| n > 0)
38}
39
40/// Seat policy from env (PH-S486).
41pub fn telegram_seat_policy_from_env() -> TelegramSeatPolicy {
42    match std::env::var(ENV_TELEGRAM_SEAT_POLICY)
43        .unwrap_or_default()
44        .trim()
45        .to_ascii_lowercase()
46        .as_str()
47    {
48        "bound_wallet_session" | "bound-wallet-session" => TelegramSeatPolicy::BoundWalletSession,
49        _ => TelegramSeatPolicy::Flat,
50    }
51}
52
53#[derive(Debug, Clone, Copy, PartialEq, Eq)]
54pub enum TelegramSeatPolicy {
55    Flat,
56    BoundWalletSession,
57}
58
59/// Effective seat cap: `min(admin_max, bound_wallets)` under bound_wallet_session (PH-S486).
60pub fn compute_seat_limit(admin_max: u32, bound_wallets: usize) -> u32 {
61    match telegram_seat_policy_from_env() {
62        TelegramSeatPolicy::Flat => admin_max,
63        TelegramSeatPolicy::BoundWalletSession => admin_max.min(bound_wallets.max(1) as u32),
64    }
65}
66
67fn effective_seat_limit() -> Option<u32> {
68    let admin_max = telegram_seat_limit_from_env()?;
69    let bound = VirtualNodeTelegramBindingService::list().len();
70    Some(compute_seat_limit(admin_max, bound))
71}

73/// Admit one `telegram_edge` peer; returns `Err(())` when seat cap exhausted.
74#[allow(clippy::result_unit_err)]
75pub fn try_admit_telegram_edge(peer_id: &str) -> Result<(), ()> {
76    let Some(limit) = effective_seat_limit() else {
77        return Ok(());
78    };
79    let mut active = ACTIVE_TELEGRAM_EDGE
80        .lock()
81        .unwrap_or_else(|e| e.into_inner());
82    if active.contains(peer_id) {
83        return Ok(());
84    }
85    if active.len() >= limit as usize {
86        return Err(());
87    }
88    active.insert(peer_id.to_string());
89    Ok(())
90}
91
92/// Active telegram_edge seat count (in-process stub).
93pub fn active_telegram_edge_seats() -> usize {
94    ACTIVE_TELEGRAM_EDGE.lock().map(|g| g.len()).unwrap_or(0)
95}

97/// Coordinator seat snapshot for read API (PH-S505, Galaxy §3.1).
98#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
99pub struct TelegramSeatCoordinatorSnapshot {
100    pub seat_policy: String,
101    pub admin_max_seats: Option<u32>,
102    pub bound_wallets_count: usize,
103    pub seat_limit: Option<u32>,
104    pub active_telegram_edge_workers: usize,
105}
106
107pub fn telegram_seat_coordinator_snapshot() -> TelegramSeatCoordinatorSnapshot {
108    let policy = match telegram_seat_policy_from_env() {
109        TelegramSeatPolicy::Flat => "flat",
110        TelegramSeatPolicy::BoundWalletSession => "bound_wallet_session",
111    };
112    let admin_max = telegram_seat_limit_from_env();
113    let bound_wallets = VirtualNodeTelegramWalletService::list().len();
114    let seat_limit = admin_max.map(|max| compute_seat_limit(max, bound_wallets));
115    TelegramSeatCoordinatorSnapshot {
116        seat_policy: policy.to_string(),
117        admin_max_seats: admin_max,
118        bound_wallets_count: bound_wallets,
119        seat_limit,
120        active_telegram_edge_workers: active_telegram_edge_seats(),
121    }
122}

124#[cfg(any(test, feature = "test-utils"))]
125pub fn reset_telegram_seats_for_test() {
126    if let Ok(mut active) = ACTIVE_TELEGRAM_EDGE.lock() {
127        active.clear();
128    }
129}

137    fn try_admit_telegram_edge_ph_s475() {
138        static ENV_LOCK: std::sync::Mutex<()> = std::sync::Mutex::new(());
139        let _guard = ENV_LOCK.lock().unwrap_or_else(|e| e.into_inner());
140        reset_telegram_seats_for_test();
141        std::env::set_var(ENV_TELEGRAM_SEAT_LIMIT, "1");
142        assert!(try_admit_telegram_edge("peer-a").is_ok());

144        assert!(try_admit_telegram_edge("peer-b").is_err());
145        assert!(try_admit_telegram_edge("peer-a").is_ok());
146        std::env::remove_var(ENV_TELEGRAM_SEAT_LIMIT);
