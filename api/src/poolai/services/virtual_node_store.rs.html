1//! File-backed store for virtual-node coordinator state (FM-016+).
2//!
3//! Set `POOLAI_VIRTUAL_NODE_DATA_DIR` (e.g. `data/virtual_nodes`) to persist bindings and
4//! task queues across coordinator restarts. When unset, only in-memory state is used.
5
6use serde::Serialize;
7use serde_json::Value;
8use std::fs;
9use std::path::{Path, PathBuf};
10
11pub fn data_dir() -> Option<PathBuf> {
12    std::env::var("POOLAI_VIRTUAL_NODE_DATA_DIR")
13        .ok()
14        .filter(|s| !s.trim().is_empty())
15        .map(PathBuf::from)
16}
17
18pub fn load_json(relative: &str) -> Result<Value, String> {
19    let path = data_dir()
20        .ok_or_else(|| "POOLAI_VIRTUAL_NODE_DATA_DIR not set".to_string())?
21        .join(relative);
22    if !path.exists() {
23        return Ok(Value::Null);
24    }
25    let bytes = fs::read(&path).map_err(|e| format!("read {}: {e}", path.display()))?;
26    serde_json::from_slice(&bytes).map_err(|e| format!("parse {}: {e}", path.display()))
27}
28
29pub fn save_json<T: Serialize>(relative: &str, value: &T) -> Result<(), String> {
30    let dir = data_dir().ok_or_else(|| "POOLAI_VIRTUAL_NODE_DATA_DIR not set".to_string())?;
31    let path = dir.join(relative);
32    write_json_atomic(&path, value)
33}
34
35pub fn peer_tasks_path(peer_id: &str) -> Option<PathBuf> {
36    data_dir().map(|d| d.join("tasks").join(format!("{peer_id}.json")))
37}
38
39pub fn write_json_atomic<T: Serialize>(path: &Path, value: &T) -> Result<(), String> {
40    if let Some(parent) = path.parent() {
41        fs::create_dir_all(parent).map_err(|e| format!("mkdir {}: {e}", parent.display()))?;
42    }
43    let data = serde_json::to_vec_pretty(value)
44        .map_err(|e| format!("serialize {}: {e}", path.display()))?;
45    let tmp = path.with_extension("json.tmp");
46    fs::write(&tmp, &data).map_err(|e| format!("write {}: {e}", tmp.display()))?;
47    fs::rename(&tmp, path).map_err(|e| format!("rename {}: {e}", path.display()))
48}
