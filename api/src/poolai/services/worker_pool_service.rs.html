3use crate::core::error::AppError;

11/// Dashboard-friendly worker row (matches admin UI expectations).
12#[derive(Debug, Clone, Serialize)]
13pub struct WorkerInfo {
14    pub id: String,
15    /// High-level state for dashboards: `idle`, `busy`, or `error`.
16    pub status: String,
17    pub current_task: Option<String>,
18    /// Matches admin UI and detailed panels (pool `WorkerStatus`).
19    pub is_healthy: bool,
20    pub total_requests_processed: u64,
21    pub queue_size: usize,
22    pub active_connections: usize,
23    pub average_response_time_ms: f64,
24}
25
26/// Validated fields for registering a worker in the pool (HTTP layer maps requests here).
27#[derive(Debug, Clone)]
28pub struct CreateWorkerInput {
29    pub worker_id: String,
30    pub max_concurrent_requests: usize,
31    pub request_timeout_ms: u64,
32    pub health_check_interval_ms: u64,
33    pub enable_caching: bool,
34    pub cache_size: usize,
35    pub max_memory_mb: usize,
36    pub cpu_priority: u8,
37    pub gpu_device: Option<usize>,
38    pub auto_restart: bool,
39    pub resource_monitoring: bool,
40}
41
42#[derive(Debug)]
43pub enum AddWorkerError {
44    PoolNotReady,
45    Operation(AppError),
46}
47
48#[derive(Debug)]
49pub enum RemoveWorkerError {
50    PoolNotReady,
51    Operation(AppError),
52}
53
54pub struct WorkerPoolService;
55
56impl WorkerPoolService {
57    fn status_label(is_healthy: bool, active_connections: usize) -> String {
58        match is_healthy {
59            true => {
60                if active_connections > 0 {
61                    "busy".to_string()
62                } else {
63                    "idle".to_string()
64                }
65            }
66            false => "error".to_string(),
67        }
68    }
69
70    fn mock_workers() -> Vec<WorkerInfo> {
71        vec![
72            WorkerInfo {
73                id: "worker-1".to_string(),
74                status: "busy".to_string(),
75                current_task: Some("text-generation".to_string()),
76                is_healthy: true,
77                total_requests_processed: 128,
78                queue_size: 0,
79                active_connections: 1,
80                average_response_time_ms: 24.5,
81            },
82            WorkerInfo {
83                id: "worker-2".to_string(),
84                status: "idle".to_string(),
85                current_task: None,
86                is_healthy: true,
87                total_requests_processed: 64,
88                queue_size: 0,
89                active_connections: 0,
90                average_response_time_ms: 18.0,
91            },
92        ]
93    }
94
95    pub async fn list_workers(ctx: &ApiContext) -> Vec<WorkerInfo> {
96        if let Some(pool) = ctx.pool.get() {
97            let worker_statuses = {
98                let pool_guard = pool.read().await;
99                pool_guard.get_worker_status().await
100            };
101            if !worker_statuses.is_empty() {
102                return worker_statuses
103                    .iter()
104                    .map(|(id, status)| WorkerInfo {
105                        id: id.clone(),
106                        status: Self::status_label(status.is_healthy, status.active_connections),
107                        current_task: status.current_task.clone(),
108                        is_healthy: status.is_healthy,
109                        total_requests_processed: status.total_requests_processed,
110                        queue_size: status.queue_size,
111                        active_connections: status.active_connections,
112                        average_response_time_ms: status.average_response_time_ms,
113                    })
114                    .collect();
115            }
116        }
117        Self::mock_workers()
118    }
119
120    pub async fn add_worker(
121        ctx: &ApiContext,
122        input: CreateWorkerInput,
123    ) -> Result<(), AddWorkerError> {
124        let pool: Arc<TokioRwLock<Pool>> = ctx
125            .pool
126            .get()
127            .cloned()
128            .ok_or(AddWorkerError::PoolNotReady)?;
129        let worker_config = pool::worker::WorkerConfig {
130            worker_id: input.worker_id.clone(),
131            max_concurrent_requests: input.max_concurrent_requests,
132            request_timeout_ms: input.request_timeout_ms,
133            health_check_interval_ms: input.health_check_interval_ms,
134            enable_caching: input.enable_caching,
135            cache_size: input.cache_size,
136            max_memory_mb: input.max_memory_mb,
137            cpu_priority: input.cpu_priority,
138            gpu_device: input.gpu_device,
139            auto_restart: input.auto_restart,
140            resource_monitoring: input.resource_monitoring,
141        };
142        let worker = pool::worker::Worker::new(worker_config);
143        let pool_guard = pool.write().await;
144        pool_guard
145            .add_worker(input.worker_id, worker)
146            .await
147            .map_err(AddWorkerError::Operation)?;
148        Ok(())
149    }
150
151    pub async fn remove_worker(ctx: &ApiContext, worker_id: &str) -> Result<(), RemoveWorkerError> {
152        let pool: Arc<TokioRwLock<Pool>> = ctx
153            .pool
154            .get()
155            .cloned()
156            .ok_or(RemoveWorkerError::PoolNotReady)?;
157        let pool_guard = pool.write().await;
158        pool_guard
159            .remove_worker(worker_id)
160            .await
161            .map_err(RemoveWorkerError::Operation)?;
162        Ok(())
163    }
