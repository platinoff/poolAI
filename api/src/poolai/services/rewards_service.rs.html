1//! Rewards-facing operations for the HTTP API.
2//!
3//! Uses [`crate::core::state::AppState::rewards_engine`] after bootstrap; falls back to
4//! [`crate::rewards::shared_reward_engine`] so lightweight tests with `ApiContext::default()`
5//! still hit the same process-wide engine.
6
7use crate::core::state::ApiContext;
8use crate::rewards::{shared_reward_engine, Reward, RewardSystem, UserProgress};
9use std::collections::HashMap;
10use std::sync::Arc;
11
12/// Default leaderboard size for `GET /rewards/top`.
13pub const TOP_USERS_DEFAULT_LIMIT: usize = 10;
14
15fn engine(ctx: &ApiContext) -> Arc<RewardSystem> {
16    ctx.rewards_engine
17        .get()
18        .cloned()
19        .unwrap_or_else(|| shared_reward_engine())
20}
21
22pub struct RewardsService;
23
24impl RewardsService {
25    pub async fn reward_statistics(ctx: &ApiContext) -> HashMap<String, f64> {
26        engine(ctx).get_reward_statistics().await
27    }
28
29    pub async fn user_rewards(ctx: &ApiContext, user_id: &str) -> Vec<Reward> {
30        engine(ctx).get_user_rewards(user_id).await
31    }
32
33    pub async fn user_progress(ctx: &ApiContext, user_id: &str) -> Option<UserProgress> {
34        engine(ctx).get_user_progress(user_id).await
35    }
36
37    pub async fn top_users(ctx: &ApiContext, limit: usize) -> Vec<(String, f64)> {
38        engine(ctx).get_top_users(limit).await
39    }
40}
