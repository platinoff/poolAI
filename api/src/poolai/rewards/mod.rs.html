1//! Rewards module for Stage 3 - Endorphin-based reward system
2//!
3//! Provides a comprehensive reward system with multiple reward types, user progress
4//! tracking, achievement-based rewards, and reward history/statistics.
5//!
6//! # Features
7//!
8//! - **Multiple Reward Types**: Performance, Efficiency, Quality, Innovation, Collaboration, Maintenance
9//! - **Reward Levels**: Bronze, Silver, Gold, Platinum, Diamond with multipliers
10//! - **User Progress Tracking**: Total rewards, streaks, achievements, levels, experience
11//! - **Achievement System**: Automatic achievement unlocking based on milestones
12//! - **Statistics**: Reward statistics and top users leaderboard
13//!
14//! # Example
15//!
16//! ```no_run
17//! use poolai::rewards::{create_reward, get_user_progress, RewardType, RewardLevel};
18//! use std::collections::HashMap;
19//!
20//! # async fn example() {
21//! // Create a reward for a user
22//! let reward = create_reward(
23//!     "user123".to_string(),
24//!     RewardType::Performance,
25//!     RewardLevel::Gold,
26//!     100.0,
27//!     "Outstanding performance".to_string(),
28//!     HashMap::new(),
29//! ).await;
30//!
31//! println!("Reward amount: {}", reward.amount);
32//!
33//! // Get user progress
34//! if let Some(progress) = get_user_progress("user123").await {
35//!     println!("Total rewards: {}", progress.total_rewards);
36//!     println!("Level: {}", progress.level);
37//!     println!("Achievements: {:?}", progress.achievements);
38//! }
39//! # }
40//! ```

48/// Reward types
49///
50/// Different types of rewards that can be awarded to users based on their activities.
51///
52/// # Example
53///
54/// ```rust
55/// use poolai::rewards::RewardType;
56///
57/// let reward_type = RewardType::Performance;
58/// match reward_type {
59///     RewardType::Performance => println!("High performance reward"),
60///     RewardType::Efficiency => println!("Efficient resource usage"),
61///     RewardType::Quality => println!("Quality result"),
62///     RewardType::Innovation => println!("Innovative solution"),
63///     RewardType::Collaboration => println!("Great collaboration"),
64///     RewardType::Maintenance => println!("System maintenance"),
65/// }
66/// ```
67#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq, Hash)]
68pub enum RewardType {
69    Performance,   // For high performance
70    Efficiency,    // For efficient resource usage
71    Quality,       // For result quality
72    Innovation,    // For innovative solutions
73    Collaboration, // For collaboration
74    Maintenance,   // For system maintenance
75}

77/// Reward levels
78///
79/// Different levels of rewards with increasing multipliers:
80/// - Bronze: 1x base reward
81/// - Silver: 2x base reward
82/// - Gold: 3x base reward
83/// - Platinum: 4x base reward
84/// - Diamond: 5x base reward
85///
86/// # Example
87///
88/// ```rust
89/// use poolai::rewards::RewardLevel;
90///
91/// let level = RewardLevel::Gold;
92/// let multiplier = level as u8 as f64; // 3.0
93/// println!("Level multiplier: {}", multiplier);
94/// ```
95#[derive(Debug, Clone, Copy, Serialize, Deserialize, PartialEq, PartialOrd)]
96pub enum RewardLevel {
97    Bronze = 1,   // 1x base reward
98    Silver = 2,   // 2x base reward
99    Gold = 3,     // 3x base reward
100    Platinum = 4, // 4x base reward
101    Diamond = 5,  // 5x base reward
102}

104/// Reward structure
105///
106/// Represents a single reward awarded to a user with type, level, amount,
107/// description, timestamp, and optional metadata.
108///
109/// # Example
110///
111/// ```rust
112/// use poolai::rewards::{Reward, RewardType, RewardLevel};
113/// use chrono::Utc;
114/// use std::collections::HashMap;
115///
116/// let reward = Reward {
117///     id: "reward-123".to_string(),
118///     user_id: "user-456".to_string(),
119///     reward_type: RewardType::Performance,
120///     level: RewardLevel::Gold,
121///     amount: 450.0,
122///     description: "Outstanding performance".to_string(),
123///     timestamp: Utc::now(),
124///     metadata: HashMap::new(),
125/// };
126/// ```
127#[derive(Debug, Clone, Serialize, Deserialize)]
128pub struct Reward {
129    pub id: String,
130    pub user_id: String,
131    pub reward_type: RewardType,
132    pub level: RewardLevel,
133    pub amount: f64,
134    pub description: String,
135    pub timestamp: DateTime<Utc>,
136    pub metadata: HashMap<String, String>,
137}

139/// User progress structure
140///
141/// Tracks a user's progress including total rewards, streaks, achievements,
142/// level, and experience points.
143///
144/// # Example
145///
146/// ```rust
147/// use poolai::rewards::UserProgress;
148/// use chrono::Utc;
149///
150/// let progress = UserProgress {
151///     user_id: "user-123".to_string(),
152///     total_rewards: 1000.0,
153///     reward_count: 10,
154///     current_streak: 5,
155///     longest_streak: 7,
156///     last_reward_date: Some(Utc::now()),
157///     achievements: vec!["First Decade".to_string()],
158///     level: 5,
159///     experience: 5000,
160/// };
161/// ```
162#[derive(Debug, Clone, Serialize, Deserialize)]
163pub struct UserProgress {
164    pub user_id: String,
165    pub total_rewards: f64,
166    pub reward_count: u32,
167    pub current_streak: u32,
168    pub longest_streak: u32,
169    pub last_reward_date: Option<DateTime<Utc>>,
170    pub achievements: Vec<String>,
171    pub level: u32,
172    pub experience: u64,
173}

175/// Reward system
176///
177/// Manages rewards, user progress, and achievements. Provides methods for
178/// creating rewards, tracking progress, and generating statistics.
179///
180/// # Example
181///
182/// ```no_run
183/// use poolai::rewards::RewardSystem;
184///
185/// # async fn example() {
186/// let system = RewardSystem::new();
187/// // System is automatically initialized and ready to use
188/// # }
189/// ```
190pub struct RewardSystem {
191    rewards: Arc<RwLock<HashMap<String, Reward>>>,
192    user_progress: Arc<RwLock<HashMap<String, UserProgress>>>,
193    _reward_multipliers: Arc<RwLock<HashMap<RewardType, f64>>>,
194}
195
196impl Default for RewardSystem {
197    fn default() -> Self {
198        Self::new()
199    }
200}

202impl RewardSystem {
203    pub fn new() -> Self {
204        let mut multipliers = HashMap::new();
205        multipliers.insert(RewardType::Performance, 1.5);
206        multipliers.insert(RewardType::Efficiency, 1.3);
207        multipliers.insert(RewardType::Quality, 1.4);
208        multipliers.insert(RewardType::Innovation, 2.0);
209        multipliers.insert(RewardType::Collaboration, 1.2);
210        multipliers.insert(RewardType::Maintenance, 1.1);
211
212        Self {
213            rewards: Arc::new(RwLock::new(HashMap::new())),
214            user_progress: Arc::new(RwLock::new(HashMap::new())),
215            _reward_multipliers: Arc::new(RwLock::new(multipliers)),
216        }
217    }

219    // Create a new reward
220    pub async fn create_reward(
221        &self,
222        user_id: String,
223        reward_type: RewardType,
224        level: RewardLevel,
225        base_amount: f64,
226        description: String,
227        metadata: HashMap<String, String>,
228    ) -> Reward {
229        let multiplier = self.get_reward_multiplier(&reward_type).await;
230        let amount = base_amount * multiplier * (level as u8 as f64);
231
232        let reward = Reward {
233            id: uuid::Uuid::new_v4().to_string(),
234            user_id: user_id.clone(),
235            reward_type: reward_type.clone(),
236            level: level,
237            amount,
238            description,
239            timestamp: Utc::now(),
240            metadata,
241        };
242
243        // Save the reward
244        let mut rewards = self.rewards.write().await;
245        rewards.insert(reward.id.clone(), reward.clone());
246
247        // Update user progress
248        self.update_user_progress(&user_id, &reward).await;
249
250        reward
251    }
252
253    // Get reward multiplier
254    async fn get_reward_multiplier(&self, reward_type: &RewardType) -> f64 {
255        let multipliers = self._reward_multipliers.read().await;
256        *multipliers.get(reward_type).unwrap_or(&1.0)
257    }
258
259    // Update user progress
260    async fn update_user_progress(&self, user_id: &str, reward: &Reward) {
261        let mut progress_map = self.user_progress.write().await;
262
263        let progress = progress_map
264            .entry(user_id.to_string())
265            .or_insert_with(|| UserProgress {
266                user_id: user_id.to_string(),
267                total_rewards: 0.0,
268                reward_count: 0,
269                current_streak: 0,
270                longest_streak: 0,
271                last_reward_date: None,
272                achievements: Vec::new(),
273                level: 1,
274                experience: 0,
275            });
276
277        // Update total amount and reward count
278        progress.total_rewards += reward.amount;
279        progress.reward_count += 1;
280
281        // Update streak
282        let now = Utc::now();
283        if let Some(last_date) = progress.last_reward_date {
284            let days_diff = (now - last_date).num_days();
285            if days_diff <= 1 {
286                progress.current_streak += 1;
287                if progress.current_streak > progress.longest_streak {
288                    progress.longest_streak = progress.current_streak;
289                }
290            } else {
291                progress.current_streak = 1;
292            }
293        } else {
294            progress.current_streak = 1;
295        }
296
297        progress.last_reward_date = Some(now);
298
299        // Update experience and level
300        progress.experience += (reward.amount * 100.0) as u64;
301        progress.level = (progress.experience / 1000) as u32 + 1;
302
303        // Check achievements
304        self.check_achievements(progress).await;
305    }

307    // Check achievements
308    async fn check_achievements(&self, progress: &mut UserProgress) {
309        let mut new_achievements = Vec::new();
310
311        // Achievements for reward count
312        if progress.reward_count >= 10
313            && !progress.achievements.contains(&"First Decade".to_string())
314        {
315            new_achievements.push("First Decade".to_string());
316        }
317        if progress.reward_count >= 50
318            && !progress.achievements.contains(&"Half Century".to_string())
319        {
320            new_achievements.push("Half Century".to_string());
321        }
322        if progress.reward_count >= 100 && !progress.achievements.contains(&"Century".to_string()) {
323            new_achievements.push("Century".to_string());
324        }
325
326        // Achievements for streak
327        if progress.current_streak >= 7
328            && !progress.achievements.contains(&"Week Warrior".to_string())
329        {
330            new_achievements.push("Week Warrior".to_string());
331        }
332        if progress.current_streak >= 30
333            && !progress
334                .achievements
335                .contains(&"Monthly Master".to_string())
336        {
337            new_achievements.push("Monthly Master".to_string());
338        }
339
340        // Achievements for level
341        if progress.level >= 5 && !progress.achievements.contains(&"Level 10".to_string()) {
342            new_achievements.push("Level 5".to_string());
343        }
344        if progress.level >= 10 && !progress.achievements.contains(&"Level 10".to_string()) {
345            new_achievements.push("Level 10".to_string());
346        }
347
348        // Add new achievements
349        progress.achievements.extend(new_achievements);
350    }

352    // Get user rewards
353    pub async fn get_user_rewards(&self, user_id: &str) -> Vec<Reward> {
354        let rewards = self.rewards.read().await;
355        rewards
356            .values()
357            .filter(|r| r.user_id == user_id)
358            .cloned()
359            .collect()
360    }

362    // Get user progress
363    pub async fn get_user_progress(&self, user_id: &str) -> Option<UserProgress> {
364        let progress = self.user_progress.read().await;
365        progress.get(user_id).cloned()
366    }

368    // Get reward statistics
369    pub async fn get_reward_statistics(&self) -> HashMap<String, f64> {
370        let rewards = self.rewards.read().await;
371        let mut stats = HashMap::new();
372
373        for reward in rewards.values() {
374            let key = format!("{:?}_{:?}", reward.reward_type, reward.level);
375            *stats.entry(key).or_insert(0.0) += reward.amount;
376        }
377
378        stats
379    }

381    // Get top users
382    pub async fn get_top_users(&self, limit: usize) -> Vec<(String, f64)> {
383        let progress = self.user_progress.read().await;
384        let mut users: Vec<(String, f64)> = progress
385            .values()
386            .map(|p| (p.user_id.clone(), p.total_rewards))
387            .collect();
388
389        users.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap());
390        users.truncate(limit);
391        users
392    }

394    // Clean up old rewards
395    pub async fn cleanup_old_rewards(&self, days_old: i64) {
396        let cutoff_date = Utc::now() - chrono::Duration::days(days_old);
397        let mut rewards = self.rewards.write().await;
398
399        rewards.retain(|_, reward| reward.timestamp > cutoff_date);
400    }

408/// Create a new reward for a user
409///
410/// Creates a reward with the specified type, level, and base amount.
411/// The final amount is calculated using reward type multiplier and level multiplier.
412///
413/// # Arguments
414///
415/// * `user_id` - ID of the user receiving the reward
416/// * `reward_type` - Type of reward (Performance, Efficiency, etc.)
417/// * `level` - Reward level (Bronze, Silver, Gold, etc.)
418/// * `base_amount` - Base reward amount before multipliers
419/// * `description` - Description of the reward
420/// * `metadata` - Optional metadata as key-value pairs
421///
422/// # Returns
423///
424/// Returns the created `Reward` with calculated amount.
425///
426/// # Example
427///
428/// ```no_run
429/// use poolai::rewards::{create_reward, RewardType, RewardLevel};
430/// use std::collections::HashMap;
431///
432/// # async fn example() {
433/// let reward = create_reward(
434///     "user123".to_string(),
435///     RewardType::Performance,
436///     RewardLevel::Gold,
437///     100.0,
438///     "Outstanding performance".to_string(),
439///     HashMap::new(),
440/// ).await;
441///
442/// // Performance multiplier = 1.5, Gold level = 3
443/// // Final amount = 100.0 * 1.5 * 3.0 = 450.0
444/// println!("Reward amount: {}", reward.amount);
445/// # }
446/// ```
447pub async fn create_reward(
448    user_id: String,
449    reward_type: RewardType,
450    level: RewardLevel,
451    base_amount: f64,
452    description: String,
453    metadata: HashMap<String, String>,
454) -> Reward {
455    REWARD_SYSTEM
456        .create_reward(
457            user_id,
458            reward_type,
459            level,
460            base_amount,
461            description,
462            metadata,
463        )
464        .await
465}

577/// Award performance bonus automatically
578///
579/// Automatically awards a performance bonus to a user based on their performance score.
580/// Awards Gold level for scores >= 0.9 and Silver level for scores >= 0.8.
581///
582/// # Arguments
583///
584/// * `user_id` - ID of the user
585/// * `performance_score` - Performance score (0.0 - 1.0)
586///
587/// # Example
588///
589/// ```no_run
590/// use poolai::rewards::award_performance_bonus;
591///
592/// # async fn example() {
593/// // Award bonus for high performance
594/// award_performance_bonus("user123".to_string(), 0.95).await;
595/// # }
596/// ```
597pub async fn award_performance_bonus(user_id: String, performance_score: f64) {
598    if performance_score >= 0.9 {
599        let metadata = HashMap::new();
600        create_reward(
601            user_id,
602            RewardType::Performance,
603            RewardLevel::Gold,
604            100.0,
605            "Outstanding Performance Bonus".to_string(),
606            metadata,
607        )
608        .await;
609    } else if performance_score >= 0.8 {
610        let metadata = HashMap::new();
611        create_reward(
612            user_id,
613            RewardType::Performance,
614            RewardLevel::Silver,
615            50.0,
616            "Good Performance Bonus".to_string(),
617            metadata,
618        )
619        .await;
620    }
621}
