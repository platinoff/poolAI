1//! UI Themes module
2//!
3//! Provides theme customization support for the PoolAI dashboard.
4//! Supports light/dark themes and custom color schemes.
5//!
6//! Structural design tokens (spacing, typography, shadows) live in [`DESIGN_TOKENS_CSS`].
7
8/// Shared structural CSS variables (PH-S09 design system).
9pub const DESIGN_TOKENS_CSS: &str = include_str!("design_tokens.css");
10
11/// Theme configuration
12#[derive(Debug, Clone)]
13pub struct Theme {
14    pub name: &'static str,
15    pub background: &'static str,
16    pub surface: &'static str,
17    pub surface_secondary: &'static str,
18    pub text: &'static str,
19    pub text_muted: &'static str,
20    pub border: &'static str,
21    pub primary: &'static str,
22    pub primary_hover: &'static str,
23    pub danger: &'static str,
24    pub danger_hover: &'static str,
25    pub secondary: &'static str,
26    pub secondary_hover: &'static str,
27    pub success: &'static str,
28    pub warning: &'static str,
29    pub info: &'static str,
30    pub link: &'static str,
31    pub link_hover: &'static str,
32}

100impl Theme {
101    /// Generate CSS variables for the theme with comprehensive design tokens
102    pub fn to_css_variables(&self) -> String {
103        format!(
104            r#"
105  :root {{
106    /* Theme Identity */
107    --theme-name: "{}";
108    
109    /* Color System */
110    --bg: {};
111    --surface: {};
112    --surface-secondary: {};
113    --text: {};
114    --text-muted: {};
115    --border: {};
116    --primary: {};
117    --primary-hover: {};
118    --danger: {};
119    --danger-hover: {};
120    --secondary: {};
121    --secondary-hover: {};
122    --success: {};
123    --warning: {};
124    --info: {};
125    --link: {};
126    --link-hover: {};
127  }}
128"#,
129            self.name,
130            self.background,
131            self.surface,
132            self.surface_secondary,
133            self.text,
134            self.text_muted,
135            self.border,
136            self.primary,
137            self.primary_hover,
138            self.danger,
139            self.danger_hover,
140            self.secondary,
141            self.secondary_hover,
142            self.success,
143            self.warning,
144            self.info,
145            self.link,
146            self.link_hover
147        )
148    }
149
150    /// Generate CSS with theme variables applied
151    pub fn to_css(&self) -> String {
152        format!(
153            r#"
154{design_tokens}
155{theme_vars}
156  body {{
157    background: var(--bg);
158    color: var(--text);
159  }}
160  .card {{
161    background: var(--surface);
162    border-color: var(--border);
163  }}
164  .btn-primary {{
165    background: var(--primary);
166    border-color: var(--primary);
167    color: var(--bg);
168  }}
169  [data-poolai-theme="high-contrast"] .btn-primary {{
170    background: #006600;
171    border-color: #00ff00;
172    color: #ffffff;
173  }}
174  [data-poolai-theme="high-contrast"] .btn-danger {{
175    background: #cc0000;
176    border-color: #ff0000;
177    color: #ffffff;
178  }}
179  .btn-primary:hover {{
180    background: var(--primary-hover);
181  }}
182  .btn-danger {{
183    background: var(--danger);
184    border-color: var(--danger);
185    color: #fff;
186  }}
187  .btn-danger:hover {{
188    background: var(--danger-hover);
189  }}
190  .btn-secondary {{
191    background: var(--secondary);
192    border-color: var(--secondary);
193    color: var(--bg);
194  }}
195  .btn-secondary:hover {{
196    background: var(--secondary-hover);
197  }}
198  .pill-success {{
199    background: var(--success);
200    color: var(--bg);
201    border-color: var(--success);
202  }}
203  .pill-error {{
204    background: var(--danger);
205    color: #fff;
206    border-color: var(--danger);
207  }}
208  .pill-warning {{
209    background: var(--warning);
210    color: var(--bg);
211    border-color: var(--warning);
212  }}
213  .pill-info {{
214    background: var(--info);
215    color: var(--bg);
216    border-color: var(--info);
217  }}
218  a {{
219    color: var(--link);
220  }}
221  a:hover {{
222    color: var(--link-hover);
223  }}
224  .muted {{
225    color: var(--text-muted);
226  }}
227  .topbar {{
228    background: var(--surface);
229    border-color: var(--border);
230  }}
231  .brand h1 {{
232    color: var(--primary);
233  }}
234"#,
235            design_tokens = DESIGN_TOKENS_CSS,
236            theme_vars = self.to_css_variables(),
237        )
238    }
239}

248        assert!(DESIGN_TOKENS_CSS.contains("--font-size-base: 16px"));

259/// Get theme by name
260pub fn get_theme(name: &str) -> &'static Theme {
261    match name {
262        "light" => &LIGHT_THEME,
263        "high-contrast" => &HIGH_CONTRAST_THEME,
264        _ => &DARK_THEME, // default
265    }
266}
267
268/// Get all available themes
269pub fn get_all_themes() -> Vec<&'static Theme> {
270    vec![&DARK_THEME, &LIGHT_THEME, &HIGH_CONTRAST_THEME]
