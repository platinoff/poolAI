1//! UI module
2//!
3//! Provides web dashboard interface with read/write operations, authentication,
4//! theme customization, and reusable UI components.
5//!
6//! # Features
7//!
8//! - **Dashboard Pages**: Status, health, metrics, workers, libs, VM, RAID
9//! - **Authentication**: JWT-based authentication with role-based access control
10//! - **Write Operations**: Create/delete artifacts and workers through UI
11//! - **Theme Support**: Dark, light, and high-contrast themes
12//! - **Components**: Reusable UI components (buttons, cards, forms, modals)
13//!
14//! # Routes
15//!
16//! - `/ui` - Home page
17//! - `/ui/status` - System status
18//! - `/ui/health` - Health check
19//! - `/ui/metrics` - System metrics
20//! - `/ui/workers` - Worker management
21//! - `/ui/libs` - Library management
22//! - `/ui/vm` - VM instance management
23//! - `/ui/raid` - RAID artifact management
24//!
25//! Concept alignment (planned in `docs/concept/poolAI_concept.txt`):
26//! - Web dashboard (basic)

842// Token validation and refresh
843async function validateToken() {
844  const token = getToken();
845  if (!token) return false;
846  
847  try {
848    // Decode token to check expiration (simple base64 decode for dev tokens)
849    const parts = token.split('.');
850    if (parts.length === 3) {
851      // Real JWT format
852      const payload = JSON.parse(atob(parts[1]));
853      const now = Math.floor(Date.now() / 1000);
854      if (payload.exp && payload.exp < now) {
855        // Token expired, try to refresh
856        return await refreshToken();
857      }
858      return true;
859    } else if (token.startsWith('dev_token_')) {
860      // Dev token format - check expiration from localStorage
861      const tokenData = localStorage.getItem('poolai_token_exp');
862      if (tokenData) {
863        const exp = parseInt(tokenData, 10);
864        const now = Math.floor(Date.now() / 1000);
865        if (exp && exp < now) {
866          return await refreshToken();
867        }
868      }
869      return true;
870    }
871    return false;
872  } catch (e) {
873    console.error('Token validation error:', e);
874    return false;
875  }
876}

878async function refreshToken() {
879  try {
880    const token = getToken();
881    if (!token) return false;
882    
883    // Try to refresh token via API (if endpoint exists)
884    const res = await fetch('/api/v1/refresh', {
885      method: 'POST',
886      headers: getAuthHeaders(),
887    });
888    
889    if (res.ok) {
890      const data = await res.json();
891      setToken(data.token);
892      if (data.role) {
893        const user = getUser();
894        if (user) setUser(user.username, data.role);
895      }
896      return true;
897    }
898    return false;
899  } catch (e) {
900    console.error('Token refresh error:', e);
901    return false;
902  }

1229async function fetchJsonWithRetry(url, options = {}, maxRetries = 3, retryDelay = 1000) {
1230  let lastError;
1231  const errors = [];
1232  
1233  for (let attempt = 0; attempt <= maxRetries; attempt++) {
1234    try {
1235      const res = await fetchJson(url, options);
1236      // Clear any retry notifications on success
1237      if (attempt > 0) {
1238        showNotification(`Request succeeded after ${attempt} ${attempt === 1 ? 'retry' : 'retries'}`, 'success', 2000);
1239      }
1240      return res;
1241    } catch (error) {
1242      lastError = error;
1243      errors.push(error);
1244      
1245      if (attempt < maxRetries) {
1246        // Exponential backoff with notification
1247        const delay = retryDelay * Math.pow(2, attempt);
1248        showNotification(
1249          `Request failed. Retrying in ${(delay / 1000).toFixed(1)}s... (${attempt + 1}/${maxRetries})`,
1250          'warning',
1251          2000
1252        );
1253        await new Promise(resolve => setTimeout(resolve, delay));
1254        continue;
1255      }
1256    }
1257  }
1258  
1259  // Create enhanced error with context
1260  const enhancedError = {
1261    message: lastError.message || 'Request failed after all retries',
1262    originalError: lastError,
1263    attempts: maxRetries + 1,
1264    url: url,
1265    suggestions: [
1266      'Check your internet connection',
1267      'Verify the server is running',
1268      'Try refreshing the page',
1269      'Contact support if the problem persists'
1270    ]
1271  };

1497function sortTable(table, columnIndex, ascending = true) {
1498  const tbody = table.querySelector('tbody');
1499  if (!tbody) return;
1500  
1501  const rows = Array.from(tbody.querySelectorAll('tr:not(.no-results-row)'));
1502  const isNumeric = rows.every(row => {
1503    const cell = row.cells[columnIndex];
1504    return cell && !isNaN(parseFloat(cell.textContent));
1505  });
1506  
1507  rows.sort((a, b) => {
1508    const aCell = a.cells[columnIndex];
1509    const bCell = b.cells[columnIndex];
1510    
1511    if (!aCell || !bCell) return 0;
1512    
1513    const aValue = isNumeric ? parseFloat(aCell.textContent) : aCell.textContent.trim();
1514    const bValue = isNumeric ? parseFloat(bCell.textContent) : bCell.textContent.trim();
1515    
1516    if (aValue < bValue) return ascending ? -1 : 1;
1517    if (aValue > bValue) return ascending ? 1 : -1;
1518    return 0;
1519  });
1520  
1521  // Remove all rows
1522  rows.forEach(row => row.remove());
1523  
1524  // Re-append sorted rows
1525  rows.forEach(row => tbody.appendChild(row));
1526  
1527  // Update sort indicators
1528  const headers = table.querySelectorAll('th');
1529  headers.forEach((header, index) => {
1530    if (index === columnIndex) {
1531      header.setAttribute('data-sort', ascending ? 'asc' : 'desc');
1532    } else {
1533      header.removeAttribute('data-sort');
1534    }
1535  });
1536}
