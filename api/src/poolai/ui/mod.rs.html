1//! UI module
2//!
3//! Provides web dashboard interface with read/write operations, authentication,
4//! theme customization, and reusable UI components.
5//!
6//! # Features
7//!
8//! - **Dashboard Pages**: Status, health, metrics, workers, libs, VM, RAID
9//! - **Authentication**: JWT-based authentication with role-based access control
10//! - **Write Operations**: Create/delete artifacts and workers through UI
11//! - **Theme Support**: Dark, light, and high-contrast themes
12//! - **Components**: Reusable UI components (buttons, cards, forms, modals)
13//!
14//! # Routes
15//!
16//! - `/ui` - Home page
17//! - `/ui/status` - System status
18//! - `/ui/health` - Health check
19//! - `/ui/metrics` - System metrics
20//! - `/ui/workers` - Worker management
21//! - `/ui/libs` - Library management
22//! - `/ui/vm` - VM instance management
23//! - `/ui/raid` - RAID artifact management
24//!
25//! Concept alignment (planned in `docs/concept/poolAI_concept.txt`):
26//! - Web dashboard (basic)
27//! - UI components/themes/layouts (planned)

69pub fn create_ui_routes() -> Router<ApiContext> {
70    let router = Router::new()
71        .route("/", get(home_handler))
72        .route("/auth", get(login_page))
73        .route("/login", get(login_page))
74        .route("/status", get(status_page))
75        .route("/health", get(health_page))
76        .route("/metrics", get(metrics_page))
77        .route("/workers", get(workers_page))
78        .route("/libs", get(libs_page))
79        .route("/vm", get(vm_page))
80        .route("/raid", get(raid_page));
81
82    // Add admin routes if enterprise feature is enabled
83    #[cfg(feature = "enterprise")]
84    {
85        router.merge(create_admin_routes())
86    }
87    #[cfg(not(feature = "enterprise"))]
88    {
89        router
90    }
91}

938// Token validation and refresh
939async function validateToken() {
940  const token = getToken();
941  if (!token) return false;
942  
943  try {
944    // Decode token to check expiration (simple base64 decode for dev tokens)
945    const parts = token.split('.');
946    if (parts.length === 3) {
947      // Real JWT format
948      const payload = JSON.parse(atob(parts[1]));
949      const now = Math.floor(Date.now() / 1000);
950      if (payload.exp && payload.exp < now) {
951        // Token expired, try to refresh
952        return await refreshToken();
953      }
954      return true;
955    } else if (token.startsWith('dev_token_')) {
956      // Dev token format - check expiration from localStorage
957      const tokenData = localStorage.getItem('poolai_token_exp');
958      if (tokenData) {
959        const exp = parseInt(tokenData, 10);
960        const now = Math.floor(Date.now() / 1000);
961        if (exp && exp < now) {
962          return await refreshToken();
963        }
964      }
965      return true;
966    }
967    return false;
968  } catch (e) {
969    console.error('Token validation error:', e);
970    return false;
971  }
972}

977    if (!token) return false;
978

994    return false;

1363async function fetchJsonWithRetry(url, options = {}, maxRetries = 3, retryDelay = 1000) {
1364  let lastError;
1365  const errors = [];
1366  
1367  for (let attempt = 0; attempt <= maxRetries; attempt++) {
1368    try {
1369      const res = await fetchJson(url, options);
1370      // Clear any retry notifications on success
1371      if (attempt > 0) {
1372        const successMsg = T('ui.requestSucceededAfter', 'Request succeeded after {count} {unit}')
1373          .replace('{count}', String(attempt))
1374          .replace('{unit}', attempt === 1 ? T('ui.retryUnit.one', 'retry') : T('ui.retryUnit.many', 'retries'));
1375        showNotification(successMsg, 'success', 2000);
1376      }
1377      return res;
1378    } catch (error) {
1379      lastError = error;
1380      errors.push(error);
1381      
1382      if (attempt < maxRetries) {
1383        // Exponential backoff with notification
1384        const delay = retryDelay * Math.pow(2, attempt);
1385        const retryingMsg = T('ui.requestRetrying', 'Request failed. Retrying in {seconds}s... ({attempt}/{max})')
1386          .replace('{seconds}', (delay / 1000).toFixed(1))
1387          .replace('{attempt}', String(attempt + 1))
1388          .replace('{max}', String(maxRetries));
1389        showNotification(retryingMsg, 'warning', 2000);
1390        await new Promise(resolve => setTimeout(resolve, delay));
1391        continue;
1392      }
1393    }
1394  }
1395  
1396  // Create enhanced error with context
1397  const enhancedError = {
1398    message: lastError.message || T('ui.requestFailedAllRetries', 'Request failed after all retries'),
1399    originalError: lastError,
1400    attempts: maxRetries + 1,
1401    url: url,
1402    suggestions: [
1403      T('ui.suggestion.checkInternet', 'Check your internet connection'),
1404      T('ui.suggestion.verifyServer', 'Verify the server is running'),
1405      T('ui.suggestion.refreshPage', 'Try refreshing the page'),

1407    ]
1408  };
1409

1411}

1651    return cell && !isNaN(parseFloat(cell.textContent));
1652  });
1653  
1654  rows.sort((a, b) => {
1655    const aCell = a.cells[columnIndex];
1656    const bCell = b.cells[columnIndex];
1657    
1658    if (!aCell || !bCell) return 0;
1659    
1660    const aValue = isNumeric ? parseFloat(aCell.textContent) : aCell.textContent.trim();
1661    const bValue = isNumeric ? parseFloat(bCell.textContent) : bCell.textContent.trim();
1662    
1663    if (aValue < bValue) return ascending ? -1 : 1;
1664    if (aValue > bValue) return ascending ? 1 : -1;
1665    return 0;
1666  });
1667  
1668  // Remove all rows
1669  rows.forEach(row => row.remove());
1670  
1671  // Re-append sorted rows
1672  rows.forEach(row => tbody.appendChild(row));
1673  
1674  // Update sort indicators
1675  const headers = table.querySelectorAll('th');
1676  headers.forEach((header, index) => {
1677    if (index === columnIndex) {
1678      header.setAttribute('data-sort', ascending ? 'asc' : 'desc');
1679    } else {
1680      header.removeAttribute('data-sort');
1681    }
1682  });

2346function validateField(input) {
2347  let isValid = true;
2348  let errorMessage = '';
2349  
2350  // Remove previous error message
2351  const existingError = input.parentElement.querySelector('.error-text');
2352  if (existingError) {
2353    existingError.remove();
2354  }
2355  
2356  // Required validation
2357  if (input.hasAttribute('required') && !input.value.trim()) {
2358    isValid = false;
2359    errorMessage = T('form.fieldRequired', 'This field is required');
2360  }
2361  
2362  // Number validation
2363  if (input.type === 'number' && input.value) {
2364    const min = input.getAttribute('min');
2365    const max = input.getAttribute('max');
2366    const value = parseFloat(input.value);
2367    
2368    if (isNaN(value)) {
2369      isValid = false;
2370      errorMessage = T('form.validNumber', 'Please enter a valid number');
2371    } else if (min && value < parseFloat(min)) {
2372      isValid = false;
2373      errorMessage = T('form.valueMin', 'Value must be at least {min}').replace('{min}', min);
2374    } else if (max && value > parseFloat(max)) {
2375      isValid = false;
2376      errorMessage = T('form.valueMax', 'Value must be at most {max}').replace('{max}', max);
2377    }
2378  }
2379  
2380  // Email validation
2381  if (input.type === 'email' && input.value) {
2382    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
2383    if (!emailRegex.test(input.value)) {
2384      isValid = false;
2385      errorMessage = T('form.validEmail', 'Please enter a valid email address');
2386    }
2387  }
2388  
2389  // Pattern validation
2390  if (input.hasAttribute('pattern') && input.value) {
2391    const pattern = new RegExp(input.getAttribute('pattern'));
2392    if (!pattern.test(input.value)) {
2393      isValid = false;
2394      errorMessage = input.getAttribute('data-pattern-error') || T('form.invalidFormat', 'Invalid format');
2395    }
2396  }
2397  
2398  // Update UI
2399  if (isValid) {
2400    input.style.borderColor = '';
2401    input.classList.remove('error');
2402  } else {
2403    input.style.borderColor = 'var(--danger, #ff5555)';
2404    input.classList.add('error');
2405    
2406    // Show error message
2407    const errorDiv = document.createElement('div');
2408    errorDiv.className = 'error-text';
2409    errorDiv.textContent = errorMessage;
2410    input.parentElement.appendChild(errorDiv);
2411  }
2412

2414}

2660// Polling with request deduplication and retry logic
2661const activePolls = new Map();
2662const pollRetries = new Map();
2663
2664async function poll(url, renderFn, containerId, retries = 3) {
2665  // Prevent duplicate requests for the same URL
2666  if (activePolls.has(url)) {
2667    return activePolls.get(url);
2668  }
2669  
2670  const pollPromise = (async () => {
2671    try {
2672      const data = await fetchJson(url);
2673      renderFn(containerId, data);
2674      setUpdated();
2675      pollRetries.delete(url); // Reset retry count on success
2676      activePolls.delete(url);
2677      return data;
2678    } catch (e) {
2679      activePolls.delete(url);
2680      
2681      // Retry logic with exponential backoff
2682      const retryCount = pollRetries.get(url) || 0;
2683      if (retryCount < retries) {
2684        pollRetries.set(url, retryCount + 1);
2685        const delay = Math.min(1000 * Math.pow(2, retryCount), 10000); // Max 10s delay
2686        
2687        await new Promise(resolve => setTimeout(resolve, delay));
2688        return poll(url, renderFn, containerId, retries);
2689      }
2690      
2691      // Max retries reached, show error
2692      pollRetries.delete(url);
2693      const el = document.getElementById(containerId);
2694      if (el) {
2695        const errorMsg = String(e);
2696        const retryLabel = escapeHtml(T('ui.retry', 'Retry'));
2697        const errorPrefix = escapeHtml(T('err.errorPrefix', 'Error: '));
2698        el.innerHTML = '<div style="color:#ff5555; padding:12px; border:1px solid #ff5555; border-radius:8px;">' + errorPrefix + errorMsg + '<br><button onclick="location.reload()" style="margin-top:8px; padding:6px 12px; background:#ff5555; color:white; border:none; border-radius:4px; cursor:pointer;">' + retryLabel + '</button></div>';
2699      }
2700      console.error('Poll error (max retries reached):', e);
2701    }
2702  })();
2703  
2704  activePolls.set(url, pollPromise);
2705  return pollPromise;
2706}

2708// Token validation and refresh
2709async function validateToken() {
2710  const token = getToken();
2711  if (!token) return false;
2712  
2713  try {
2714    // Decode token to check expiration (simple base64 decode for dev tokens)
2715    const parts = token.split('.');
2716    if (parts.length === 3) {
2717      // Real JWT format
2718      const payload = JSON.parse(atob(parts[1]));
2719      const now = Math.floor(Date.now() / 1000);
2720      if (payload.exp && payload.exp < now) {
2721        // Token expired, try to refresh
2722        return await refreshToken();
2723      }
2724      return true;
2725    } else if (token.startsWith('dev_token_')) {
2726      // Dev token format - check expiration from localStorage
2727      const tokenData = localStorage.getItem('poolai_token_exp');
2728      if (tokenData) {
2729        const exp = parseInt(tokenData, 10);
2730        const now = Math.floor(Date.now() / 1000);
2731        if (exp && exp < now) {
2732          return await refreshToken();
2733        }
2734      }
2735      return true;
2736    }
2737    return false;
2738  } catch (e) {
2739    console.error('Token validation error:', e);
2740    return false;
2741  }
2742}
