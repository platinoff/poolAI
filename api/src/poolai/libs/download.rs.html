19/// Download a library from URL
20pub async fn download_library(
21    url: &str,
22    destination: &Path,
23    expected_checksum: Option<&str>,
24) -> Result<PathBuf, AppError> {
25    info!("Downloading library from: {}", url);
26
27    // Create HTTP client
28    let client = reqwest::Client::builder()
29        .timeout(std::time::Duration::from_secs(300)) // 5 minutes timeout
30        .build()
31        .map_err(|e| AppError::ConfigError(format!(
32            "Failed to create HTTP client. Context: Cannot initialize HTTP client for library download. \
33            Suggestion: Check network configuration and ensure reqwest crate is properly configured. \
34            URL: '{}', Error: {}",
35            url, e
36        )))?;
37
38    // Download file
39    let response = client
40        .get(url)
41        .send()
42        .await
43        .map_err(|e| AppError::ConfigError(format!(
44            "Failed to download library. Context: HTTP request failed during library download. \
45            Suggestion: Check network connectivity, verify URL is accessible, and ensure firewall allows outbound connections. \
46            URL: '{}', Error: {}",
47            url, e
48        )))?;
49
50    if !response.status().is_success() {
51        return Err(AppError::ConfigError(format!(
52            "Failed to download library: HTTP {}. Context: Server returned non-success status code. \
53            Suggestion: Verify the download URL is correct and the server is accessible. \
54            URL: '{}', Status: {}",
55            response.status(), url, response.status()
56        )));
57    }
58
59    // Get content length for progress tracking
60    let content_length = response.content_length().unwrap_or(0);
61    info!("Downloading {} bytes", content_length);
62
63    // Create destination file
64    let mut file = tokio::fs::File::create(destination)
65        .await
66        .map_err(|e| AppError::ConfigError(format!(
67            "Failed to create destination file. Context: Cannot create file for downloaded library. \
68            Suggestion: Check filesystem permissions and ensure parent directory exists. \
69            Path: '{}', Error: {}",
70            destination.display(), e
71        )))?;
72
73    // Download with progress tracking
74    let mut stream = response.bytes_stream();
75    let mut hasher = Sha256::new();
76    let mut downloaded = 0u64;
77
78    while let Some(chunk) = stream.next().await {
79        let chunk = chunk.map_err(|e| {
80            AppError::ConfigError(format!(
81                "Download error. Context: Failed to read chunk from HTTP response stream. \
82            Suggestion: Check network stability and retry the download. \
83            URL: '{}', Downloaded: {} bytes, Error: {}",
84                url, downloaded, e
85            ))
86        })?;
87
88        // Write chunk to file
89        file.write_all(&chunk).await.map_err(|e| {
90            AppError::ConfigError(format!(
91                "Failed to write file. Context: Cannot write downloaded chunk to destination file. \
92                Suggestion: Check disk space and filesystem permissions. \
93                Path: '{}', Downloaded: {} bytes, Error: {}",
94                destination.display(), downloaded, e
95            ))
96        })?;
97
98        // Update checksum
99        hasher.update(&chunk);
100
101        // Update progress
102        downloaded += chunk.len() as u64;
103        if content_length > 0 && downloaded.is_multiple_of((content_length / 10).max(1)) {
104            let progress = (downloaded * 100) / content_length;
105            info!("Download progress: {}%", progress);
106        }
107    }
108
109    // Finalize file
110    file.sync_all().await.map_err(|e| {
111        AppError::ConfigError(format!(
112            "Failed to sync file. Context: Cannot flush file data to disk after download. \
113            Suggestion: Check disk space and filesystem integrity. \
114            Path: '{}', Error: {}",
115            destination.display(),
116            e
117        ))
118    })?;
119
120    // Verify checksum if provided
121    let calculated_checksum = format!("{:x}", hasher.finalize());
122    if let Some(expected) = expected_checksum {
123        if calculated_checksum != expected {
124            return Err(AppError::ConfigError(format!(
125                "Checksum mismatch. Context: Downloaded file integrity verification failed. \
126                Suggestion: File may be corrupted or tampered with. Try downloading again or verify the source. \
127                Expected: '{}', Calculated: '{}', Path: '{}'",
128                expected, calculated_checksum, destination.display()
129            )));
130        }
131        info!("Checksum verified successfully");
132    } else {
133        info!("Downloaded checksum: {}", calculated_checksum);
134    }
135
136    info!("Library downloaded successfully to: {:?}", destination);
137    Ok(destination.to_path_buf())
138}

140/// Extract archive (tar.gz, tar, zip)
141pub async fn extract_archive(archive_path: &Path, destination: &Path) -> Result<PathBuf, AppError> {
142    info!(
143        "Extracting archive: {:?} to {:?}",
144        archive_path, destination
145    );
146
147    // Ensure destination directory exists
148    tokio::fs::create_dir_all(destination).await.map_err(|e| {
149        AppError::ConfigError(format!(
150            "Failed to create destination directory. Context: Cannot create directory for archive extraction. \
151            Suggestion: Check filesystem permissions and ensure parent directory exists. \
152            Path: '{}', Error: {}",
153            destination.display(), e
154        ))
155    })?;
156
157    // Determine archive type from extension
158    let extension = archive_path
159        .extension()
160        .and_then(|s| s.to_str())
161        .unwrap_or("");
162
163    match extension {
164        "gz" | "tgz" => {
165            // Handle tar.gz
166            extract_tar_gz(archive_path, destination).await?;
167        }
168        "tar" => {
169            // Handle tar
170            extract_tar(archive_path, destination).await?;
171        }
172        "zip" => {
173            // Handle zip
174            extract_zip(archive_path, destination).await?;
175        }
176        _ => {
177            return Err(AppError::ConfigError(format!(
178                "Unsupported archive format. Context: Archive file extension is not recognized. \
179                Suggestion: Supported formats are: .tar.gz, .tgz, .tar, .zip. \
180                Archive: '{}', Extension: '{}'",
181                archive_path.display(),
182                extension
183            )));
184        }
185    }
186
187    info!("Archive extracted successfully to: {:?}", destination);
188    Ok(destination.to_path_buf())
189}

191/// Extract tar.gz archive
192async fn extract_tar_gz(archive_path: &Path, destination: &Path) -> Result<(), AppError> {
193    use flate2::read::GzDecoder;
194    use std::fs::File;
195    use std::io::BufReader;
196
197    let file = File::open(archive_path).map_err(|e| {
198        AppError::ConfigError(format!(
199            "Failed to open archive. Context: Cannot open tar.gz archive file for extraction. \
200            Suggestion: Check file permissions and verify archive file is not corrupted. \
201            Archive: '{}', Error: {}",
202            archive_path.display(),
203            e
204        ))
205    })?;
206
207    let gz = GzDecoder::new(BufReader::new(file));
208    let mut archive = tar::Archive::new(gz);
209
210    archive.unpack(destination).map_err(|e| {
211        AppError::ConfigError(format!(
212            "Failed to extract tar.gz. Context: Cannot extract contents from tar.gz archive. \
213            Suggestion: Verify archive integrity and ensure destination directory is writable. \
214            Archive: '{}', Destination: '{}', Error: {}",
215            archive_path.display(),
216            destination.display(),
217            e
218        ))
219    })?;
220
221    Ok(())
222}

224/// Extract tar archive
225async fn extract_tar(archive_path: &Path, destination: &Path) -> Result<(), AppError> {
226    use std::fs::File;
227
228    // Run blocking I/O in spawn_blocking
229    let archive_path = archive_path.to_path_buf();
230    let destination = destination.to_path_buf();
231
232    let archive_path_str = archive_path.display().to_string();
233    let destination_str = destination.display().to_string();
234    tokio::task::spawn_blocking(move || {
235        let file = File::open(&archive_path).map_err(|e| {
236            AppError::ConfigError(format!(
237                "Failed to open archive. Context: Cannot open tar archive file for extraction. \
238                Suggestion: Check file permissions and verify archive file is not corrupted. \
239                Archive: '{}', Error: {}",
240                archive_path_str, e
241            ))
242        })?;
243
244        let mut archive = tar::Archive::new(file);
245
246        archive.unpack(&destination).map_err(|e| {
247            AppError::ConfigError(format!(
248                "Failed to extract tar. Context: Cannot extract contents from tar archive. \
249                Suggestion: Verify archive integrity and ensure destination directory is writable. \
250                Archive: '{}', Destination: '{}', Error: {}",
251                archive_path_str, destination_str, e
252            ))
253        })?;
254
255        Ok::<(), AppError>(())
256    })
257    .await
258    .map_err(|e| AppError::ConfigError(format!("Task join error: {}", e)))??;
259
260    Ok(())
261}

263/// Extract zip archive
264async fn extract_zip(archive_path: &Path, destination: &Path) -> Result<(), AppError> {
265    use std::fs::File;
266    use std::io::Read;
267
268    // Run blocking I/O in spawn_blocking
269    let archive_path = archive_path.to_path_buf();
270    let destination = destination.to_path_buf();
271    let archive_path_str = archive_path.display().to_string();
272
273    tokio::task::spawn_blocking(move || {
274        let file = File::open(&archive_path)
275            .map_err(|e| AppError::ConfigError(format!(
276                "Failed to open archive. Context: Cannot open zip archive file for extraction. \
277                Suggestion: Check file permissions and verify archive file is not corrupted. \
278                Archive: '{}', Error: {}",
279                archive_path_str, e
280            )))?;
281
282        let mut archive = zip::ZipArchive::new(file)
283            .map_err(|e| AppError::ConfigError(format!(
284                "Failed to read zip archive. Context: Cannot parse zip archive structure. \
285                Suggestion: Verify archive integrity and ensure file is a valid ZIP archive. \
286                Archive: '{}', Error: {}",
287                archive_path_str, e
288            )))?;
289
290        for i in 0..archive.len() {
291            let mut file = archive
292                .by_index(i)
293                .map_err(|e| AppError::ConfigError(format!(
294                    "Failed to read zip entry. Context: Cannot read entry from zip archive. \
295                    Suggestion: Archive may be corrupted or incomplete. Try re-downloading. \
296                    Archive: '{}', Entry index: {}, Error: {}",
297                    archive_path_str, i, e
298                )))?;
299
300            let outpath = destination.join(file.name());
301
302            if file.name().ends_with('/') {
303                // Directory
304                std::fs::create_dir_all(&outpath).map_err(|e| {
305                    AppError::ConfigError(format!(
306                        "Failed to create directory. Context: Cannot create directory from zip archive. \
307                        Suggestion: Check filesystem permissions and ensure parent directory exists. \
308                        Path: '{}', Error: {}",
309                        outpath.display(), e
310                    ))
311                })?;
312            } else {
313                // File
314                if let Some(parent) = outpath.parent() {
315                    std::fs::create_dir_all(parent).map_err(|e| {
316                        AppError::ConfigError(format!(
317                            "Failed to create parent directory. Context: Cannot create parent directory for extracted file. \
318                            Suggestion: Check filesystem permissions. \
319                            Path: '{}', Error: {}",
320                            parent.display(), e
321                        ))
322                    })?;
323                }
324
325                let mut outfile = std::fs::File::create(&outpath)
326                    .map_err(|e| AppError::ConfigError(format!(
327                        "Failed to create file. Context: Cannot create extracted file from zip archive. \
328                        Suggestion: Check filesystem permissions and disk space. \
329                        Path: '{}', Error: {}",
330                        outpath.display(), e
331                    )))?;
332
333                let mut buffer = Vec::new();
334                file.read_to_end(&mut buffer).map_err(|e| {
335                    AppError::ConfigError(format!(
336                        "Failed to read zip entry. Context: Cannot read file contents from zip archive. \
337                        Suggestion: Archive may be corrupted. Try re-downloading. \
338                        Entry: '{}', Error: {}",
339                        file.name(), e
340                    ))
341                })?;
342
343                std::io::Write::write_all(&mut outfile, &buffer)
344                    .map_err(|e| AppError::ConfigError(format!(
345                        "Failed to write file. Context: Cannot write extracted file contents to disk. \
346                        Suggestion: Check disk space and filesystem permissions. \
347                        Path: '{}', Error: {}",
348                        outpath.display(), e
349                    )))?;
350            }
351        }
352
353        Ok::<(), AppError>(())
354    })
355    .await
356    .map_err(|e| AppError::ConfigError(format!("Task join error: {}", e)))??;
357
358    Ok(())
359}

361/// Calculate file checksum
362pub async fn calculate_checksum(file_path: &Path) -> Result<String, AppError> {
363    let mut file = tokio::fs::File::open(file_path).await.map_err(|e| {
364        AppError::ConfigError(format!(
365            "Failed to open file. Context: Cannot open file for checksum calculation. \
366            Suggestion: Check file permissions and ensure file exists. \
367            Path: '{}', Error: {}",
368            file_path.display(),
369            e
370        ))
371    })?;
372
373    let mut hasher = Sha256::new();
374    let mut buffer = vec![0u8; 8192];
375
376    loop {
377        let n = file
378            .read(&mut buffer)
379            .await
380            .map_err(|e| AppError::ConfigError(format!(
381                "Failed to read file. Context: Cannot read file data for checksum calculation. \
382                Suggestion: Check file permissions and verify file is not locked by another process. \
383                Path: '{}', Error: {}",
384                file_path.display(), e
385            )))?;
386
387        if n == 0 {
388            break;
389        }
390
391        hasher.update(&buffer[..n]);
392    }
393
394    Ok(format!("{:x}", hasher.finalize()))
395}

397/// Create a tar.gz archive from a directory
398pub async fn create_artifact_archive(
399    source_dir: &Path,
400    archive_path: &Path,
401) -> Result<PathBuf, AppError> {
402    info!(
403        "Creating artifact archive from {:?} to {:?}",
404        source_dir, archive_path
405    );
406
407    // Ensure parent directory exists
408    if let Some(parent) = archive_path.parent() {
409        tokio::fs::create_dir_all(parent).await.map_err(|e| {
410            AppError::ConfigError(format!(
411                "Failed to create archive parent directory. Context: Cannot create parent directory for artifact archive. \
412                Suggestion: Check filesystem permissions and ensure parent directory path is valid. \
413                Path: '{}', Error: {}",
414                parent.display(), e
415            ))
416        })?;
417    }
418
419    let source_dir = source_dir.to_path_buf();
420    let archive_path = archive_path.to_path_buf();
421
422    let archive_path_clone = archive_path.clone();
423    let archive_path_str = archive_path_clone.display().to_string();
424    let source_dir_str = source_dir.display().to_string();
425
426    // Run blocking I/O in spawn_blocking
427    tokio::task::spawn_blocking(move || {
428        use flate2::write::GzEncoder;
429        use flate2::Compression;
430        use std::fs::File;
431
432        let file = File::create(&archive_path)
433            .map_err(|e| AppError::ConfigError(format!(
434                "Failed to create archive file. Context: Cannot create tar.gz archive file. \
435                Suggestion: Check filesystem permissions and disk space. \
436                Path: '{}', Error: {}",
437                archive_path_str, e
438            )))?;
439
440        let gz = GzEncoder::new(file, Compression::default());
441        let mut tar = tar::Builder::new(gz);
442
443        tar.append_dir_all(".", &source_dir).map_err(|e| {
444            AppError::ConfigError(format!(
445                "Failed to add directory to archive. Context: Cannot add source directory contents to tar archive. \
446                Suggestion: Check source directory permissions and verify directory is accessible. \
447                Source: '{}', Archive: '{}', Error: {}",
448                source_dir_str, archive_path_str, e
449            ))
450        })?;
451
452        tar.finish()
453            .map_err(|e| AppError::ConfigError(format!(
454                "Failed to finish archive. Context: Cannot finalize tar.gz archive creation. \
455                Suggestion: Check disk space and filesystem integrity. \
456                Archive: '{}', Error: {}",
457                archive_path_str, e
458            )))?;
459
460        Ok::<PathBuf, AppError>(archive_path)
461    })
462    .await
463    .map_err(|e| AppError::ConfigError(format!("Task join error: {}", e)))??;
464
465    info!(
466        "Artifact archive created successfully: {:?}",
467        archive_path_clone
468    );
469    Ok(archive_path_clone)
