1//! Network topology management for load balancing
2//!
3//! This module provides:
4//! - Latency matrix between nodes
5//! - Network bandwidth measurement
6//! - Topology-aware placement strategies
7//!
8//! Inspired by exo's topology-aware load balancing feature.

21/// Network topology information
22#[derive(Debug, Clone, Serialize, Deserialize)]
23pub struct Topology {
24    /// Latency matrix between nodes (in milliseconds)
25    /// Key format: "node_id1:node_id2"
26    pub latency_matrix: HashMap<String, f64>,
27    /// Bandwidth matrix between nodes (in Mbps)
28    /// Key format: "node_id1:node_id2"
29    pub bandwidth_matrix: HashMap<String, f64>,
30    /// Resource information per node
31    pub node_resources: HashMap<String, NodeResources>,
32    /// Last topology update timestamp
33    pub last_updated: DateTime<Utc>,
34}
35
36/// Resource information for a node
37#[derive(Debug, Clone, Serialize, Deserialize)]
38pub struct NodeResources {
39    /// Node ID
40    pub node_id: String,
41    /// Available GPU memory (MB)
42    pub available_gpu_memory_mb: u64,
43    /// Total GPU memory (MB)
44    pub total_gpu_memory_mb: u64,
45    /// Available CPU cores
46    pub available_cpu_cores: usize,
47    /// Total CPU cores
48    pub total_cpu_cores: usize,
49    /// Available system memory (MB)
50    pub available_memory_mb: u64,
51    /// Total system memory (MB)
52    pub total_memory_mb: u64,
53    /// Current load (0.0-1.0)
54    pub current_load: f32,
55}

57/// Topology manager
58pub struct TopologyManager {
59    topology: Arc<RwLock<Topology>>,
60    /// Latency measurement timeout (seconds)
61    latency_timeout_secs: u64,
62    // Note: update_interval_secs is configured in main.rs topology update task
63    // Stored here for future use in self-managed update loops
64    #[allow(dead_code)]
65    update_interval_secs: u64,
66}

68impl TopologyManager {
69    /// Create a new topology manager
70    pub fn new() -> Self {
71        Self {
72            topology: Arc::new(RwLock::new(Topology {
73                latency_matrix: HashMap::new(),
74                bandwidth_matrix: HashMap::new(),
75                node_resources: HashMap::new(),
76                last_updated: Utc::now(),
77            })),
78            latency_timeout_secs: 5,
79            update_interval_secs: 30,
80        }
81    }

83    /// Get latency between two nodes (in milliseconds)
84    pub async fn get_latency(&self, node_id1: &str, node_id2: &str) -> Option<f64> {
85        let topology = self.topology.read().await;
86
87        // Try direct path
88        let key = format!("{}:{}", node_id1, node_id2);
89        if let Some(&latency) = topology.latency_matrix.get(&key) {
90            return Some(latency);
91        }
92
93        // Try reverse path (latency is symmetric)
94        let reverse_key = format!("{}:{}", node_id2, node_id1);
95        if let Some(&latency) = topology.latency_matrix.get(&reverse_key) {
96            return Some(latency);
97        }
98
99        None
100    }
101
102    /// Get bandwidth between two nodes (in Mbps)
103    pub async fn get_bandwidth(&self, node_id1: &str, node_id2: &str) -> Option<f64> {
104        let topology = self.topology.read().await;
105
106        let key = format!("{}:{}", node_id1, node_id2);
107        topology.bandwidth_matrix.get(&key).copied().or_else(|| {
108            // Try reverse path
109            let reverse_key = format!("{}:{}", node_id2, node_id1);
110            topology.bandwidth_matrix.get(&reverse_key).copied()
111        })
112    }
113
114    /// Get resource information for a node
115    pub async fn get_node_resources(&self, node_id: &str) -> Option<NodeResources> {
116        let topology = self.topology.read().await;
117        topology.node_resources.get(node_id).cloned()
118    }

120    /// Measure latency to a peer (simplified ping-based)
121    async fn measure_latency(&self, peer: &PeerInfo) -> Option<f64> {
122        // Simplified latency measurement using TCP connect time
123        // In real implementation, this would use ICMP ping or HTTP ping
124        let addr = format!("{}:{}", peer.address, peer.port);
125
126        // Try to establish a TCP connection and measure time
127        let start = std::time::Instant::now();
128        let result = timeout(
129            Duration::from_secs(self.latency_timeout_secs),
130            tokio::net::TcpStream::connect(&addr),
131        )
132        .await;
133
134        match result {
135            Ok(Ok(_stream)) => {
136                let elapsed = start.elapsed();
137                Some(elapsed.as_secs_f64() * 1000.0) // Convert to milliseconds
138            }
139            Ok(Err(_)) => {
140                warn!("Failed to connect to {} for latency measurement", addr);
141                None
142            }
143            Err(_) => {
144                warn!("Latency measurement timeout for {}", addr);
145                None
146            }
147        }
148    }

150    /// Update topology with discovered peers
151    pub async fn update_topology(&self) -> Result<(), AppError> {
152        use crate::runtime::instance::get_global_instance_manager;
153
154        info!("Updating network topology");
155
156        // Get discovered peers from discovery service
157        let peers = if let Some(discovery) = get_global_discovery_service() {
158            discovery.get_peers().await
159        } else {
160            Vec::new()
161        };
162
163        let mut topology = self.topology.write().await;
164
165        // Calculate load per node from active instances
166        let mut node_loads: HashMap<String, (usize, usize)> = HashMap::new(); // (active_requests, capacity)
167
168        if let Some(instance_manager) = get_global_instance_manager() {
169            let manager = instance_manager.read().await;
170            let instances = manager.list_instances().await;
171
172            for instance in instances {
173                // Count active instances per node (active = Ready or Active status)
174                let status = instance.status.read().await;
175                let is_active = matches!(
176                    *status,
177                    crate::runtime::instance::InstanceStatus::Ready
178                        | crate::runtime::instance::InstanceStatus::Active
179                );
180                drop(status);
181
182                if is_active {
183                    for node_id in &instance.placement.node_ids {
184                        let entry = node_loads.entry(node_id.clone()).or_insert((0, 0));
185                        entry.0 += 1; // active_requests
186                        entry.1 += 10; // capacity (estimate: 10 instances per node)
187                    }
188                }
189            }
190        }
191
192        // Update node resources from peers with load
193        for peer in &peers {
194            // Get load for this node
195            let (active_requests, capacity) =
196                node_loads.get(&peer.peer_id).copied().unwrap_or((0, 10));
197            let calculated_load = if capacity > 0 {
198                (active_requests as f32 / capacity as f32).min(1.0)
199            } else {
200                0.0
201            };
202
203            // Use peer capabilities load if available (from heartbeat), otherwise calculate from instances
204            let current_load = if peer.capabilities.capacity > 0 {
205                peer.capabilities.current_load
206            } else {
207                calculated_load
208            };
209
210            let resources = NodeResources {
211                node_id: peer.peer_id.clone(),
212                available_gpu_memory_mb: peer.capabilities.memory_mb as u64, // Simplified
213                total_gpu_memory_mb: peer.capabilities.memory_mb as u64,
214                available_cpu_cores: peer.capabilities.cpu_cores,
215                total_cpu_cores: peer.capabilities.cpu_cores,
216                available_memory_mb: peer.capabilities.memory_mb as u64,
217                total_memory_mb: peer.capabilities.memory_mb as u64,
218                current_load,
219            };
220
221            topology
222                .node_resources
223                .insert(peer.peer_id.clone(), resources);
224        }
225
226        // Measure latency between nodes
227        let local_peer_id = get_global_discovery_service()
228            .map(|d| d.local_peer_id().to_string())
229            .unwrap_or_else(|| "local".to_string());
230
231        for peer in &peers {
232            if peer.peer_id == local_peer_id {
233                continue;
234            }
235
236            let key = format!("{}:{}", local_peer_id, peer.peer_id);
237
238            // Measure latency (non-blocking, don't fail if measurement fails)
239            if let Some(latency) = self.measure_latency(peer).await {
240                topology.latency_matrix.insert(key.clone(), latency);
241                debug!("Latency to {}: {:.2}ms", peer.peer_id, latency);
242            }
243
244            // Estimate bandwidth (simplified - in real implementation would measure)
245            // For now, use default values based on network type
246            let bandwidth = 1000.0; // Default 1 Gbps for local network
247            topology.bandwidth_matrix.insert(key, bandwidth);
248        }
249
250        topology.last_updated = Utc::now();
251
252        info!(
253            "Topology updated: {} nodes, {} latency measurements, load tracking enabled",
254            topology.node_resources.len(),
255            topology.latency_matrix.len()
256        );
257
258        Ok(())
259    }

261    /// Find best nodes for placement based on topology
262    pub async fn find_best_nodes(
263        &self,
264        required_memory_mb: u64,
265        required_cpu_cores: usize,
266        node_count: usize,
267    ) -> Vec<String> {
268        let topology = self.topology.read().await;
269
270        // Filter nodes by resource availability
271        let mut candidates: Vec<_> = topology
272            .node_resources
273            .iter()
274            .filter(|(_, resources)| {
275                resources.available_gpu_memory_mb >= required_memory_mb
276                    && resources.available_cpu_cores >= required_cpu_cores
277                    && resources.current_load < 0.9 // Not overloaded
278            })
279            .collect();
280
281        // Sort by current load (prefer less loaded nodes)
282        candidates.sort_by(|a, b| {
283            a.1.current_load
284                .partial_cmp(&b.1.current_load)
285                .unwrap_or(std::cmp::Ordering::Equal)
286        });
287
288        // Take top N nodes
289        candidates
290            .into_iter()
291            .take(node_count)
292            .map(|(node_id, _)| node_id.clone())
293            .collect()
294    }

296    /// Get average latency between a set of nodes
297    pub async fn get_average_latency(&self, node_ids: &[String]) -> Option<f64> {
298        if node_ids.len() < 2 {
299            return None;
300        }
301
302        let topology = self.topology.read().await;
303        let mut total_latency = 0.0;
304        let mut count = 0;
305
306        for i in 0..node_ids.len() {
307            for j in (i + 1)..node_ids.len() {
308                let key = format!("{}:{}", node_ids[i], node_ids[j]);
309                if let Some(&latency) = topology.latency_matrix.get(&key) {
310                    total_latency += latency;
311                    count += 1;
312                }
313            }
314        }
315
316        if count > 0 {
317            Some(total_latency / count as f64)
318        } else {
319            None
320        }
321    }

336// Global topology manager instance
337static GLOBAL_TOPOLOGY_MANAGER: OnceLock<Arc<RwLock<TopologyManager>>> = OnceLock::new();
338
339/// Initialize global topology manager
340pub fn initialize_global_topology_manager() -> Result<(), AppError> {
341    let manager = TopologyManager::new();
342    GLOBAL_TOPOLOGY_MANAGER
343        .set(Arc::new(RwLock::new(manager)))
344        .map_err(|_| AppError::ConfigError("Topology manager already initialized".to_string()))?;
345    Ok(())
346}

348/// Get global topology manager
349pub fn get_global_topology_manager() -> Option<&'static Arc<RwLock<TopologyManager>>> {
350    GLOBAL_TOPOLOGY_MANAGER.get()
