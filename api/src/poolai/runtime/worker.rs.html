16/// Worker status enumeration
17///
18/// Represents the current state of a worker instance in the runtime system.
19///
20/// # Example
21///
22/// ```rust
23/// use poolai::runtime::worker::WorkerStatus;
24///
25/// let status = WorkerStatus::Ready;
26/// match status {
27///     WorkerStatus::Initializing => println!("Worker is initializing"),
28///     WorkerStatus::Ready => println!("Worker is ready"),
29///     WorkerStatus::Busy => println!("Worker is busy"),
30///     WorkerStatus::Idle => println!("Worker is idle"),
31///     WorkerStatus::Error => println!("Worker has error"),
32///     WorkerStatus::Shutdown => println!("Worker is shutdown"),
33/// }
34/// ```
35#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
36pub enum WorkerStatus {
37    Initializing,
38    Ready,
39    Busy,
40    Idle,
41    Error,
42    Shutdown,
43}

45/// Worker configuration

70    /// CPU priority (1-10)
71    pub cpu_priority: u8,
72    /// GPU device ID (if applicable)
73    pub gpu_device: Option<usize>,
74    /// Auto-restart on failure
75    pub auto_restart: bool,

118/// Worker instance
119///
120/// Manages the lifecycle of an AI mining worker process including spawning,
121/// monitoring, health checks, and graceful shutdown.
122///
123/// # Example
124///
125/// ```no_run
126/// use poolai::runtime::worker::Worker;
127///
128/// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
129/// let mut worker = Worker::new(1);
130/// worker.initialize().await?;
131/// worker.start().await?;
132///
133/// // Get worker status
134/// let status = worker.get_status().await;
135/// println!("Worker status: {:?}", status);
136///
137/// // Get worker metrics
138/// let metrics = worker.get_metrics().await;
139/// println!("CPU usage: {:.1}%", metrics.cpu_usage);
140///
141/// worker.shutdown().await?;
142/// # Ok(())
143/// # }
144/// ```
145pub struct Worker {
146    config: WorkerConfig,
147    status: Arc<RwLock<WorkerStatus>>,
148    metrics: Arc<RwLock<WorkerMetrics>>,
149    process: Option<Child>,
150    #[allow(dead_code)] // Will be used for task distribution in future
151    task_channel: mpsc::Sender<WorkerTask>,
152    health_monitor: tokio::task::JoinHandle<()>,
153}

205    /// Initialize worker
206    ///
207    /// Prepares the worker for operation by setting up health monitoring
208    /// and transitioning to Ready status.
209    ///
210    /// # Returns
211    ///
212    /// Returns `Ok(())` if initialization succeeds.
213    ///
214    /// # Example
215    ///
216    /// ```no_run
217    /// use poolai::runtime::worker::Worker;
218    ///
219    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
220    /// let mut worker = Worker::new(1);
221    /// worker.initialize().await?;
222    /// # Ok(())
223    /// # }
224    /// ```
225    pub async fn initialize(&mut self) -> Result<(), Box<dyn std::error::Error>> {
226        info!("Initializing worker {}", self.config.id);
227
228        // Set status to ready
229        {
230            let mut status = self.status.write().await;
231            *status = WorkerStatus::Ready;
232        }
233
234        // Start health monitoring
235        self.start_health_monitoring().await?;
236
237        info!("Worker {} initialized successfully", self.config.id);
238        Ok(())
239    }

243    /// Spawns the worker process and transitions the worker to Ready status.
244    /// The worker process will run as a separate child process.

246    /// # Returns

248    /// Returns `Ok(())` if the worker process is spawned successfully.

262    pub async fn start(&mut self) -> Result<(), Box<dyn std::error::Error>> {
263        info!("Starting worker {}", self.config.id);
264
265        // Spawn worker process
266        self.spawn_process().await?;
267
268        // Update status
269        {

272        }
273
274        info!("Worker {} started successfully", self.config.id);

278    /// Shutdown worker
279    ///
280    /// Gracefully shuts down the worker by terminating the process and
281    /// stopping health monitoring.
282    ///
283    /// # Returns
284    ///
285    /// Returns `Ok(())` if shutdown completes successfully.
286    ///
287    /// # Example
288    ///
289    /// ```no_run
290    /// use poolai::runtime::worker::Worker;
291    ///
292    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
293    /// let mut worker = Worker::new(1);
294    /// worker.initialize().await?;
295    /// worker.start().await?;
296    /// worker.shutdown().await?;
297    /// # Ok(())
298    /// # }
299    /// ```
300    pub async fn shutdown(&mut self) -> Result<(), Box<dyn std::error::Error>> {
301        info!("Shutting down worker {}", self.config.id);
302
303        // Update status
304        {
305            let mut status = self.status.write().await;
306            *status = WorkerStatus::Shutdown;
307        }
308
309        // Terminate process if running
310        if let Some(mut process) = self.process.take() {
311            if let Err(e) = process.kill().await {
312                warn!("Failed to kill worker process: {}", e);
313            }
314        }
315
316        // Cancel health monitoring
317        self.health_monitor.abort();
318
319        info!("Worker {} shutdown completed", self.config.id);
320        Ok(())
321    }

333    /// Spawn worker process
334    async fn spawn_process(&mut self) -> Result<(), Box<dyn std::error::Error>> {
335        // Prefer spawning the sibling worker binary located next to the current executable.
336        // This avoids relying on PATH (common source of `ErrorKind::NotFound` on Windows/MSYS).
337        let worker_exe = match std::env::current_exe() {
338            Ok(exe) => {
339                #[cfg(target_os = "windows")]
340                let name = "poolai-worker.exe";
341                #[cfg(not(target_os = "windows"))]
342                let name = "poolai-worker";
343                exe.with_file_name(name)
344            }
345            Err(_) => {
346                #[cfg(target_os = "windows")]
347                let name = "poolai-worker.exe";
348                #[cfg(not(target_os = "windows"))]
349                let name = "poolai-worker";
350                std::path::PathBuf::from(name)
351            }
352        };
353
354        let mut command = if worker_exe.exists() {
355            Command::new(worker_exe)
356        } else {
357            // Fallback to PATH lookup for custom deployments.
358            #[cfg(target_os = "windows")]
359            let name = "poolai-worker.exe";
360            #[cfg(not(target_os = "windows"))]
361            let name = "poolai-worker";
362            Command::new(name)
363        };
364
365        // Ensure the child process is terminated when the parent drops the handle (graceful shutdown path).
366        // This prevents orphaned `poolai-worker` processes during development and normal Ctrl+C shutdown.
367        command.kill_on_drop(true);
368
369        // Set process priority (Windows-specific code commented out for now)
370        #[cfg(target_os = "windows")]
371        {
372            // Future improvement: Implement Windows process priority setting
373            // 1. Open process handle using OpenProcess() Windows API
374            //    - Use PROCESS_SET_INFORMATION access right
375            //    - Handle must be closed with CloseHandle() when done
376            // 2. Set priority class using SetPriorityClass() Windows API
377            //    - Use HIGH_PRIORITY_CLASS, NORMAL_PRIORITY_CLASS, or BELOW_NORMAL_PRIORITY_CLASS
378            //    - Example: SetPriorityClass(process_handle, HIGH_PRIORITY_CLASS)
379            // 3. Optionally set thread priority using SetThreadPriority() Windows API
380            //    - Use THREAD_PRIORITY_* constants for fine-grained control
381            //    - Requires thread handle from process
382            //
383            // This requires:
384            // - Windows API bindings (windows-sys crate or winapi crate)
385            // - Process handle management (proper cleanup)
386            // - Understanding of Windows priority classes
387        }
388
389        // Spawn process
390        let child = match command
391            .arg("--worker-id")
392            .arg(&self.config.id)
393            .arg("--max-memory")
394            .arg(self.config.max_memory_mb.to_string())
395            .spawn()
396        {
397            Ok(child) => child,
398            Err(e) if e.kind() == ErrorKind::NotFound => {
399                warn!(
400                    "Worker binary not found; continuing without external worker process (id={}). \
401                     Build/run `poolai-worker` or ensure it is next to the main executable.",
402                    self.config.id
403                );
404                return Ok(());
405            }
406            Err(e) => return Err(e.into()),
407        };
408
409        self.process = Some(child);
410        Ok(())
411    }

413    /// Start health monitoring
414    async fn start_health_monitoring(&mut self) -> Result<(), Box<dyn std::error::Error>> {
415        let status = Arc::clone(&self.status);
416        let metrics = Arc::clone(&self.metrics);
417        let config = self.config.clone();
418
419        self.health_monitor = tokio::task::spawn(async move {
420            let mut interval = tokio::time::interval(tokio::time::Duration::from_secs(
421                config.health_check_interval,
422            ));
423
424            loop {
425                interval.tick().await;
426
427                // Check worker health
428                if let Err(e) = Self::check_worker_health(&status, &metrics).await {
429                    warn!("Health check failed for worker {}: {}", config.id, e);
430
431                    // Auto-restart if enabled
432                    if config.auto_restart {
433                        warn!("Auto-restarting worker {}", config.id);
434                        // Future improvement: Implement restart logic
435                        // 1. Wait for process to fully terminate using wait() or kill()
436                        //    - Ensure process is completely stopped before restart
437                        //    - Use timeout to avoid hanging on unresponsive processes
438                        // 2. Check restart attempts counter against max_restart_attempts
439                        //    - Increment restart attempts counter
440                        //    - If max attempts reached, mark worker as failed
441                        // 3. Calculate restart delay using exponential backoff (if enabled)
442                        //    - Use exponential backoff: delay = initial * 2^attempts
443                        //    - Cap delay at max_restart_delay_secs
444                        //    - Sleep for calculated delay using tokio::time::sleep()
445                        // 4. Re-spawn worker process using the same spawn logic
446                        //    - Use command.spawn() to create new process
447                        //    - Update process_id in worker status
448                        //    - Reset restart attempts counter on successful restart
449                        // 5. Re-attach monitoring and health checks for new process
450                        //    - Re-register health checks
451                        //    - Re-attach process monitoring
452                        //
453                        // This requires:
454                        // - Process state tracking (restart attempts, last restart time)
455                        // - Delay calculation logic (exponential backoff)
456                        // - Process spawning (reuse existing spawn logic)
457                        // - Health check re-registration
458                    }
459                }
460            }
461        });
462
463        Ok(())
464    }

466    /// Check worker health
467    async fn check_worker_health(
468        status: &Arc<RwLock<WorkerStatus>>,
469        metrics: &Arc<RwLock<WorkerMetrics>>,
470    ) -> Result<(), Box<dyn std::error::Error>> {
471        let current_status = status.read().await;
472        let current_metrics = metrics.read().await;
473
474        // Check if worker is responsive
475        if *current_status == WorkerStatus::Error {
476            return Err("Worker is in error state".into());
477        }
478
479        // Check memory usage
480        if current_metrics.memory_usage_mb > 2048.0 {
481            warn!(
482                "Worker memory usage high: {} MB",
483                current_metrics.memory_usage_mb
484            );
485        }
486
487        // Check CPU usage
488        if current_metrics.cpu_usage > 90.0 {
489            warn!("Worker CPU usage high: {}%", current_metrics.cpu_usage);
490        }
491
492        Ok(())
493    }
