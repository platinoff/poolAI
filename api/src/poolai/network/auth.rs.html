1//! Authentication and authorization module
2//!
3//! Provides JWT-based authentication, role-based access control (RBAC),
4//! and middleware for protecting API endpoints.
5//!
6//! # Features
7//!
8//! - **JWT Token Generation**: Create and validate JWT tokens with claims
9//! - **Role-Based Access Control**: Admin, Operator, and Viewer roles with permissions
10//! - **Authentication Middleware**: Protect routes with token validation
11//! - **Permission Middleware**: Check specific permissions for route access
12//! - **User Authentication**: Authenticate users and generate tokens
13//!
14//! # Example
15//!
16//! ```no_run
17//! use poolai::network::auth::{authenticate_user, AuthRequest, UserRole};
18//!
19//! # async fn example() -> Result<(), Box<dyn std::error::Error>> {

27//!     format!("Authentication failed: {} - {:?}", code, json)

53/// JWT Claims structure
54///
55/// Contains user information embedded in JWT tokens including user ID,
56/// expiration time, role, and permissions.
57///
58/// # Example
59///
60/// ```rust
61/// use poolai::network::auth::{Claims, UserRole};
62/// use std::time::{SystemTime, UNIX_EPOCH};
63///
64/// let now = SystemTime::now()
65///     .duration_since(UNIX_EPOCH)
66///     .unwrap()
67///     .as_secs() as usize;
68///
69/// let claims = Claims {
70///     sub: "user123".to_string(),
71///     exp: now + 3600,
72///     iat: now,
73///     role: UserRole::Admin,
74///     permissions: vec!["read:all".to_string(), "write:all".to_string()],
75/// };
76/// ```
77#[derive(Debug, Serialize, Deserialize, Clone)]
78pub struct Claims {
79    pub sub: String,              // User ID
80    pub exp: usize,               // Expiration time
81    pub iat: usize,               // Issued at
82    pub role: UserRole,           // User role
83    pub permissions: Vec<String>, // User permissions
84}

86/// User roles for role-based access control
87///
88/// Defines three roles with different permission levels:
89/// - **Admin**: Full access to all resources
90/// - **Operator**: Read access and write access to workers/models
91/// - **Viewer**: Read-only access to status, metrics, and models
92///
93/// # Example
94///
95/// ```rust
96/// use poolai::network::auth::UserRole;
97///
98/// let role = UserRole::Admin;
99/// let permissions = role.get_permissions();
100/// println!("Admin permissions: {:?}", permissions);
101/// ```
102#[derive(Debug, Serialize, Deserialize, Clone, PartialEq)]
103pub enum UserRole {
104    Admin,
105    Operator,
106    Viewer,
107}
108
109impl UserRole {
110    pub fn get_permissions(&self) -> Vec<String> {
111        match self {
112            UserRole::Admin => vec![
113                "read:all".to_string(),
114                "write:all".to_string(),
115                "delete:all".to_string(),
116                "admin:all".to_string(),
117            ],
118            UserRole::Operator => vec![
119                "read:all".to_string(),
120                "write:workers".to_string(),
121                "write:models".to_string(),
122                "read:metrics".to_string(),
123            ],
124            UserRole::Viewer => vec![
125                "read:status".to_string(),
126                "read:metrics".to_string(),
127                "read:models".to_string(),
128            ],
129        }
130    }

263/// Validate JWT token
264///
265/// Validates a JWT token and extracts claims. Checks token format, expiration,
266/// and signature (when JWT feature is enabled).
267///
268/// # Arguments
269///
270/// * `token` - JWT token string to validate
271///
272/// # Returns
273///
274/// Returns `Claims` if token is valid, or an error if validation fails.
275///
276/// # Example
277///
278/// ```rust
279/// use poolai::network::auth::validate_token;
280///
281/// let token = "dev_token_...";
282/// match validate_token(token) {
283///     Ok(claims) => println!("User: {}, Role: {:?}", claims.sub, claims.role),
284///     Err(e) => println!("Token validation failed: {}", e),
285/// }
286/// ```
287pub fn validate_token(token: &str) -> Result<Claims, String> {
288    #[cfg(feature = "jwt")]
289    {
290        // Real JWT token validation
291        let config = JwtConfig::default();
292        use jsonwebtoken::{decode, DecodingKey, Validation};
293        let key = DecodingKey::from_secret(config.secret.as_ref());
294        let validation = Validation::default();
295
296        decode::<Claims>(token, &key, &validation)
297            .map(|data| data.claims)
298            .map_err(|e| format!("Token validation failed: {}", e))
299    }
300
301    #[cfg(not(feature = "jwt"))]
302    {
303        // Fallback: Simple validation (NOT SECURE - for development only)
304        if !token.starts_with("dev_token_") {
305            return Err("Invalid token format".to_string());
306        }
307
308        let token_data = &token[10..]; // Skip "dev_token_"
309        let decoded = base64::engine::general_purpose::STANDARD
310            .decode(token_data)
311            .map_err(|e| format!("Decode error: {}", e))?;
312        let claims: Claims =
313            serde_json::from_slice(&decoded).map_err(|e| format!("Parse error: {}", e))?;
314
315        // Check expiration
316        let now = SystemTime::now()
317            .duration_since(UNIX_EPOCH)
318            .unwrap()
319            .as_secs() as usize;
320
321        if claims.exp < now {
322            return Err("Token expired".to_string());
323        }
324
325        Ok(claims)
326    }
327}

329/// Authentication middleware
330///
331/// Middleware that validates JWT tokens from the `Authorization` header
332/// and adds claims to request extensions for use in route handlers.
333///
334/// # Returns
335///
336/// Returns the response from the next middleware/handler, or an error
337/// if authentication fails.
338///
339/// # Example
340///
341/// ```no_run
342/// use poolai::network::auth::auth_middleware;
343/// use axum::{middleware, Router, routing::post};
344///
345/// // Protect a route with authentication
346/// async fn handler() -> &'static str { "ok" }
347/// let app: Router<()> = Router::new()
348///     .route("/api/workers", post(handler))
349///     .layer(middleware::from_fn(auth_middleware));
350/// ```
351pub async fn auth_middleware(
352    mut req: Request,
353    next: Next,
354) -> Result<Response, (StatusCode, Json<serde_json::Value>)> {
355    // Отримуємо токен з заголовка Authorization
356    let auth_header = req
357        .headers()
358        .get(AUTHORIZATION)
359        .and_then(|auth| auth.to_str().ok())
360        .and_then(|auth_str| {
361            if auth_str.starts_with("Bearer ") {
362                Some(auth_str[7..].to_string())
363            } else {
364                None
365            }
366        });
367
368    let token = auth_header.ok_or_else(|| {
369        (
370            StatusCode::UNAUTHORIZED,
371            Json(serde_json::json!({
372                "error": "Missing or invalid authorization header"
373            })),
374        )
375    })?;
376
377    // Валідуємо токен
378    let claims = validate_token(&token).map_err(|_| {
379        (
380            StatusCode::UNAUTHORIZED,
381            Json(serde_json::json!({
382                "error": "Invalid or expired token"
383            })),
384        )
385    })?;
386
387    // Додаємо claims до request extensions
388    req.extensions_mut().insert(claims);
389
390    Ok(next.run(req).await)
391}

393// Middleware для перевірки прав доступу
394pub async fn permission_middleware(
395    req: Request,
396    next: Next,
397    required_permission: &str,
398) -> Result<Response, (StatusCode, Json<serde_json::Value>)> {
399    let claims = req.extensions().get::<Claims>().ok_or_else(|| {
400        (
401            StatusCode::UNAUTHORIZED,
402            Json(serde_json::json!({
403                "error": "User not authenticated"
404            })),
405        )
406    })?;
407
408    // Перевіряємо права доступу
409    if !claims
410        .permissions
411        .contains(&required_permission.to_string())
412    {
413        return Err((
414            StatusCode::FORBIDDEN,
415            Json(serde_json::json!({
416                "error": "Insufficient permissions",
417                "required": required_permission,
418                "user_permissions": claims.permissions
419            })),
420        ));
421    }
422
423    Ok(next.run(req).await)
424}

426/// Authenticate user and generate token
427///
428/// Validates user credentials and generates a JWT token with appropriate
429/// role and permissions. Currently uses hardcoded credentials for development.
430///
431/// # Arguments

444///

455/// # Ok(())
456/// # }
457/// ```
458pub async fn authenticate_user(
459    auth_req: AuthRequest,
460) -> Result<AuthResponse, (StatusCode, Json<serde_json::Value>)> {
461    // Future improvement: Реальна перевірка користувача з бази даних
462    // 1. Підключення до бази даних (SQLite, PostgreSQL, MySQL, тощо)
463    //    - Використовувати async database driver (sqlx, diesel, sea-orm)
464    //    - Зберігати connection pool в global state або config
465    // 2. Запит до бази для перевірки користувача
466    //    - SELECT * FROM users WHERE username = ? AND password_hash = ?
467    //    - Використовувати prepared statements для безпеки
468    //    - Хешувати пароль перед порівнянням (bcrypt, argon2, scrypt)
469    // 3. Перевірка пароля
470    //    - Використовувати secure password hashing (argon2 рекомендовано)
471    //    - Перевіряти password hash з stored hash
472    //    - Handle timing attacks (constant-time comparison)
473    // 4. Отримання ролі користувача
474    //    - Завантажувати role з users table або roles join table
475    //    - Map database role to UserRole enum
476    // 5. Error handling
477    //    - Handle database connection errors gracefully
478    //    - Return generic error for invalid credentials (don't leak user existence)
479    //    - Log authentication failures for security monitoring
480    // Example:
481    //    let user = db.query_user_by_username(&auth_req.username).await?;
482    //    if !verify_password(&auth_req.password, &user.password_hash)? {
483    //        return Err((StatusCode::UNAUTHORIZED, Json(json!({"error": "Invalid credentials"}))));
484    //    }
485    //    let role = UserRole::from_str(&user.role)?;
486    // Use UserManager for authentication
487    let manager = get_global_user_manager();
488
489    // Ensure manager is initialized
490    if let Err(e) = manager.initialize().await {
491        return Err((
492            StatusCode::INTERNAL_SERVER_ERROR,
493            Json(serde_json::json!({
494                "error": format!("User manager initialization failed: {}", e)
495            })),
496        ));
497    }
498
499    // Verify password
500    let is_valid = manager
501        .verify_password(&auth_req.username, &auth_req.password)
502        .await
503        .map_err(|e| {
504            (
505                StatusCode::INTERNAL_SERVER_ERROR,
506                Json(serde_json::json!({
507                    "error": format!("Authentication error: {}", e)
508                })),
509            )
510        })?;
511
512    if !is_valid {
513        return Err((
514            StatusCode::UNAUTHORIZED,
515            Json(serde_json::json!({
516                "error": "Invalid credentials"
517            })),
518        ));
519    }
520
521    // Get user to retrieve role
522    let user = manager
523        .get_user_by_username(&auth_req.username)
524        .await
525        .map_err(|e| {
526            (
527                StatusCode::INTERNAL_SERVER_ERROR,
528                Json(serde_json::json!({
529                    "error": format!("User retrieval error: {}", e)
530                })),
531            )
532        })?;
533
534    let (role, username) = if let Some(u) = user {
535        (u.role, u.username)
536    } else {
537        return Err((
538            StatusCode::UNAUTHORIZED,
539            Json(serde_json::json!({
540                "error": "Invalid credentials"
541            })),
542        ));
543    };
544
545    let token = generate_token(&username, role.clone()).map_err(|_| {
546        (
547            StatusCode::INTERNAL_SERVER_ERROR,
548            Json(serde_json::json!({
549                "error": "Failed to generate token"
550            })),
551        )
552    })?;
553
554    let config = JwtConfig::default();
555
556    Ok(AuthResponse {
557        token,
558        token_type: "Bearer".to_string(),
559        expires_in: config.expiration,
560        role,
561    })
562}

748        let mut username_index = self.username_index.write().await;
749        let now = chrono::Utc::now();
750
751        for (username, password, role) in default_users {
752            let id = Uuid::new_v4();
753            let user = User {
754                id,
755                username: username.to_string(),
756                password_hash: password.to_string(), // In production, use proper password hashing
757                role,
758                active: true,
759                created_at: now,
760                updated_at: now,
761            };
762            users.insert(id, user.clone());
763            username_index.insert(username.to_string(), id);
764        }
765
766        *initialized = true;
767        info!(
768            "User manager initialized with {} default users",
769            users.len()
770        );
771        Ok(())
772    }

835    /// Updates a user
836    pub async fn update_user(
837        &self,
838        id: Uuid,
839        username: Option<String>,
840        password: Option<String>,
841        role: Option<UserRole>,
842        active: Option<bool>,
843    ) -> Result<UserInfo, String> {
844        let mut users = self.users.write().await;
845        let mut username_index = self.username_index.write().await;
846
847        let user = users.get_mut(&id).ok_or_else(|| {
848            format!("User not found: {}. Context: Cannot update non-existent user. Suggestion: Check user ID and ensure user exists.", id)
849        })?;
850
851        if let Some(new_username) = username {
852            if new_username != user.username {
853                // Check if new username already exists
854                if username_index.contains_key(&new_username) {
855                    return Err(format!("Username '{}' already exists", new_username));
856                }
857                // Update username index
858                username_index.remove(&user.username);
859                username_index.insert(new_username.clone(), id);
860                user.username = new_username;
861            }
862        }
863
864        if let Some(new_password) = password {
865            user.password_hash = new_password; // In production, use proper password hashing
866        }
867
868        if let Some(new_role) = role {
869            user.role = new_role;
870        }
871
872        if let Some(new_active) = active {
873            user.active = new_active;
874        }
875
876        user.updated_at = chrono::Utc::now();
877        let updated_user = user.clone();
878
879        info!("Updated user: {} ({})", updated_user.username, id);
880        Ok(updated_user.into())
881    }

885        let username = {
886            let users = self.users.read().await;
887            let user = users.get(&id).ok_or_else(|| {
888                format!("User not found: {}. Context: Cannot delete non-existent user. Suggestion: Check user ID.", id)
889            })?;
890            user.username.clone()
891        };
892
893        let mut users = self.users.write().await;
894        let mut username_index = self.username_index.write().await;
895
896        username_index.remove(&username);
897        users.remove(&id);
898
899        info!("Deleted user: {} ({})", username, id);
900        Ok(())
901    }

904    pub async fn verify_password(&self, username: &str, password: &str) -> Result<bool, String> {
905        if let Some(user) = self.get_user_by_username(username).await? {
906            if !user.active {
907                return Ok(false);
908            }
909            // In production, use proper password verification (bcrypt, argon2, etc.)
910            Ok(user.password_hash == password)
911        } else {
912            Ok(false)
913        }
