20/// Handle PutArtifact protocol message
21pub async fn put_artifact_handler(
22    State(ctx): State<ApiContext>,
23    Json(message): Json<ProtocolMessage>,
24) -> impl IntoResponse {
25    info!("Received PutArtifact message: {}", message.id);
26
27    let payload = match message.extract_put_artifact() {
28        Ok(p) => p,
29        Err(e) => {
30            error!("Failed to extract PutArtifact payload: {}", e);
31            return create_error_response(
32                &message,
33                ErrorCode::InvalidRequest,
34                format!("Invalid payload: {}", e),
35            );
36        }
37    };
38
39    // Decode base64 data if present
40    let artifact_data = if let Some(data_base64) = payload.data {
41        match base64_decode(&data_base64) {
42            Ok(data) => Some(data),
43            Err(e) => {
44                error!("Failed to decode base64 data: {}", e);
45                return create_error_response(
46                    &message,
47                    ErrorCode::InvalidRequest,
48                    format!("Invalid base64 data: {}", e),
49                );
50            }
51        }
52    } else {
53        None
54    };
55
56    // Store artifact locally (same path as REST via RaidService)
57    let artifact_name = format!("{}-{}", payload.metadata.name, payload.metadata.version);
58    match artifact_data {
59        Some(data) => match RaidService::put_artifact(&ctx, &artifact_name, &data).await {
60            Ok(artifact_ref) => {
61                info!("Artifact stored successfully: {}", artifact_ref.id);
62                let response = PutArtifactResponse {
63                    status: OperationStatus::Success,
64                    artifact_id: payload.artifact_id,
65                    stored_at: Utc::now(),
66                    error: None,
67                };
68                create_success_response(&message, response)
69            }
70            Err(RaidServiceError::ManagerUnavailable) => create_error_response(
71                &message,
72                ErrorCode::InvalidRequest,
73                "RAID manager not initialized".to_string(),
74            ),
75            Err(RaidServiceError::Operation(e)) => {
76                error!("Failed to store artifact: {}", e);
77                create_error_response(
78                    &message,
79                    ErrorCode::ReplicationFailed,
80                    format!("Storage failed: {}", e),
81                )
82            }
83            Err(e) => {
84                error!("Failed to store artifact: {:?}", e);
85                create_error_response(
86                    &message,
87                    ErrorCode::ReplicationFailed,
88                    format!("Storage failed: {:?}", e),
89                )
90            }
91        },
92        None => {
93            // Metadata-only replication (data will be fetched separately)
94            warn!("PutArtifact without data - metadata-only replication");
95            let response = PutArtifactResponse {
96                status: OperationStatus::Success,
97                artifact_id: payload.artifact_id,
98                stored_at: Utc::now(),
99                error: None,
100            };
101            create_success_response(&message, response)
102        }
103    }
104}

573fn create_success_response<T: serde::Serialize>(
574    original_message: &ProtocolMessage,
575    payload: T,
576) -> Response {
577    let response_message = ProtocolMessage {
578        message_type: format!("{}_response", original_message.message_type),
579        id: Uuid::new_v4().to_string(),
580        timestamp: Utc::now(),
581        // Future improvement: Use actual node ID instead of copying from request
582        // 1. Get node ID from RaidManager using manager.get_node_id().await
583        //    - This returns the actual node ID from Raft configuration
584        //    - More reliable than trusting client-provided node_id
585        // 2. Handle errors gracefully if node ID cannot be retrieved
586        //    - Fallback to original_message.node_id if get_node_id fails
587        // Example: node_id: manager.get_node_id().await.to_string()
588        // Note: For now, using original_message.node_id for compatibility
589        node_id: original_message.node_id.clone(),
590        payload: serde_json::to_value(payload).unwrap_or(serde_json::Value::Null),
591    };
592
593    (StatusCode::OK, Json(response_message)).into_response()
594}

596fn create_error_response(
597    original_message: &ProtocolMessage,
598    error_code: ErrorCode,
599    error_message: String,
600) -> Response {
601    let error = ProtocolError {
602        error_code,
603        error_message,
604        details: None,
605    };
606
607    let response_message = ProtocolMessage {
608        message_type: "error".to_string(),
609        id: Uuid::new_v4().to_string(),
610        timestamp: Utc::now(),
611        node_id: original_message.node_id.clone(),
612        payload: serde_json::to_value(error).unwrap_or(serde_json::Value::Null),
613    };
614
615    (StatusCode::BAD_REQUEST, Json(response_message)).into_response()
616}

618fn base64_decode(data: &str) -> Result<Vec<u8>, AppError> {
619    use base64::{engine::general_purpose, Engine as _};
620    general_purpose::STANDARD
621        .decode(data)
622        .map_err(|e| AppError::ValidationError(format!(
623            "Base64 decode error: {}. Context: Failed to decode base64-encoded artifact data. \
624            Suggestion: Verify the data is valid base64 encoding. Ensure no whitespace or invalid characters are present. \
625            Error details: {}, Data length: {}",
626            e, e, data.len()
627        )))
