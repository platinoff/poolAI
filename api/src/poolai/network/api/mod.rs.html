43/// Create API routes
44///
45/// Composes all API endpoint modules into a single router.
46///
47/// Composes modular `api/*` routers (canonical REST surface under `/api/v1`).
48pub fn create_api_routes() -> Router<ApiContext> {
49    Router::new()
50        .merge(admin::create_admin_routes())
51        .merge(system::create_system_routes())
52        .merge(workers::create_workers_routes())
53        .merge(rewards::create_rewards_routes())
54        .merge(vm::create_vm_routes())
55        .merge(raid::create_raid_routes())
56        .merge(raid_admin::create_raid_admin_routes())
57        .merge(libraries::create_libraries_routes())
58        .merge(users::create_users_routes())
59        .merge(discovery::create_discovery_routes())
60        .merge(virtual_nodes::create_virtual_node_routes())
61        .merge(instances::create_instance_routes())
62        .merge(completions::create_completions_routes())
63        .merge(topology::create_topology_routes())
64        .merge(ui::create_ui_routes())
65}
