37#[cfg(feature = "enterprise")]
38fn ui_monitoring_http_err(
39    e: EnterpriseMonitoringError,
40    api_code: &'static str,
41    operation: &'static str,
42    message_for_operation: impl Into<String>,
43) -> UiHttpErr {
44    match e {
45        EnterpriseMonitoringError::Init(err) => api_json_error(
46            "MONITORING_MANAGER_UNAVAILABLE",
47            format!("Monitoring manager not initialized: {}", err),
48            Some(ErrorContext::new(operation).with_hint("Check system startup sequence.")),
49            StatusCode::SERVICE_UNAVAILABLE,
50        ),
51        EnterpriseMonitoringError::Operation(err) => api_json_error(
52            api_code,
53            format!("{}: {}", message_for_operation.into(), err),
54            Some(ErrorContext::new(operation)),
55            StatusCode::INTERNAL_SERVER_ERROR,
56        ),
57    }
58}

60/// Create UI management routes
61pub fn create_ui_routes() -> Router<ApiContext> {
62    Router::new()
63        // Dashboard management endpoints
64        .route("/ui/dashboards", get(ui_dashboards_handler))
65        .route(
66            "/ui/dashboards",
67            post(ui_dashboard_create_handler).layer(middleware::from_fn(auth_middleware)),
68        )
69        .route("/ui/dashboards/{id}", get(ui_dashboard_get_handler))
70        .route(
71            "/ui/dashboards/{id}",
72            put(ui_dashboard_update_handler).layer(middleware::from_fn(auth_middleware)),
73        )
74        .route(
75            "/ui/dashboards/{id}",
76            delete(ui_dashboard_delete_handler).layer(middleware::from_fn(auth_middleware)),
77        )
78        // Theme management endpoints
79        .route("/ui/themes", get(ui_themes_handler))
80        .route("/ui/themes/{name}", get(ui_theme_get_handler))
81        // Component registry endpoints
82        .route("/ui/components", get(ui_components_handler))
83        .route("/ui/components/{name}", get(ui_component_get_handler))
84}

90/// Request structure for creating a dashboard
91///
92/// Note: Fields are used via move semantics in handlers, which may not be detected by the compiler.
93/// This is part of the API contract and fields are actively used in dashboard creation.
94#[derive(Deserialize)]
95#[allow(dead_code)] // Fields are used via move in handlers
96struct CreateDashboardRequest {
97    name: String,
98    description: String,
99    metrics: Vec<String>,
100    layout: String,
101    is_public: Option<bool>,
102    tenant_id: Option<Uuid>,
103}

105#[cfg(feature = "enterprise")]
106async fn ui_dashboards_handler(State(ctx): State<ApiContext>) -> impl IntoResponse {
107    match UiService::list_dashboards(&ctx).await {
108        Ok(dashboards) => AxumJson(dashboards).into_response(),
109        Err(e) => {
110            let (s, j) = ui_monitoring_http_err(
111                e,
112                "LIST_DASHBOARDS_FAILED",
113                "ui_dashboards_list",
114                "Failed to list dashboards",
115            );
116            (s, AxumJson(j.0)).into_response()
117        }
118    }
119}

135#[cfg(feature = "enterprise")]
136async fn ui_dashboard_get_handler(
137    State(ctx): State<ApiContext>,
138    Path(id): Path<String>,
139) -> impl IntoResponse {
140    let uuid = match Uuid::parse_str(&id) {
141        Ok(u) => u,
142        Err(_) => {
143            let (s, j) = api_json_error(
144                "INVALID_UUID",
145                format!("Invalid UUID format: {}", id),
146                Some(ErrorContext::new("ui_dashboard_get").with_resource("dashboard_id", &id)),
147                StatusCode::BAD_REQUEST,
148            );
149            return (s, AxumJson(j.0)).into_response();
150        }
151    };
152
153    match UiService::get_dashboard(&ctx, uuid).await {
154        Ok(Some(dashboard)) => AxumJson(dashboard).into_response(),
155        Ok(None) => {
156            let (s, j) = api_json_error(
157                "DASHBOARD_NOT_FOUND",
158                format!("Dashboard not found: {}", id),
159                Some(ErrorContext::new("ui_dashboard_get").with_resource("dashboard_id", &id)),
160                StatusCode::NOT_FOUND,
161            );
162            (s, AxumJson(j.0)).into_response()
163        }
164        Err(e) => {
165            let (s, j) = ui_monitoring_http_err(
166                e,
167                "GET_DASHBOARD_FAILED",
168                "ui_dashboard_get",
169                "Failed to get dashboard",
170            );
171            (s, AxumJson(j.0)).into_response()
172        }
173    }
174}

190#[cfg(feature = "enterprise")]
191async fn ui_dashboard_create_handler(
192    State(ctx): State<ApiContext>,
193    Extension(claims): Extension<Claims>,
194    Json(payload): Json<CreateDashboardRequest>,
195) -> impl IntoResponse {
196    if let Err(err) = check_permission(&claims, "write:all")
197        .or_else(|_| check_permission(&claims, "write:monitoring"))
198    {
199        return err.into_response();
200    }
201
202    if payload.name.is_empty() {
203        let (s, j) = api_json_error(
204            "VALIDATION_ERROR",
205            "Dashboard name cannot be empty",
206            Some(ErrorContext::new("ui_dashboard_create").with_resource("field", "name")),
207            StatusCode::BAD_REQUEST,
208        );
209        return (s, AxumJson(j.0)).into_response();
210    }
211
212    let input = DashboardCreateInput {
213        name: payload.name,
214        description: Some(payload.description),
215        metrics: payload.metrics,
216        layout: Some(payload.layout),
217        is_public: payload.is_public,
218        tenant_id: payload.tenant_id.map(|u| u.to_string()),
219    };
220
221    match UiService::create_dashboard(&ctx, input).await {
222        Ok(dashboard) => (StatusCode::CREATED, AxumJson(dashboard)).into_response(),
223        Err(e) => {
224            let (s, j) = ui_monitoring_http_err(
225                e,
226                "CREATE_DASHBOARD_FAILED",
227                "ui_dashboard_create",
228                "Failed to create dashboard",
229            );
230            (s, AxumJson(j.0)).into_response()
231        }
232    }
233}
