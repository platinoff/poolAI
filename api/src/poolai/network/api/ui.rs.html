25/// Create UI management routes
26pub fn create_ui_routes() -> Router {
27    Router::new()
28        // Dashboard management endpoints
29        .route("/ui/dashboards", get(ui_dashboards_handler))
30        .route(
31            "/ui/dashboards",
32            post(ui_dashboard_create_handler).layer(middleware::from_fn(auth_middleware)),
33        )
34        .route(
35            "/ui/dashboards/{id}",
36            get(ui_dashboard_get_handler),
37        )
38        .route(
39            "/ui/dashboards/{id}",
40            put(ui_dashboard_update_handler).layer(middleware::from_fn(auth_middleware)),
41        )
42        .route(
43            "/ui/dashboards/{id}",
44            delete(ui_dashboard_delete_handler).layer(middleware::from_fn(auth_middleware)),
45        )
46        // Theme management endpoints
47        .route("/ui/themes", get(ui_themes_handler))
48        .route(
49            "/ui/themes/{name}",
50            get(ui_theme_get_handler),
51        )
52        // Component registry endpoints
53        .route("/ui/components", get(ui_components_handler))
54        .route(
55            "/ui/components/{name}",
56            get(ui_component_get_handler),
57        )
58}

74#[cfg(feature = "enterprise")]
75async fn ui_dashboards_handler() -> impl IntoResponse {
76    let manager = get_global_monitoring_manager();
77    match manager.list_dashboards(None).await {
78        Ok(dashboards) => AxumJson(dashboards).into_response(),
79        Err(e) => (
80            StatusCode::INTERNAL_SERVER_ERROR,
81            AxumJson(serde_json::json!({
82                "error": format!("Failed to list dashboards: {}", e)
83            })),
84        )
85            .into_response(),
86    }
87}

101async fn ui_dashboard_get_handler(Path(id): Path<String>) -> impl IntoResponse {
102    let manager = get_global_monitoring_manager();
103    let uuid = match Uuid::parse_str(&id) {
104        Ok(u) => u,
105        Err(_) => {
106            return (
107                StatusCode::BAD_REQUEST,
108                AxumJson(serde_json::json!({
109                    "error": format!("Invalid UUID format: {}", id)
110                })),
111            )
112                .into_response();
113        }
114    };
115
116    match manager.get_dashboard(uuid).await {
117        Ok(Some(dashboard)) => AxumJson(dashboard).into_response(),
118        Ok(None) => (
119            StatusCode::NOT_FOUND,
120            AxumJson(serde_json::json!({
121                "error": format!("Dashboard not found: {}", id)
122            })),
123        )
124            .into_response(),
125        Err(e) => (
126            StatusCode::INTERNAL_SERVER_ERROR,
127            AxumJson(serde_json::json!({
128                "error": format!("Failed to get dashboard: {}", e)

146#[cfg(feature = "enterprise")]
147async fn ui_dashboard_create_handler(
148    Extension(claims): Extension<Claims>,
149    Json(payload): Json<CreateDashboardRequest>,
150) -> impl IntoResponse {
151    if let Err(err) = check_permission(&claims, "write:all")
152        .or_else(|_| check_permission(&claims, "write:monitoring"))
153    {
154        return err.into_response();
155    }
156
157    if payload.name.is_empty() {
158        return (
159            StatusCode::BAD_REQUEST,
160            AxumJson(serde_json::json!({
161                "error": "Dashboard name cannot be empty".to_string()
162            })),
163        )
164            .into_response();
165    }
166
167    let manager = get_global_monitoring_manager();
168    let dashboard = Dashboard {
169        id: Uuid::new_v4(),
170        name: payload.name.clone(),
171        description: payload.description,
172        metrics: payload.metrics,
173        layout: payload.layout,
174        is_public: payload.is_public.unwrap_or(false),
175        tenant_id: payload.tenant_id,
176        created_at: chrono::Utc::now(),
177    };
178
179    match manager.create_dashboard(dashboard.clone()).await {
180        Ok(_) => (StatusCode::CREATED, AxumJson(dashboard)).into_response(),
181        Err(e) => (
182            StatusCode::INTERNAL_SERVER_ERROR,
183            AxumJson(serde_json::json!({
184                "error": format!("Failed to create dashboard: {}", e)
185            })),
186        )
187            .into_response(),
188    }
189}

205#[cfg(feature = "enterprise")]
206async fn ui_dashboard_update_handler(
207    Extension(claims): Extension<Claims>,
208    Path(id): Path<String>,
209    Json(payload): Json<CreateDashboardRequest>,
210) -> impl IntoResponse {
211    if let Err(err) = check_permission(&claims, "write:all")
212        .or_else(|_| check_permission(&claims, "write:monitoring"))
213    {
214        return err.into_response();
215    }
216
217    let manager = get_global_monitoring_manager();
218    let uuid = match Uuid::parse_str(&id) {
219        Ok(u) => u,
220        Err(_) => {
221            return (
222                StatusCode::BAD_REQUEST,
223                AxumJson(serde_json::json!({
224                    "error": format!("Invalid UUID format: {}", id)
225                })),
226            )
227                .into_response();
228        }
229    };
230
231    // Get existing dashboard first
232    let existing = match manager.get_dashboard(uuid).await {
233        Ok(Some(d)) => d,
234        Ok(None) => {
235            return (
236                StatusCode::NOT_FOUND,
237                AxumJson(serde_json::json!({
238                    "error": format!("Dashboard not found: {}", id)
239                })),
240            )
241                .into_response();
242        }
243        Err(e) => {
244            return (
245                StatusCode::INTERNAL_SERVER_ERROR,
246                AxumJson(serde_json::json!({
247                    "error": format!("Failed to get dashboard: {}", e)
248                })),
249            )
250                .into_response();
251        }
252    };
253
254    // Update dashboard
255    let updated = Dashboard {
256        id: existing.id,
257        name: payload.name,
258        description: payload.description,
259        metrics: payload.metrics,
260        layout: payload.layout,
261        is_public: payload.is_public.unwrap_or(existing.is_public),
262        tenant_id: payload.tenant_id.or(existing.tenant_id),
263        created_at: existing.created_at,
264    };
265
266    match manager.create_dashboard(updated.clone()).await {
267        Ok(_) => AxumJson(updated).into_response(),
268        Err(e) => (
269            StatusCode::INTERNAL_SERVER_ERROR,
270            AxumJson(serde_json::json!({
271                "error": format!("Failed to update dashboard: {}", e)
272            })),
273        )
274            .into_response(),
275    }
276}

334#[derive(Serialize)]
335struct ThemeResponse {
336    name: String,
337    css_variables: String,
338    css: String,
339}
340
341async fn ui_themes_handler() -> impl IntoResponse {
342    let themes = get_all_themes();
343    let response: Vec<ThemeResponse> = themes
344        .into_iter()
345        .map(|theme| ThemeResponse {
346            name: theme.name.to_string(),
347            css_variables: theme.to_css_variables(),
348            css: theme.to_css(),
349        })
350        .collect();
351    AxumJson(response).into_response()
352}
353
354async fn ui_theme_get_handler(Path(name): Path<String>) -> impl IntoResponse {
355    let theme = get_theme(&name);
356    let response = ThemeResponse {
357        name: theme.name.to_string(),
358        css_variables: theme.to_css_variables(),
359        css: theme.to_css(),
360    };
361    AxumJson(response).into_response()
362}

408            styles: components::BUTTON_STYLES.to_string(),

410        },
411        "card" => ComponentInfo {
412            name: "card".to_string(),
413            component_type: "card".to_string(),
414            styles: components::CARD_STYLES.to_string(),
415            description: Some("Card component for content containers".to_string()),
416        },
417        "form" => ComponentInfo {
418            name: "form".to_string(),
419            component_type: "form".to_string(),
420            styles: components::FORM_STYLES.to_string(),
421            description: Some("Form component for input fields and validation".to_string()),
