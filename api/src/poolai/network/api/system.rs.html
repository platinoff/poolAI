35    #[tokio::test]
36    async fn status_handler_works_with_api_context() {
37        let app_state = ApiContext::default();
38
39        let app = create_system_routes().with_state(app_state);
40
41        let response = app
42            .oneshot(
43                HttpRequest::builder()
44                    .uri("/status")
45                    .body(Body::empty())
46                    .unwrap(),
47            )
48            .await
49            .expect("request should succeed");
50
51        assert_eq!(response.status(), StatusCode::OK);
52    }

55/// Create system routes
56pub fn create_system_routes() -> Router<ApiContext> {
57    Router::new()
58        .route("/status", get(status_handler))
59        .route("/health", get(health_handler))
60        .route("/login", axum::routing::post(login_handler))
61        .route("/metrics", get(metrics_handler))
62        .route("/models", get(models_handler))
63        .route("/gpu", get(gpu_info))
64        .route("/ws/metrics", get(websocket_handler))
65        .route("/config", get(config_get_handler))
66        .route(
67            "/config",
68            put(config_update_handler)
69                .layer(middleware::from_fn(crate::network::auth::auth_middleware)),
70        )
71}

505async fn login_handler(
506    State(ctx): State<ApiContext>,
507    Json(auth_req): Json<AuthRequest>,
508) -> impl IntoResponse {
509    match SystemService::login(auth_req, ctx.user_manager.clone()).await {
510        Ok(auth_response) => Json(auth_response).into_response(),
511        Err((status, error)) => (status, error).into_response(),
512    }
513}
514
515/// Get system configuration
516async fn config_get_handler() -> impl IntoResponse {
517    match SystemService::get_configuration() {
518        Ok(config) => Json(config).into_response(),
519        Err(e) => {
520            let (s, j) = api_json_error(
521                "CONFIG_GET_FAILED",
522                format!("Failed to get configuration: {}", e),
523                Some(ErrorContext::new("config_get")),
524                StatusCode::INTERNAL_SERVER_ERROR,
525            );
526            (s, j).into_response()
527        }
528    }
529}

531/// Update system configuration
532async fn config_update_handler(
533    Extension(claims): Extension<Claims>,
534    Json(config): Json<PoolAIConfig>,
535) -> impl IntoResponse {
536    // Check permission: admin:all
537    if let Err(err) = check_permission(&claims, "admin:all") {
538        return err.into_response();
539    }
540
541    match SystemService::apply_configuration(config) {
542        Ok(()) => Json(serde_json::json!({
543            "message": "Configuration updated successfully"
544        }))
545        .into_response(),
546        Err(e) => {
547            let (s, j) = api_json_error(
548                "CONFIG_UPDATE_FAILED",
549                format!("Failed to update configuration: {}", e),
550                Some(ErrorContext::new("config_update")),
551                StatusCode::BAD_REQUEST,
552            );
553            (s, j).into_response()
554        }
555    }
556}
