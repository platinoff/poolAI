29#[derive(Deserialize)]
30struct CreateWorkerRequest {
31    worker_id: String,
32    max_concurrent_requests: Option<usize>,
33    request_timeout_ms: Option<u64>,
34    health_check_interval_ms: Option<u64>,
35    enable_caching: Option<bool>,
36    cache_size: Option<usize>,
37    max_memory_mb: Option<usize>,
38    cpu_priority: Option<u8>,
39    gpu_device: Option<usize>,
40    auto_restart: Option<bool>,
41    resource_monitoring: Option<bool>,
42}

44#[derive(Serialize)]
45struct CreateWorkerResponse {
46    worker_id: String,
47    message: String,
48}

50#[derive(Serialize)]
51struct DeleteWorkerResponse {

53    message: String,

56/// Create worker management routes
57pub fn create_workers_routes() -> Router<ApiContext> {
58    Router::new()
59        .route("/workers", get(workers_handler))
60        .route(
61            "/workers",
62            post(worker_create_handler).layer(axum::middleware::from_fn(auth_middleware)),
63        )
64        .route(
65            "/workers/{id}",
66            delete(worker_delete_handler).layer(axum::middleware::from_fn(auth_middleware)),
67        )
68}

70async fn workers_handler(
71    State(ctx): State<ApiContext>,
72) -> Result<AxumJson<Vec<WorkerInfo>>, AppError> {
73    Ok(AxumJson(WorkerPoolService::list_workers(&ctx).await))

76async fn worker_create_handler(
77    State(ctx): State<ApiContext>,
78    Json(payload): Json<CreateWorkerRequest>,
79) -> impl IntoResponse {
80    if let Err(e) = validation::validate_worker_id(&payload.worker_id) {
81        let (s, j) = api_json_error(
82            "VALIDATION_ERROR",
83            e.to_string(),
84            Some(ErrorContext::new("create_worker").with_resource("worker_id", &payload.worker_id)),
85            StatusCode::BAD_REQUEST,
86        );
87        return (s, AxumJson(j.0)).into_response();
88    }
89
90    let max_concurrent_requests = payload.max_concurrent_requests.unwrap_or(10);
91    let request_timeout_ms = payload.request_timeout_ms.unwrap_or(5000);
92    let health_check_interval_ms = payload.health_check_interval_ms.unwrap_or(1000);
93    let cache_size = payload.cache_size.unwrap_or(1000);
94    let max_memory_mb = payload.max_memory_mb.unwrap_or(2048);
95    let cpu_priority = payload.cpu_priority.unwrap_or(5);
96
97    if let Err(e) = validation::validate_worker_config(
98        max_concurrent_requests,
99        request_timeout_ms,
100        health_check_interval_ms,
101        cache_size,
102        max_memory_mb,
103        cpu_priority,
104    ) {
105        let (s, j) = api_json_error(
106            "VALIDATION_ERROR",
107            e.to_string(),
108            Some(ErrorContext::new("create_worker").with_resource("worker_id", &payload.worker_id)),
109            StatusCode::BAD_REQUEST,
110        );
111        return (s, AxumJson(j.0)).into_response();
112    }
113
114    let input = CreateWorkerInput {
115        worker_id: payload.worker_id.clone(),
116        max_concurrent_requests,
117        request_timeout_ms,
118        health_check_interval_ms,
119        enable_caching: payload.enable_caching.unwrap_or(true),
120        cache_size,
121        max_memory_mb,
122        cpu_priority,
123        gpu_device: payload.gpu_device,
124        auto_restart: payload.auto_restart.unwrap_or(true),
125        resource_monitoring: payload.resource_monitoring.unwrap_or(true),
126    };
127
128    match WorkerPoolService::add_worker(&ctx, input).await {
129        Ok(()) => {
130            let response = CreateWorkerResponse {
131                worker_id: payload.worker_id,
132                message: "Worker created successfully".to_string(),
133            };
134            (StatusCode::CREATED, AxumJson(response)).into_response()
135        }
136        Err(AddWorkerError::PoolNotReady) => {
137            let (s, j) = api_json_error(
138                "SUBSYSTEM_UNAVAILABLE",
139                "Pool not initialized; worker pool manager is not available",
140                Some(ErrorContext::new("create_worker").with_resource("pool", "default")),
141                StatusCode::SERVICE_UNAVAILABLE,
142            );
143            (s, AxumJson(j.0)).into_response()
144        }
145        Err(AddWorkerError::Operation(e)) => {
146            let (s, j) = api_json_error(
147                "INTERNAL_ERROR",
148                format!(
149                    "Failed to create worker: {} (worker_id='{}')",
150                    e, payload.worker_id
151                ),
152                Some(
153                    ErrorContext::new("create_worker")
154                        .with_resource("worker_id", &payload.worker_id),
155                ),
156                StatusCode::INTERNAL_SERVER_ERROR,
157            );
158            (s, AxumJson(j.0)).into_response()
159        }
160    }
161}
162
163async fn worker_delete_handler(
164    State(ctx): State<ApiContext>,
165    Path(worker_id): Path<String>,
166) -> impl IntoResponse {
167    match WorkerPoolService::remove_worker(&ctx, &worker_id).await {
168        Ok(()) => {
169            let response = DeleteWorkerResponse {
170                worker_id,
171                message: "Worker deleted successfully".to_string(),
172            };
173            AxumJson(response).into_response()
174        }
175        Err(RemoveWorkerError::PoolNotReady) => {
176            let (s, j) = api_json_error(
177                "SUBSYSTEM_UNAVAILABLE",
178                "Pool not initialized; worker pool manager is not available",
179                Some(ErrorContext::new("delete_worker").with_resource("pool", "default")),
180                StatusCode::SERVICE_UNAVAILABLE,
181            );
182            (s, AxumJson(j.0)).into_response()
183        }
184        Err(RemoveWorkerError::Operation(e)) => {
185            let (s, j) = api_json_error(
186                "NOT_FOUND",
187                format!("Failed to delete worker: {} (worker_id='{}')", e, worker_id),
188                Some(ErrorContext::new("delete_worker").with_resource("worker_id", &worker_id)),
189                StatusCode::NOT_FOUND,
190            );
191            (s, AxumJson(j.0)).into_response()
192        }
193    }
194}
