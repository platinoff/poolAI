1//! Admin aggregation API (JSON for `/ui/admin` and tools).
2
3use crate::core::state::ApiContext;
4use crate::services::admin_service::{AdminOverview, AdminService};
5use axum::{extract::State, routing::get, Json, Router};
6
7async fn admin_overview_handler(State(ctx): State<ApiContext>) -> Json<AdminOverview> {
8    Json(AdminService::overview(&ctx).await)
9}
10
11/// Routes under `/api/v1` (see `network::start_server` nest).
12pub fn create_admin_routes() -> Router<ApiContext> {
13    Router::new().route("/admin/overview", get(admin_overview_handler))
14}
15
16#[cfg(test)]
17mod tests {
18    use super::*;
19    use axum::body::Body;
20    use axum::http::{Request, StatusCode};
21    use tower::ServiceExt;
22
23    #[tokio::test]
24    async fn admin_overview_returns_ok_json() {
25        let ctx = ApiContext::default();
26        ctx.initialize().await.expect("init");
27        let app = create_admin_routes().with_state(ctx);
28        let res = app
29            .oneshot(
30                Request::builder()
31                    .uri("/admin/overview")
32                    .body(Body::empty())
33                    .unwrap(),
34            )
35            .await
36            .expect("oneshot");
37        assert_eq!(res.status(), StatusCode::OK);
38    }
39}
