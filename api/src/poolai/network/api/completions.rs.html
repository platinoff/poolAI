21/// OpenAI-compatible chat message
22#[derive(Debug, Clone, Serialize, Deserialize)]
23pub struct ChatMessage {
24    /// Message role (system, user, assistant)
25    pub role: String,
26    /// Message content
27    pub content: String,
28}

30/// OpenAI-compatible chat completion request
31#[derive(Debug, Deserialize)]
32pub struct ChatCompletionRequest {
33    /// Model name/ID
34    pub model: String,
35    /// Chat messages
36    pub messages: Vec<ChatMessage>,
37    /// Temperature (0.0-2.0)
38    #[serde(default = "default_temperature")]
39    pub temperature: f32,
40    /// Max tokens
41    #[serde(default = "default_max_tokens")]
42    pub max_tokens: usize,
43    /// Stream response
44    #[serde(default)]
45    pub stream: bool,
46    /// Top-p sampling
47    #[serde(default = "default_top_p")]
48    pub top_p: f32,
49    /// Frequency penalty
50    #[serde(default)]
51    pub frequency_penalty: f32,
52    /// Presence penalty
53    #[serde(default)]
54    pub presence_penalty: f32,
55    /// Stop sequences
56    #[serde(default)]
57    pub stop: Option<Vec<String>>,
58}

153/// Handler for POST /v1/chat/completions
154/// OpenAI-compatible chat completions endpoint
155async fn chat_completions_handler(
156    Extension(_claims): Extension<Claims>,
157    Json(request): Json<ChatCompletionRequest>,
158) -> impl IntoResponse {
159    // Convert OpenAI request to internal ModelRequest
160    let prompt = request
161        .messages
162        .iter()
163        .map(|m| format!("{}: {}", m.role, m.content))
164        .collect::<Vec<_>>()
165        .join("\n");
166
167    let model_params = ModelParameters {
168        temperature: request.temperature,
169        max_tokens: request.max_tokens,
170        top_p: request.top_p,
171        frequency_penalty: request.frequency_penalty,
172        presence_penalty: request.presence_penalty,
173        stop_sequences: request.stop.unwrap_or_default(),
174        seed: None,
175    };
176
177    let model_request = ModelRequest {
178        input: prompt,
179        parameters: model_params,
180        session_id: None,
181        priority: 5,
182        timeout: Some(30),
183    };
184
185    // Process request (simplified - in real implementation would use instance manager)
186    if request.stream {
187        // Streaming response
188        let stream = create_streaming_response(&request.model, model_request);
189        Sse::new(stream).into_response()
190    } else {
191        // Non-streaming response
192        let response = process_chat_completion(&request.model, model_request).await;
193
194        match response {
195            Ok(completion) => (StatusCode::OK, Json(completion)).into_response(),
196            Err(e) => (
197                StatusCode::INTERNAL_SERVER_ERROR,
198                Json(serde_json::json!({"error": {"message": e.to_string(), "type": "internal_error"}})),
199            )
200                .into_response(),
201        }
202    }
203}

239/// Create streaming response
240fn create_streaming_response(
241    model: &str,
242    request: ModelRequest,
243) -> impl Stream<Item = Result<Event, Infallible>> {
244    let model = model.to_string();
245    let response_id = format!("chatcmpl-{}", uuid::Uuid::new_v4());
246    let created = chrono::Utc::now().timestamp();
247
248    // Simulate streaming by chunking the response
249    let response_text = format!("Processed: {}", request.input);
250    let chunks: Vec<String> = response_text
251        .chars()
252        .collect::<Vec<_>>()
253        .chunks(5)
254        .map(|chunk| chunk.iter().collect())
255        .collect();
256
257    let chunks_len = chunks.len();
258    let model_clone = model.clone();
259    let response_id_clone = response_id.clone();
260
261    stream::iter(chunks.into_iter().enumerate().map(move |(index, chunk)| {
262        let is_first = index == 0;
263        let is_last = index == chunks_len - 1;
264
265        let chunk_data = ChatCompletionChunk {
266            id: response_id_clone.clone(),
267            object: "chat.completion.chunk".to_string(),
268            created,
269            model: model_clone.clone(),
270            choices: vec![ChatCompletionChunkChoice {
271                index: 0,
272                delta: ChatMessageDelta {
273                    role: if is_first {
274                        Some("assistant".to_string())
275                    } else {
276                        None
277                    },
278                    content: Some(chunk),
279                },
280                finish_reason: if is_last {
281                    Some("stop".to_string())
282                } else {
283                    None
284                },
285            }],
286        };
287
288        let json = serde_json::to_string(&chunk_data).unwrap_or_default();
289        Ok(Event::default().data(json))
290    }))
291    .chain(stream::iter(vec![Ok(Event::default().data("[DONE]"))]))
