125pub async fn ingest_grid_envelope_handler(
126    envelope: Json<GridEnvelope>,
127) -> Result<(StatusCode, Json<GridIngestResponse>), HttpAppError> {
128    let outcome = ingest_envelope(envelope.0, jobs(), memory())?;
129    Ok((StatusCode::OK, Json(response_from_outcome(outcome))))
130}

157async fn get_grid_seed_inventory(
158    State(_ctx): State<ApiContext>,
159) -> Result<(StatusCode, Json<GridSeedInventoryResponse>), HttpAppError> {
160    let entries = coordinator_seed_inventory_snapshot();
161    let persist = memory().persist_enabled();
162    let registered_count = memory().registered_shard_count()?;
163    let registered_ids = memory().registered_shard_ids()?;
164    let store_depth = memory_store_depth_stub(persist, registered_count);
165    let layer_depth = memory_layer_depth_stub(persist, registered_count, entries.len() as u32);
166    Ok((
167        StatusCode::OK,
168        Json(GridSeedInventoryResponse {
169            ok: true,
170            entries,
171            generated_at: chrono::Utc::now().to_rfc3339(),
172            memory_persist: persist,
173            registered_shard_count: registered_count,
174            memory_store_depth: memory_store_depth_wire_label(store_depth),
175            memory_layer_depth: memory_layer_depth_wire_label(layer_depth),
176            registered_shard_ids: registered_ids,
177        }),
178    ))
179}

244async fn get_grid_payout_batch(
245    State(_ctx): State<ApiContext>,
246) -> Result<(StatusCode, Json<GridPayoutBatchResponse>), HttpAppError> {
247    let mode = current_settlement_mode();
248    let entry = crate::grid::galaxy_settlement_metrics::last_payout_batch_ledger_entry();
249    let routing = entry.as_ref().map(|e| payout_routing_snapshot(mode, e));
250    let onchain_depth = current_settlement_onchain_depth();
251    let events_configured = crate::grid::galaxy_settlement_onchain::onchain_events_dir_configured();
252    let solana_depth = solana_depth_stub(
253        true,
254        events_configured,
255        settlement_on_chain_pending(),
256        false,
257    );
258    Ok((
259        StatusCode::OK,
260        Json(GridPayoutBatchResponse {
261            ok: true,
262            settlement_mode: mode,
263            on_chain_pending: Some(settlement_on_chain_pending()),
264            onchain_depth: settlement_onchain_depth_wire_label(onchain_depth),
265            solana_depth: solana_depth_wire_label(solana_depth),
266            onchain_events_configured: events_configured,
267            routing,
268            entry,
269        }),
270    ))
271}

795fn freshness_to_response(f: CacheFreshness) -> Option<GridPricingSnapshotFreshness> {
796    match f {
797        CacheFreshness::Fresh => Some(GridPricingSnapshotFreshness::Fresh),
798        CacheFreshness::Stale => Some(GridPricingSnapshotFreshness::Stale),
799        CacheFreshness::Expired => None,

803fn cache_hit_response(
804    entry: GalaxyPricingCacheEntry,
805    freshness: CacheFreshness,
806    config: &GalaxyPricingConfig,
807    now_secs: u64,
808) -> GridPricingSnapshotResponse {
809    match freshness {
810        CacheFreshness::Fresh => record_l1_fresh_served(entry.quote.unit_key),
811        CacheFreshness::Stale => record_l1_stale_served(entry.quote.unit_key),
812        CacheFreshness::Expired => {}
813    }
814    let l1_cache = Some(cache_metadata(now_secs, entry.quote.cached_at_secs, config));
815    if let Some(ref meta) = l1_cache {
816        observe_l1_cache_age_secs(meta.cache_age_secs);
817    }
818    GridPricingSnapshotResponse {
819        ok: true,
820        source: GridPricingSnapshotSource::Cache,
821        freshness: freshness_to_response(freshness).expect("fresh or stale"),
822        snapshot: snapshot_from_quote(entry.quote),
823        l1_cache,
824    }
825}

827async fn get_grid_pricing_snapshot(
828    State(_ctx): State<ApiContext>,
829    Query(query): Query<GridPricingSnapshotQuery>,
830) -> Result<(StatusCode, Json<GridPricingSnapshotResponse>), HttpAppError> {
831    let unit_key = GalaxyPriceUnitKey::from_str(&query.unit_key).map_err(|_| {
832        AppError::ValidationError(format!(
833            "invalid `unit_key` '{}'; expected one of: inference_input_token, inference_output_token, inference_blended_token, gpu_second, job_flat",
834            query.unit_key
835        ))
836    })?;
837    let cache_key = GalaxyPricingCacheKey {
838        task_profile: query.task_profile,
839        model_profile: query.model_profile,
840        unit_key,
841    };
842    let now = now_secs()?;
843    let (config, provider_catalog) = {
844        let oracle = pricing_oracle()
845            .lock()
846            .map_err(|_| AppError::InternalError("pricing oracle mutex poisoned".to_string()))?;
847        if let Some((entry, freshness)) = oracle.lookup(now, &cache_key) {
848            let serve_cached = if oracle.config().force_fallback {
849                entry.quote.provider_id_at_min == "fallback_l2_config"
850            } else {
851                true
852            };
853            if serve_cached && freshness_to_response(freshness).is_some() {
854                return Ok((
855                    StatusCode::OK,
856                    Json(cache_hit_response(entry, freshness, oracle.config(), now)),
857                ));
858            }
859        }
860        (*oracle.config(), oracle.provider_catalog().clone())
861    };
862
863    let live_providers = fetch_live_provider_quotes(
864        &provider_catalog,
865        &cache_key.task_profile,
866        &cache_key.model_profile,
867        cache_key.unit_key,
868        provider_http_timeout_ms_from_env(),
869    )
870    .await;
871
872    let mut oracle = pricing_oracle()
873        .lock()
874        .map_err(|_| AppError::InternalError("pricing oracle mutex poisoned".to_string()))?;
875    if let Some((entry, freshness)) = oracle.lookup(now, &cache_key) {
876        let serve_cached = if config.force_fallback {
877            entry.quote.provider_id_at_min == "fallback_l2_config"
878        } else {
879            true
880        };
881        if serve_cached && freshness_to_response(freshness).is_some() {
882            return Ok((
883                StatusCode::OK,
884                Json(cache_hit_response(entry, freshness, &config, now)),
885            ));
886        }
887    }
888
889    let quote = oracle
890        .try_quote(now, cache_key, &live_providers)
891        .map_err(|_| {
892            HttpAppError::new(AppError::RestError {
893            code: PRICING_UNAVAILABLE_ERROR_CODE,
894            message:
895                "grid pricing unavailable: no fresh or stale cache and L2 fallback not configured"
896                    .to_string(),
897        })
898        .with_status(StatusCode::SERVICE_UNAVAILABLE)
899        })?;
900    Ok((
901        StatusCode::OK,
902        Json(GridPricingSnapshotResponse {
903            ok: true,
904            source: GridPricingSnapshotSource::Oracle,
905            freshness: GridPricingSnapshotFreshness::Fresh,
906            snapshot: snapshot_from_quote(quote),
907            l1_cache: None,
908        }),
909    ))
910}
