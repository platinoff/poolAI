17#[derive(Deserialize)]
18struct CreateJobRequest {
19    kind: JobKind,
20    #[serde(default)]
21    priority: u8,
22    #[serde(default)]
23    input_artifact_ids: Vec<String>,
24    #[serde(default)]
25    verification_policy: Option<String>,

28#[derive(Serialize)]
29struct JobSummary {
30    id: String,
31    kind: JobKind,
32    status: JobStatus,
33    created_at: String,
34    #[serde(skip_serializing_if = "Option::is_none")]
35    worker_id: Option<String>,
36    #[serde(skip_serializing_if = "Option::is_none")]
37    vm_id: Option<String>,
38}

40#[derive(Serialize)]
41struct JobsListResponse {
42    jobs: Vec<JobSummary>,
43}

45#[derive(Serialize)]
46struct JobDetailResponse {

48}
49
50#[derive(Serialize)]
51struct ScheduleJobsResponse {
52    scheduled: usize,
53    bound_workers: usize,
54    bound_vms: usize,
55}
56
57#[derive(Deserialize)]
58struct PatchJobRequest {

60}

62fn store() -> &'static JobStore {
63    JobStore::global()
64}

66pub fn create_jobs_routes() -> Router<ApiContext> {
67    Router::new()
68        .route("/jobs", get(list_jobs).post(create_job))
69        .route("/jobs/schedule", post(schedule_jobs))
70        .route("/jobs/{id}", get(get_job).patch(patch_job))
71}

73async fn list_jobs(State(_ctx): State<ApiContext>) -> Result<Json<JobsListResponse>, HttpAppError> {
74    let jobs = store()
75        .list()?
76        .into_iter()
77        .map(|r| JobSummary {
78            id: r.spec.id.0.clone(),
79            kind: r.spec.kind,
80            status: r.status,
81            created_at: r.created_at.to_rfc3339(),
82            worker_id: r.worker_id,
83            vm_id: r.vm_id,
84        })
85        .collect();
86    Ok(Json(JobsListResponse { jobs }))
87}

89async fn create_job(
90    State(ctx): State<ApiContext>,
91    Json(body): Json<CreateJobRequest>,
92) -> Result<(StatusCode, Json<JobSummary>), HttpAppError> {
93    let id = JobId::new(uuid::Uuid::new_v4().to_string());
94    let spec = JobSpec {
95        id: id.clone(),
96        kind: body.kind,
97        resources: Default::default(),
98        priority: body.priority,
99        max_duration_secs: None,
100        input_artifact_ids: body.input_artifact_ids,
101        verification_policy: body.verification_policy,
102        deadline: None,
103    };
104    let record = JobRecord {
105        spec: spec.clone(),
106        status: JobStatus::Submitted,
107        created_at: Utc::now(),
108        worker_id: None,
109        vm_id: None,
110    };
111    store().push(record)?;
112    schedule_from_context(&ctx, store()).await?;
113    let scheduled = store()
114        .get(&id.0)?
115        .ok_or_else(|| AppError::InternalError("job missing after create".into()))?;
116    let summary = JobSummary {
117        id: id.0.clone(),
118        kind: scheduled.spec.kind,
119        status: scheduled.status,
120        created_at: scheduled.created_at.to_rfc3339(),
121        worker_id: scheduled.worker_id,
122        vm_id: scheduled.vm_id,
123    };
124    Ok((StatusCode::CREATED, Json(summary)))
125}

127async fn schedule_jobs(
128    State(ctx): State<ApiContext>,
129) -> Result<Json<ScheduleJobsResponse>, HttpAppError> {
130    let outcome = schedule_from_context(&ctx, store()).await?;
131    Ok(Json(ScheduleJobsResponse {
132        scheduled: outcome.scheduled,
133        bound_workers: outcome.bound_workers,
134        bound_vms: outcome.bound_vms,
135    }))
136}
137
138async fn get_job(
139    State(_ctx): State<ApiContext>,
140    Path(id): Path<String>,
141) -> Result<Json<JobDetailResponse>, HttpAppError> {
142    let job = store()
143        .get(&id)?
144        .ok_or_else(|| HttpAppError::new(AppError::ApiNotFound(format!("job '{id}' not found"))))?;
145    Ok(Json(JobDetailResponse { job }))
146}
147
148async fn patch_job(
149    State(_ctx): State<ApiContext>,
150    Path(id): Path<String>,
151    Json(body): Json<PatchJobRequest>,
152) -> Result<Json<JobDetailResponse>, HttpAppError> {
153    let job = store().update_status(&id, body.status)?;
154    Ok(Json(JobDetailResponse { job }))
