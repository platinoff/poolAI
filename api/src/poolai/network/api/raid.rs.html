33type RaidHttpErr = (StatusCode, AxumJson<serde_json::Value>);

35fn raid_api_err(
36    code: impl AsRef<str>,
37    message: impl Into<String>,
38    ctx: Option<ErrorContext>,
39    status: StatusCode,
40) -> RaidHttpErr {
41    let (s, j) = api_json_error(code, message, ctx, status);
42    (s, AxumJson(j.0))
43}

45fn raid_event_store_unavailable(operation: impl Into<String>) -> RaidHttpErr {
46    raid_api_err(
47        "EVENT_STORE_UNAVAILABLE",
48        "Event store is not initialized or accessible.",
49        Some(ErrorContext::new(operation.into()).with_hint(
50            "Enable event store in configuration and ensure initialization at startup.",
51        )),
52        StatusCode::SERVICE_UNAVAILABLE,
53    )
54}

56fn raid_manager_unavailable() -> RaidHttpErr {
57    raid_api_err(
58        "RAID_MANAGER_UNAVAILABLE",
59        crate::services::raid_service::RAID_MANAGER_UNAVAILABLE_MESSAGE,
60        Some(
61            ErrorContext::new("raid")
62                .with_hint("Ensure RAID manager is initialized during application startup."),
63        ),
64        StatusCode::SERVICE_UNAVAILABLE,
65    )

68fn raid_service_http_err(e: RaidServiceError) -> RaidHttpErr {
69    match e {
70        RaidServiceError::ManagerUnavailable => raid_manager_unavailable(),
71        RaidServiceError::ArtifactNotFound { id } => raid_api_err(
72            "ARTIFACT_NOT_FOUND",
73            format!("Artifact {} not found", id),
74            Some(ErrorContext::new("raid_artifact").with_resource("artifact_id", id.to_string())),
75            StatusCode::NOT_FOUND,
76        ),
77        RaidServiceError::WorkerNotFound { id } => raid_api_err(
78            "RAID_WORKER_NOT_FOUND",
79            format!("RAID worker not found: {}", id),
80            Some(ErrorContext::new("raid_worker_get").with_resource("worker_id", id.to_string())),
81            StatusCode::NOT_FOUND,
82        ),
83        RaidServiceError::EventStoreUnavailable { operation } => {
84            raid_event_store_unavailable(operation)
85        }
86        RaidServiceError::Operation(ref err) => {
87            let (s, j) = api_error_response(
88                err,
89                Some(ErrorContext::new("raid")),
90                Some(StatusCode::INTERNAL_SERVER_ERROR),
91            );
92            (s, AxumJson(j.0))
93        }
94    }
95}

250async fn raid_artifact_create_handler(
251    State(ctx): State<ApiContext>,
252    Json(payload): Json<CreateArtifactRequest>,
253) -> impl IntoResponse {
254    // Validate artifact name
255    if let Err(e) = validation::validate_artifact_name(&payload.name) {
256        let (s, j) = api_json_error(
257            "VALIDATION_ERROR",
258            e.to_string(),
259            Some(ErrorContext::new("raid_artifact_create").with_resource("field", "name")),
260            StatusCode::BAD_REQUEST,
261        );
262        return (s, AxumJson(j.0)).into_response();
263    }
264
265    // Validate base64 data size (max 100MB)
266    const MAX_ARTIFACT_SIZE: usize = 100 * 1024 * 1024; // 100MB
267    if let Err(e) = validation::validate_base64_data(&payload.data, MAX_ARTIFACT_SIZE) {
268        let (s, j) = api_json_error(
269            "VALIDATION_ERROR",
270            e.to_string(),
271            Some(ErrorContext::new("raid_artifact_create").with_resource("field", "data")),
272            StatusCode::BAD_REQUEST,
273        );
274        return (s, AxumJson(j.0)).into_response();
275    }
276
277    // Decode base64 data
278    let artifact_name = payload.name.clone();
279    let data = match base64_engine.decode(&payload.data) {
280        Ok(d) => d,
281        Err(e) => {
282            let (s, j) = api_json_error(
283                "INVALID_BASE64",
284                format!("Cannot decode base64 artifact data: {}", e),
285                Some(
286                    ErrorContext::new("raid_artifact_create")
287                        .with_resource("artifact_name", artifact_name.clone())
288                        .with_hint("Verify payload is valid base64 and not corrupted."),
289                ),
290                StatusCode::BAD_REQUEST,
291            );
292            return (s, AxumJson(j.0)).into_response();
293        }
294    };
295
296    // Validate decoded data size
297    if let Err(e) = validation::validate_artifact_data_size(data.len(), MAX_ARTIFACT_SIZE) {
298        let (s, j) = api_json_error(
299            "VALIDATION_ERROR",
300            e.to_string(),
301            Some(ErrorContext::new("raid_artifact_create").with_resource("field", "decoded_size")),
302            StatusCode::BAD_REQUEST,
303        );
304        return (s, AxumJson(j.0)).into_response();
305    }
306
307    let data_size = data.len();
308    match RaidService::put_artifact(&ctx, &payload.name, &data).await {
309        Ok(artifact) => {
310            let response = CreateArtifactResponse {
311                artifact_id: artifact.id.to_string(),
312                name: artifact.name,
313                stored_at: artifact.stored_at.to_rfc3339(),
314                message: "Artifact created successfully".to_string(),
315            };
316            (StatusCode::CREATED, AxumJson(response)).into_response()
317        }
318        Err(RaidServiceError::ManagerUnavailable) => raid_manager_unavailable().into_response(),
319        Err(RaidServiceError::ArtifactNotFound { .. }) => {
320            let (s, j) = api_json_error(
321                "RAID_UNEXPECTED_STATE",
322                "Unexpected RAID state while creating artifact.",
323                Some(ErrorContext::new("raid_artifact_create")),
324                StatusCode::INTERNAL_SERVER_ERROR,
325            );
326            (s, AxumJson(j.0)).into_response()
327        }
328        Err(RaidServiceError::Operation(e)) => {
329            let (s, j) = api_json_error(
330                "RAID_ARTIFACT_CREATE_FAILED",
331                format!("Failed to store artifact in RAID: {}", e),
332                Some(
333                    ErrorContext::new("raid_artifact_create")
334                        .with_resource("artifact_name", artifact_name.clone())
335                        .with_details(format!("data_size_bytes={}", data_size))
336                        .with_hint("Check storage quota, RAID manager status, and disk space."),
337                ),
338                StatusCode::INTERNAL_SERVER_ERROR,
339            );
340            (s, AxumJson(j.0)).into_response()
341        }
342        Err(e) => raid_service_http_err(e).into_response(),
343    }
344}

346async fn raid_artifact_delete_handler(
347    State(ctx): State<ApiContext>,
348    Path(artifact_id): Path<String>,
349) -> impl IntoResponse {
350    // Validate UUID format
351    if let Err(e) = validation::validate_uuid(&artifact_id) {
352        let (s, j) = api_json_error(
353            "VALIDATION_ERROR",
354            e.to_string(),
355            Some(
356                ErrorContext::new("raid_artifact_delete")
357                    .with_resource("artifact_id", &artifact_id),
358            ),
359            StatusCode::BAD_REQUEST,
360        );
361        return (s, AxumJson(j.0)).into_response();
362    }
363
364    // Parse artifact ID
365    let id = Uuid::parse_str(&artifact_id).unwrap(); // Safe after validation
366
367    match RaidService::delete_artifact(&ctx, id).await {
368        Ok(()) => {
369            let response = DeleteArtifactResponse {
370                artifact_id,
371                message: "Artifact deleted successfully".to_string(),
372            };
373            AxumJson(response).into_response()
374        }
375        Err(RaidServiceError::ManagerUnavailable) => raid_manager_unavailable().into_response(),
376        Err(RaidServiceError::ArtifactNotFound { id }) => {
377            let (s, j) = api_json_error(
378                "ARTIFACT_NOT_FOUND",
379                format!("Artifact {} not found", id),
380                Some(
381                    ErrorContext::new("raid_artifact_delete")
382                        .with_resource("artifact_id", id.to_string()),
383                ),
384                StatusCode::NOT_FOUND,
385            );
386            (s, AxumJson(j.0)).into_response()
387        }
388        Err(RaidServiceError::Operation(e)) => {
389            let (s, j) = api_json_error(
390                "RAID_ARTIFACT_DELETE_FAILED",
391                format!("Failed to delete artifact: {}", e),
392                Some(
393                    ErrorContext::new("raid_artifact_delete")
394                        .with_resource("artifact_id", artifact_id.clone()),
395                ),
396                StatusCode::INTERNAL_SERVER_ERROR,
397            );
398            (s, AxumJson(j.0)).into_response()
399        }
400        Err(e) => raid_service_http_err(e).into_response(),
401    }
402}

465async fn raid_events_range_handler(
466    State(ctx): State<ApiContext>,
467    Query(params): Query<EventsRangeQuery>,
468) -> impl IntoResponse {
469    let start = params
470        .start
471        .and_then(|s| DateTime::parse_from_rfc3339(&s).ok())
472        .map(|dt| dt.with_timezone(&chrono::Utc))
473        .unwrap_or_else(|| chrono::Utc::now() - chrono::Duration::days(7));
474
475    let end = params
476        .end
477        .and_then(|s| DateTime::parse_from_rfc3339(&s).ok())
478        .map(|dt| dt.with_timezone(&chrono::Utc))
479        .unwrap_or_else(chrono::Utc::now);
480
481    match RaidService::load_events_in_range(&ctx, start, end).await {
482        Ok(events) => AxumJson(serde_json::json!({
483            "start": start.to_rfc3339(),
484            "end": end.to_rfc3339(),
485            "events": events,
486            "count": events.len()
487        }))
488        .into_response(),
489        Err(RaidServiceError::Operation(ref err)) => {
490            let (s, j) = api_json_error(
491                "RAID_EVENTS_LOAD_FAILED",
492                format!("Failed to load events: {}", err),
493                Some(ErrorContext::new("raid_events_range")),
494                StatusCode::INTERNAL_SERVER_ERROR,
495            );
496            (s, AxumJson(j.0)).into_response()
497        }
498        Err(e) => raid_service_http_err(e).into_response(),
499    }
500}

649async fn raid_worker_get_handler(
650    State(ctx): State<ApiContext>,

861/// Get metrics for the active RAID strategy
862async fn raid_metrics_handler(State(ctx): State<ApiContext>) -> impl IntoResponse {
863    let manager = match raid_http_manager(&ctx) {
864        Ok(m) => m,
865        Err(e) => return e.into_response(),
866    };
867    let mode = manager.get_mode().await;
868    let mode_str = format!("{:?}", mode);
869
870    let artifacts = manager.list_artifacts().await;
871    let total_artifacts = artifacts.len();
872    let total_size = manager.get_total_size().await.unwrap_or(0);
873    let quota_bytes = manager.get_quota_bytes().await;
874    let usage_percent = quota_bytes.map(|quota| {
875        if quota > 0 {
876            (total_size as f64 / quota as f64) * 100.0
877        } else {
878            0.0
879        }
880    });
881    let nodes = manager.list_nodes().await;
882    let node_count = nodes.len();
883
884    // Strategy-specific metrics
885    let replication_factor = match mode {
886        raid::RaidMode::BurstRaid => {
887            // Get actual replication factor from BurstRAID strategy
888            if let Some(metrics) = manager.get_burst_raid_metrics().await {
889                Some(metrics.base_replication_factor)
890            } else {
891                Some(2) // Default base_replication_factor
892            }
893        }
894        raid::RaidMode::SmallWorld => {
895            // Get actual replication factor from SmallWorld strategy
896            if let Some(metrics) = manager.get_small_world_metrics().await {
897                Some(metrics.base_replication_factor)
898            } else {
899                Some(3) // Default base_replication_factor
900            }
901        }
902        _ => None,
903    };
904
905    let clustering_coefficient = match mode {
906        raid::RaidMode::SmallWorld => {
907            // Get actual clustering coefficient from SmallWorld strategy
908            if let Some(metrics) = manager.get_small_world_metrics().await {
909                Some(metrics.avg_clustering_coefficient)
910            } else {
911                None
912            }
913        }
914        _ => None,
915    };
916
917    let response = RaidMetricsResponse {
918        mode: mode_str,
919        total_artifacts,
920        total_size_bytes: total_size,
921        quota_bytes,
922        usage_percent,
923        node_count,
924        replication_factor,
925        clustering_coefficient,
926    };
927
928    AxumJson(response).into_response()
929}

973/// Health check for RAID strategies
974async fn raid_health_handler(State(ctx): State<ApiContext>) -> impl IntoResponse {
975    let manager = match raid_http_manager(&ctx) {
976        Ok(m) => m,
977        Err(e) => return e.into_response(),
978    };
979    let mode = manager.get_mode().await;
980    let mode_str = format!("{:?}", mode);
981
982    // Get strategy status
983    let strategy_status = manager.get_strategy_status().await.ok();
984    let strategy_initialized = strategy_status
985        .as_ref()
986        .map(|s| s.initialized)
987        .unwrap_or(false);
988
989    // Check storage availability
990    let total_size = manager.get_total_size().await.unwrap_or(0);
991    let quota_bytes = manager.get_quota_bytes().await;
992    let storage_available = quota_bytes
993        .map(|quota| {
994            let usage_percent = if quota > 0 {
995                (total_size as f64 / quota as f64) * 100.0
996            } else {
997                0.0
998            };
999            usage_percent < 95.0 // Consider available if usage < 95%
1000        })
1001        .unwrap_or(true);
1002
1003    // Check replication status
1004    let replication_active = match mode {
1005        raid::RaidMode::Local => false,
1006        raid::RaidMode::BurstRaid | raid::RaidMode::SmallWorld => strategy_status
1007            .as_ref()
1008            .map(|s| s.active && s.rebalancing_enabled)
1009            .unwrap_or(false),
1010    };
1011
1012    // Determine overall health status
1013    let status = if !strategy_initialized && mode != raid::RaidMode::Local {
1014        "unhealthy"
1015    } else if !storage_available {
1016        "degraded"
1017    } else if !replication_active && mode != raid::RaidMode::Local {
1018        "degraded"
1019    } else {
1020        "healthy"
1021    };
1022
1023    let response = RaidHealthResponse {
1024        status: status.to_string(),
1025        mode: mode_str,
1026        strategy_initialized,
1027        storage_available,
1028        replication_active,
1029    };
1030
1031    AxumJson(response).into_response()
1032}
