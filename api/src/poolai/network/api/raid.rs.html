195async fn raid_artifact_create_handler(
196    Json(payload): Json<CreateArtifactRequest>,
197) -> impl IntoResponse {
198    // Validate artifact name
199    if let Err(e) = validation::validate_artifact_name(&payload.name) {
200        return (
201            StatusCode::BAD_REQUEST,
202            AxumJson(serde_json::json!({
203                "error": e.to_string()
204            })),
205        )
206            .into_response();
207    }
208
209    // Validate base64 data size (max 100MB)
210    const MAX_ARTIFACT_SIZE: usize = 100 * 1024 * 1024; // 100MB
211    if let Err(e) = validation::validate_base64_data(&payload.data, MAX_ARTIFACT_SIZE) {
212        return (
213            StatusCode::BAD_REQUEST,
214            AxumJson(serde_json::json!({
215                "error": e.to_string()
216            })),
217        )
218            .into_response();
219    }
220
221    // Decode base64 data
222    let artifact_name = payload.name.clone();
223    let data = match base64_engine.decode(&payload.data) {
224        Ok(d) => d,
225        Err(e) => {
226            return (
227                StatusCode::BAD_REQUEST,
228                AxumJson(serde_json::json!({
229                    "error": format!("Invalid base64 data. Context: Cannot decode base64-encoded artifact data. Suggestion: Verify data is properly base64-encoded and not corrupted. Artifact name: '{}', Error: {}", artifact_name, e)
230                })),
231            )
232                .into_response();
233        }
234    };
235
236    // Validate decoded data size
237    if let Err(e) = validation::validate_artifact_data_size(data.len(), MAX_ARTIFACT_SIZE) {
238        return (
239            StatusCode::BAD_REQUEST,
240            AxumJson(serde_json::json!({
241                "error": e.to_string()
242            })),
243        )
244            .into_response();
245    }
246
247    // Create artifact
248    let manager = raid::get_global_manager();
249    let data_size = data.len();
250    match manager.put_artifact(&payload.name, &data).await {
251        Ok(artifact) => {
252            let response = CreateArtifactResponse {
253                artifact_id: artifact.id.to_string(),
254                name: artifact.name,
255                stored_at: artifact.stored_at.to_rfc3339(),
256                message: "Artifact created successfully".to_string(),
257            };
258            (StatusCode::CREATED, AxumJson(response)).into_response()
259        }
260        Err(e) => (
261            StatusCode::INTERNAL_SERVER_ERROR,
262            AxumJson(serde_json::json!({
263                "error": format!("Failed to create artifact. Context: Cannot store artifact in RAID storage. Suggestion: Check storage quota, verify RAID manager is initialized, and ensure sufficient disk space. Artifact name: '{}', Data size: {} bytes, Error: {}", artifact_name, data_size, e)
264            })),
265        )
266            .into_response(),
267    }
268}

270async fn raid_artifact_delete_handler(Path(artifact_id): Path<String>) -> impl IntoResponse {
271    // Validate UUID format
272    if let Err(e) = validation::validate_uuid(&artifact_id) {
273        return (
274            StatusCode::BAD_REQUEST,
275            AxumJson(serde_json::json!({
276                "error": e.to_string()
277            })),
278        )
279            .into_response();
280    }
281
282    // Parse artifact ID
283    let id = Uuid::parse_str(&artifact_id).unwrap(); // Safe after validation
284
285    // Delete artifact
286    let manager = raid::get_global_manager();
287    match manager.delete_artifact(id).await {
288        Ok(_) => {
289            let response = DeleteArtifactResponse {
290                artifact_id,
291                message: "Artifact deleted successfully".to_string(),
292            };
293            AxumJson(response).into_response()
294        }
295        Err(e) => (
296            StatusCode::NOT_FOUND,
297            AxumJson(serde_json::json!({
298                "error": format!("Failed to delete artifact. Context: Cannot delete artifact from RAID storage. Suggestion: Verify artifact ID exists, check RAID manager status, and ensure artifact is not locked. Artifact ID: '{}', Error: {}", artifact_id, e)
299            })),
300        )
301            .into_response(),
302    }
303}

305async fn raid_quota_handler() -> impl IntoResponse {
306    let manager = raid::get_global_manager();
307    let total_size = manager.get_total_size().await.unwrap_or(0);
308    let artifacts = manager.list_artifacts().await;
309    let artifact_count = artifacts.len();
310
311    // Get quota from config
312    let quota_bytes = manager.get_quota_bytes().await;
313
314    let usage_percent = quota_bytes.map(|quota| {
315        if quota > 0 {
316            (total_size as f64 / quota as f64) * 100.0
317        } else {
318            0.0
319        }
320    });
321
322    AxumJson(RaidQuotaResponse {
323        total_size_bytes: total_size,
324        quota_bytes,
325        usage_percent,
326        artifact_count,
327    })
328    .into_response()
329}

331async fn raid_status_handler() -> impl IntoResponse {
332    let manager = raid::get_global_manager();
333
334    // Get basic information
335    let nodes = manager.list_nodes().await;
336    let artifacts = manager.list_artifacts().await;
337    let node_count = nodes.len();
338    let artifact_count = artifacts.len();
339
340    // Get storage information
341    let total_size = manager.get_total_size().await.unwrap_or(0);
342    let quota_bytes = manager.get_quota_bytes().await;
343    let usage_percent = quota_bytes.map(|quota| {
344        if quota > 0 {
345            (total_size as f64 / quota as f64) * 100.0
346        } else {
347            0.0
348        }
349    });
350    let available_bytes = quota_bytes.map(|quota| quota.saturating_sub(total_size));
351
352    // Get mode from config
353    let mode = format!("{:?}", manager.get_mode().await);
354
355    // Determine cluster status
356    // Healthy: nodes > 0, usage < 90%, no errors
357    // Degraded: usage >= 90%, or some nodes unavailable
358    // Unhealthy: no nodes, or critical errors
359    let cluster_status = if node_count == 0 {
360        "unhealthy".to_string()
361    } else if let Some(usage) = usage_percent {
362        if usage >= 95.0 {
363            "unhealthy".to_string()
364        } else if usage >= 90.0 {
365            "degraded".to_string()
366        } else {
367            "healthy".to_string()
368        }
369    } else {
370        "healthy".to_string()
371    };
372
373    // Check replication status (if in distributed mode)
374    let replication_status = if mode != "Local" {
375        // For distributed modes, check if replication is active
376        // Placeholder: in future, check actual replication status
377        Some("active".to_string())
378    } else {
379        None
380    };
381
382    // Check Raft status (if enabled)
383    // TODO: Implement Raft status query when Raft integration is complete
384    // Placeholder: Raft status would be queried from the global Raft node
385    #[cfg(feature = "raft")]
386    let raft_status: Option<RaftStatus> = {
387        // Future implementation:
388        // - Query Raft node for current role (leader/follower/candidate)
389        // - Get current term from Raft state
390        // - Get leader ID from Raft state
391        None
392    };
393
394    #[cfg(not(feature = "raft"))]
395    let raft_status: Option<RaftStatus> = None;
396
397    let response = RaidStatusResponse {
398        cluster_status,
399        node_count,
400        artifact_count,
401        storage: RaidStorageStatus {
402            total_size_bytes: total_size,
403            quota_bytes,
404            usage_percent,
405            available_bytes,
406        },
407        mode,
408        replication_status,
409        raft_status,
410    };
411
412    AxumJson(response).into_response()
413}

479async fn raid_events_range_handler(Query(params): Query<EventsRangeQuery>) -> impl IntoResponse {
480    let manager = raid::get_global_manager();
481
482    if let Some(event_store) = manager.event_store() {
483        let start = params
484            .start
485            .and_then(|s| DateTime::parse_from_rfc3339(&s).ok())
486            .map(|dt| dt.with_timezone(&chrono::Utc))
487            .unwrap_or_else(|| chrono::Utc::now() - chrono::Duration::days(7));
488
489        let end = params
490            .end
491            .and_then(|s| DateTime::parse_from_rfc3339(&s).ok())
492            .map(|dt| dt.with_timezone(&chrono::Utc))
493            .unwrap_or_else(chrono::Utc::now);
494
495        match event_store
496            .read()
497            .await
498            .get_events_in_range(start, end)
499            .await
500        {
501            Ok(events) => AxumJson(serde_json::json!({
502                "start": start.to_rfc3339(),
503                "end": end.to_rfc3339(),
504                "events": events,
505                "count": events.len()
506            }))
507            .into_response(),
508            Err(e) => (
509                StatusCode::INTERNAL_SERVER_ERROR,
510                AxumJson(serde_json::json!({
511                    "error": format!("Failed to load events. Context: Cannot retrieve events from event store. Suggestion: Verify event store is accessible, check file permissions, and ensure event store is properly initialized. Error: {}", e)
512                })),
513            )
514                .into_response(),
515        }
516    } else {
517        (
518            StatusCode::SERVICE_UNAVAILABLE,
519            AxumJson(serde_json::json!({
520                "error": "Event store not available. Context: Event store is not initialized or accessible. Suggestion: Ensure event store is enabled in configuration and properly initialized during system startup."
521            })),
522        )
523            .into_response()
524    }

663async fn raid_worker_get_handler(Path(id): Path<String>) -> impl IntoResponse {
664    let manager = raid::get_global_manager();
665    let uuid = match Uuid::parse_str(&id) {
666        Ok(u) => u,
667        Err(_) => {
668            return (
669                StatusCode::BAD_REQUEST,
670                AxumJson(serde_json::json!({
671                    "error": format!("Invalid UUID format: {}", id)
672                })),
673            )
674                .into_response();
675        }
676    };
677
678    match manager.get_node(uuid).await {
679        Some(node) => {
680            let response = RaidWorkerResponse {
681                id: node.id.to_string(),
682                address: node.address,
683                last_seen: node.last_seen.to_rfc3339(),
684            };
685            AxumJson(response).into_response()
686        }
687        None => (
688            StatusCode::NOT_FOUND,

874/// Get metrics for the active RAID strategy
875async fn raid_metrics_handler() -> impl IntoResponse {
876    let manager = raid::get_global_manager();
877    let mode = manager.get_mode().await;
878    let mode_str = format!("{:?}", mode);
879
880    let artifacts = manager.list_artifacts().await;
881    let total_artifacts = artifacts.len();
882    let total_size = manager.get_total_size().await.unwrap_or(0);
883    let quota_bytes = manager.get_quota_bytes().await;
884    let usage_percent = quota_bytes.map(|quota| {
885        if quota > 0 {
886            (total_size as f64 / quota as f64) * 100.0
887        } else {
888            0.0
889        }
890    });
891    let nodes = manager.list_nodes().await;
892    let node_count = nodes.len();
893
894    // Strategy-specific metrics
895    let replication_factor = match mode {
896        raid::RaidMode::BurstRaid => {
897            // Get actual replication factor from BurstRAID strategy
898            if let Some(metrics) = manager.get_burst_raid_metrics().await {
899                Some(metrics.base_replication_factor)
900            } else {
901                Some(2) // Default base_replication_factor
902            }
903        }
904        raid::RaidMode::SmallWorld => {
905            // Get actual replication factor from SmallWorld strategy
906            if let Some(metrics) = manager.get_small_world_metrics().await {
907                Some(metrics.base_replication_factor)
908            } else {
909                Some(3) // Default base_replication_factor
910            }
911        }
912        _ => None,
913    };
914
915    let clustering_coefficient = match mode {
916        raid::RaidMode::SmallWorld => {
917            // Get actual clustering coefficient from SmallWorld strategy
918            if let Some(metrics) = manager.get_small_world_metrics().await {
919                Some(metrics.avg_clustering_coefficient)
920            } else {
921                None
922            }
923        }
924        _ => None,
925    };
926
927    let response = RaidMetricsResponse {
928        mode: mode_str,
929        total_artifacts,
930        total_size_bytes: total_size,
931        quota_bytes,
932        usage_percent,
933        node_count,
934        replication_factor,
935        clustering_coefficient,
936    };
937
938    AxumJson(response).into_response()
939}

973/// Health check for RAID strategies
974async fn raid_health_handler() -> impl IntoResponse {
975    let manager = raid::get_global_manager();
976    let mode = manager.get_mode().await;
977    let mode_str = format!("{:?}", mode);
978
979    // Get strategy status
980    let strategy_status = manager.get_strategy_status().await.ok();
981    let strategy_initialized = strategy_status
982        .as_ref()
983        .map(|s| s.initialized)
984        .unwrap_or(false);
985
986    // Check storage availability
987    let total_size = manager.get_total_size().await.unwrap_or(0);
988    let quota_bytes = manager.get_quota_bytes().await;
989    let storage_available = quota_bytes
990        .map(|quota| {
991            let usage_percent = if quota > 0 {
992                (total_size as f64 / quota as f64) * 100.0
993            } else {
994                0.0
995            };
996            usage_percent < 95.0 // Consider available if usage < 95%
997        })
998        .unwrap_or(true);
999
1000    // Check replication status
1001    let replication_active = match mode {
1002        raid::RaidMode::Local => false,
1003        raid::RaidMode::BurstRaid | raid::RaidMode::SmallWorld => strategy_status
1004            .as_ref()
1005            .map(|s| s.active && s.rebalancing_enabled)
1006            .unwrap_or(false),
1007    };
1008
1009    // Determine overall health status
1010    let status = if !strategy_initialized && mode != raid::RaidMode::Local {
1011        "unhealthy"
1012    } else if !storage_available {
1013        "degraded"
1014    } else if !replication_active && mode != raid::RaidMode::Local {
1015        "degraded"
1016    } else {
1017        "healthy"
1018    };
1019
1020    let response = RaidHealthResponse {
1021        status: status.to_string(),
1022        mode: mode_str,
1023        strategy_initialized,
1024        storage_available,
1025        replication_active,
1026    };
1027
1028    AxumJson(response).into_response()
1029}
