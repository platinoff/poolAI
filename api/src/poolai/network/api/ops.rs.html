1//! Operator power endpoints (`POST /api/v1/ops/power`, PH-S1016) and ratio96 store wire
2//! (`GET /api/v1/ops/ratio96`, PH-S1680).
3
4use axum::{
5    http::StatusCode,
6    routing::{get, post},
7    Json, Router,
8};
9
10use crate::core::error::AppError;
11use crate::core::state::ApiContext;
12use crate::network::api::common::HttpAppError;
13use crate::ops::power::{apply_power_action, PowerAction, PowerRequest, PowerResponse};
14
15pub fn create_ops_routes() -> Router<ApiContext> {
16    Router::new()
17        .route("/ops/power", post(power_handler))
18        .route("/ops/ratio96", get(ratio96_store_wire_handler))
19}
20
21async fn power_handler(
22    Json(body): Json<PowerRequest>,
23) -> Result<(StatusCode, Json<PowerResponse>), HttpAppError> {
24    let action = PowerAction::parse(body.action.trim()).ok_or_else(|| {
25        HttpAppError::new(AppError::ValidationError(
26            "action must be shutdown or reboot".into(),
27        ))
28    })?;
29    let response = apply_power_action(action);
30    Ok((StatusCode::ACCEPTED, Json(response)))
31}
32
33/// `GET /api/v1/ops/ratio96` — durable ratio store wire for the admin dashboard strip (PH-S1680).
34async fn ratio96_store_wire_handler() -> Json<serde_json::Value> {
35    Json(poolai_ui_core::ratio96_store_depth::ratio96_store_wire_json())
36}

38#[cfg(test)]
39mod tests {
40    use super::*;
41    use axum::body::Body;
42    use axum::http::Request;
43    use tower::ServiceExt;
44
45    #[tokio::test]
46    async fn ops_power_route_accepts_shutdown_ph_s1016() {
47        let app = create_ops_routes().with_state(ApiContext::default());
48        let response = app
49            .oneshot(
50                Request::builder()
51                    .method("POST")
52                    .uri("/ops/power")
53                    .header("content-type", "application/json")
54                    .body(Body::from(r#"{"action":"shutdown"}"#))
55                    .unwrap(),
56            )
57            .await
58            .unwrap();
59        assert_eq!(response.status(), StatusCode::ACCEPTED);
60    }
61
62    #[tokio::test]
63    async fn ops_ratio96_route_returns_wire_ph_s1680() {
64        let app = create_ops_routes().with_state(ApiContext::default());
65        let response = app
66            .oneshot(
67                Request::builder()
68                    .method("GET")
69                    .uri("/ops/ratio96")
70                    .body(Body::empty())
71                    .unwrap(),
72            )
73            .await
74            .unwrap();
75        assert_eq!(response.status(), StatusCode::OK);
76        let body = axum::body::to_bytes(response.into_body(), 64 * 1024)
77            .await
78            .expect("ratio96 wire body");
79        let json: serde_json::Value =
80            serde_json::from_slice(&body).expect("ratio96 wire parses as JSON");
81        assert_eq!(
82            json.get("available").and_then(serde_json::Value::as_bool),
83            Some(true)
84        );
85        assert!(json.get("stretch_gate_met").is_some());
86        assert!(json.get("hold_gate_met").is_some());
87    }
