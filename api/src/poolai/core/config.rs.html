236    pub fn from_file(path: &str) -> Result<Self, AppError> {
237        let content = std::fs::read_to_string(path)
238            .map_err(|e| AppError::ConfigError(format!("Failed to read config file: {}", e)))?;
239
240        let config: PoolAIConfig = toml::from_str(&content)
241            .map_err(|e| AppError::ConfigError(format!("Failed to parse config: {}", e)))?;
242
243        Ok(config)
244    }

248        let content = toml::to_string_pretty(self)
249            .map_err(|e| AppError::ConfigError(format!("Failed to serialize config: {}", e)))?;
250
251        std::fs::write(path, content)
252            .map_err(|e| AppError::ConfigError(format!("Failed to write config file: {}", e)))?;
253
254        Ok(())
255    }

272    /// Validate configuration
273    pub fn validate(&self) -> Result<(), AppError> {
274        // Validate system configuration
275        if self.system.max_workers == 0 {
276            return Err(AppError::ConfigError(
277                format!(
278                    "Invalid configuration: system.max_workers is 0 (must be > 0). \
279                    Context: max_workers defines the maximum number of concurrent worker processes. \
280                    Suggestion: Set max_workers to a positive value (e.g., 4-16 depending on CPU cores). \
281                    Current value: {}",
282                    self.system.max_workers
283                )
284            ));
285        }
286
287        if self.system.queue_size == 0 {
288            return Err(AppError::ConfigError(
289                format!(
290                    "Invalid configuration: system.queue_size is 0 (must be > 0). \
291                    Context: queue_size defines the maximum number of pending requests in the queue. \
292                    Suggestion: Set queue_size to a positive value (e.g., 100-1000 depending on workload). \
293                    Current value: {}",
294                    self.system.queue_size
295                )
296            ));
297        }
298
299        // Validate GPU configuration
300        if self.gpu.enabled && self.gpu.memory_limit == 0 {
301            return Err(AppError::ConfigError(
302                format!(
303                    "Invalid configuration: GPU is enabled but memory_limit is 0 (must be > 0). \
304                    Context: When GPU is enabled, memory_limit defines the maximum GPU memory in MB. \
305                    Suggestion: Set gpu.memory_limit to a positive value (e.g., 4096 for 4GB) or disable GPU if not needed. \
306                    Current value: {}",
307                    self.gpu.memory_limit
308                )
309            ));
310        }
311
312        // Validate pool configuration
313        if self.pool.max_workers == 0 {
314            return Err(AppError::ConfigError(
315                format!(
316                    "Invalid configuration: pool.max_workers is 0 (must be > 0). \
317                    Context: pool.max_workers defines the maximum number of workers in the pool. \
318                    Suggestion: Set pool.max_workers to a positive value (e.g., 8-32 depending on system capacity). \
319                    Current value: {}",
320                    self.pool.max_workers
321                )
322            ));
323        }
324
325        if self.pool.scaling_threshold < 0.0 || self.pool.scaling_threshold > 1.0 {
326            return Err(AppError::ConfigError(
327                format!(
328                    "Invalid configuration: pool.scaling_threshold is {} (must be between 0.0 and 1.0). \
329                    Context: scaling_threshold defines the load threshold for auto-scaling (0.0 = 0%, 1.0 = 100%). \
330                    Suggestion: Set scaling_threshold to a value between 0.0 and 1.0 (e.g., 0.7 for 70% load threshold). \
331                    Current value: {}",
332                    self.pool.scaling_threshold, self.pool.scaling_threshold
333                )
334            ));
335        }
336
337        // Validate monitoring configuration
338        if self.monitoring.metrics_interval == 0 {
339            return Err(AppError::ConfigError(
340                format!(
341                    "Invalid configuration: monitoring.metrics_interval is 0 (must be > 0). \
342                    Context: metrics_interval defines the interval in seconds between metric collection. \
343                    Suggestion: Set metrics_interval to a positive value (e.g., 5-60 seconds depending on monitoring needs). \
344                    Current value: {}",
345                    self.monitoring.metrics_interval
346                )
347            ));
348        }
349
350        if self.monitoring.alert_threshold < 0.0 || self.monitoring.alert_threshold > 1.0 {
351            return Err(AppError::ConfigError(
352                format!(
353                    "Invalid configuration: monitoring.alert_threshold is {} (must be between 0.0 and 1.0). \
354                    Context: alert_threshold defines the threshold for triggering alerts (0.0 = 0%, 1.0 = 100%). \
355                    Suggestion: Set alert_threshold to a value between 0.0 and 1.0 (e.g., 0.9 for 90% threshold). \
356                    Current value: {}",
357                    self.monitoring.alert_threshold, self.monitoring.alert_threshold
358                )
359            ));
360        }
361
362        Ok(())
363    }

366/// Global configuration instance
367static CONFIG: OnceLock<PoolAIConfig> = OnceLock::new();
368
369/// Initialize configuration
370///
371/// This function initializes the global configuration instance.
372/// Must be called before using any configuration-dependent functionality.
373///
374/// # Arguments
375///
376/// * `config` - The configuration to initialize
377///
378/// # Errors
379///
380/// Returns an error if:
381/// - Configuration validation fails
382/// - Configuration is already initialized
383///
384/// # Example
385///
386/// ```rust,no_run
387/// use poolai::core::config::{PoolAIConfig, initialize_config};
388///
389/// # fn example() -> Result<(), poolai::core::error::AppError> {
390/// let config = PoolAIConfig::default();
391/// initialize_config(config)?;
392/// // Now you can use get_config() to retrieve the configuration
393/// # Ok(())
394/// # }
395/// ```
396pub fn initialize_config(config: PoolAIConfig) -> Result<(), AppError> {
397    config.validate()?;
398
399    CONFIG
400        .set(config)
401        .map_err(|_| AppError::ConfigError("Configuration already initialized".to_string()))?;
402
403    Ok(())
404}
405
406/// Get configuration
407///
408/// Retrieves a clone of the global configuration instance.
409/// Must be called after `initialize_config()`.
410///
411/// # Errors
412///
413/// Returns an error if configuration has not been initialized.
414///
415/// # Example
416///
417/// ```rust,no_run
418/// use poolai::core::config::{get_config, initialize_config, PoolAIConfig};
419///
420/// # fn example() -> Result<(), poolai::core::error::AppError> {
421/// // First initialize
422/// initialize_config(PoolAIConfig::default())?;
423///
424/// // Then retrieve
425/// let config = get_config()?;
426/// println!("System name: {}", config.system.name);
427/// # Ok(())
428/// # }
429/// ```
430pub fn get_config() -> Result<PoolAIConfig, AppError> {
431    CONFIG
432        .get()
433        .cloned()
434        .ok_or_else(|| AppError::ConfigError("Configuration not initialized".to_string()))
435}

437/// Update configuration
438pub fn update_config(config: PoolAIConfig) -> Result<(), AppError> {
439    config.validate()?;
440
441    // OnceLock doesn't support updates, so we need to reinitialize
442    // This is a limitation, but ensures thread safety
443    // For true updates, consider using Arc<RwLock<PoolAIConfig>> instead
444    CONFIG.set(config).map_err(|_| {
445        AppError::ConfigError(
446            "Configuration already initialized. Use reinitialize_config() to update.".to_string(),
447        )
448    })?;
449
450    Ok(())
451}
