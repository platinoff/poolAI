1//! Dev-stand helpers (FM-003): env overrides for multi-node on one host.
2
3/// HTTP API port (`POOLAI_HTTP_PORT`, default `8080`).
4pub fn resolve_http_port() -> u16 {
5    std::env::var("POOLAI_HTTP_PORT")
6        .ok()
7        .and_then(|s| s.parse().ok())
8        .unwrap_or(8080)
9}
10
11#[cfg(test)]
12mod tests {
13    use super::*;
14
15    #[test]
16    fn resolve_http_port_env() {
17        const KEY: &str = "POOLAI_HTTP_PORT";
18        std::env::remove_var(KEY);
19        assert_eq!(resolve_http_port(), 8080);
20        std::env::set_var(KEY, "9091");
21        assert_eq!(resolve_http_port(), 9091);
22        std::env::remove_var(KEY);
23    }
