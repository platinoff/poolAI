1//! E2E stand lifecycle (PH-S158): start / restart / stop poolai for Playwright + stand smoke.
2//!
3//! ```text
4//! cargo run --bin poolai-e2e-stand -- start --port 8080 --print-stand-root
5//! cargo run --bin poolai-e2e-stand -- restart --stand-root /tmp/poolai-e2e-NNN
6//! cargo run --bin poolai-e2e-stand -- stop --stand-root /tmp/poolai-e2e-NNN
7//! ```

60fn default_port() -> u16 {
61    std::env::var(ENV_PORT)
62        .ok()
63        .and_then(|v| v.parse().ok())
64        .unwrap_or(8080)
65}
66
67fn default_profile() -> String {
68    let from_env = std::env::var(ENV_PROFILE).ok();
69    if let Some(p) = from_env.filter(|s| !s.is_empty()) {
70        return p;
71    }
72    if std::env::var("CI").ok().as_deref() == Some("true") {
73        "debug".to_string()
74    } else {
75        "release".to_string()
76    }
77}

79fn parse_cli() -> Result<Cli, String> {
80    let mut command = None;
81    let mut stand_root = None;
82    let mut port = default_port();
83    let mut profile = default_profile();
84    let mut print_stand_root = false;
85    let mut json_out = false;
86    let mut health_tries = 90;
87
88    let args: Vec<String> = std::env::args().skip(1).collect();
89    let mut i = 0;
90    while i < args.len() {
91        match args[i].as_str() {
92            "start" => command = Some(CommandKind::Start),
93            "restart" => command = Some(CommandKind::Restart),
94            "stop" => command = Some(CommandKind::Stop),
95            "--stand-root" => {
96                i += 1;
97                let val = args.get(i).ok_or("--stand-root requires path")?;
98                stand_root = Some(PathBuf::from(val));
99            }
100            "--port" => {
101                i += 1;
102                port = args
103                    .get(i)
104                    .and_then(|v| v.parse().ok())
105                    .ok_or("--port requires u16")?;
106            }
107            "--profile" => {
108                i += 1;
109                profile = args.get(i).ok_or("--profile requires value")?.clone();
110            }
111            "--print-stand-root" => print_stand_root = true,
112            "--json" => json_out = true,
113            "--health-tries" => {
114                i += 1;
115                health_tries = args
116                    .get(i)
117                    .and_then(|v| v.parse().ok())
118                    .ok_or("--health-tries requires u32")?;
119            }
120            _ if args[i].starts_with('-') => {}
121            _ => {}
122        }
123        i += 1;
124    }
125
126    let command = command.unwrap_or(CommandKind::Start);
127    if stand_root.is_none() {
128        stand_root = std::env::var(ENV_STAND_ROOT)
129            .ok()
130            .filter(|s| !s.is_empty())
131            .map(PathBuf::from);
132    }
133
134    Ok(Cli {
135        command,
136        stand_root,
137        port,
138        profile,
139        print_stand_root,
140        json_out,
141        health_tries,
142    })
143}

145fn repo_root() -> PathBuf {
146    std::env::var("POOLAI_REPO_ROOT")
147        .map(PathBuf::from)
148        .unwrap_or_else(|_| PathBuf::from(env!("CARGO_MANIFEST_DIR")))
149}

151fn resolve_poolai_bin(profile: &str) -> Result<PathBuf, String> {
152    let root = repo_root();
153    let candidates = [
154        root.join(format!("target/{profile}/poolai")),
155        root.join(format!("target/{profile}/poolai.exe")),
156    ];
157    for path in candidates {
158        if path.is_file() {
159            return Ok(path);
160        }
161    }
162    let features = std::env::var(ENV_FEATURES).unwrap_or_else(|_| {
163        if std::env::var("CI").ok().as_deref() == Some("true") {
164            "enterprise,cloud,test-utils".to_string()
165        } else {
166            "enterprise,ml,cloud,test-utils".to_string()
167        }
168    });
169    Err(format!(
170    "poolai binary not found for profile `{profile}`; run: cargo build --{profile} --features {features}"
171  ))
172}

174fn resolve_stand_bin() -> PathBuf {
175    std::env::current_exe().unwrap_or_else(|_| PathBuf::from("poolai-e2e-stand"))

178fn default_stand_root() -> PathBuf {
179    let pid = std::process::id();
180    PathBuf::from(format!("/tmp/poolai-e2e-{pid}"))

183fn base_url(port: u16) -> String {
184    std::env::var(ENV_BASE)
185        .ok()
186        .filter(|s| !s.trim().is_empty())
187        .unwrap_or_else(|| format!("http://127.0.0.1:{port}"))

190fn read_pid(stand_root: &Path) -> Option<u32> {
191    let raw = fs::read_to_string(stand_root.join("pid")).ok()?;
192    raw.trim().parse().ok()
193}
194
195fn write_pid(stand_root: &Path, pid: u32) -> Result<(), String> {
196    fs::write(stand_root.join("pid"), format!("{pid}\n")).map_err(|e| e.to_string())?;
197    Ok(())
198}
199
200fn kill_pid(pid: u32) -> Result<(), String> {
201    let status = Command::new("kill")
202        .arg(pid.to_string())
203        .status()
204        .map_err(|e| format!("kill failed: {e}"))?;
205    if !status.success() {
206        return Err(format!("kill pid {pid} exit {status}"));
207    }
208    Ok(())
209}
210
211fn stop_stand(stand_root: &Path) -> Result<Option<u32>, String> {
212    let pid = read_pid(stand_root);
213    if let Some(pid) = pid {
214        if Command::new("kill")
215            .args(["-0", &pid.to_string()])
216            .status()
217            .map(|s| s.success())
218            .unwrap_or(false)
219        {
220            kill_pid(pid)?;
221        }
222    }
223    Ok(pid)
224}

226async fn wait_health(base: &str, tries: u32) -> Result<(), String> {
227    let client = Client::builder()
228        .timeout(Duration::from_secs(3))
229        .build()
230        .map_err(|e| e.to_string())?;
231    let url = format!("{}/api/v1/health", base.trim_end_matches('/'));
232    for _ in 0..tries {
233        if let Ok(resp) = client.get(&url).send().await {
234            if resp.status().is_success() {
235                eprintln!("OK  health -> {url}");
236                return Ok(());
237            }
238        }
239        tokio::time::sleep(Duration::from_secs(2)).await;
240    }
241    Err(format!("health not ready at {url}"))
242}

257fn build_stand_paths(cli: &Cli, stand_root: PathBuf) -> Result<StandPaths, String> {
258    Ok(StandPaths {
259        stand_root,
260        poolai_bin: resolve_poolai_bin(&cli.profile)?,
261        stand_bin: resolve_stand_bin(),
262        port: cli.port,
263        profile: cli.profile.clone(),
264        job_store: std::env::var(ENV_JOB_STORE).unwrap_or_else(|_| "raid".to_string()),
265        lease_ttl: std::env::var(ENV_LEASE_TTL).unwrap_or_else(|_| "2".to_string()),
266        pricing_fallback: std::env::var(ENV_PRICING_FB)
267            .unwrap_or_else(|_| "{\"inference_blended_token\":470000}".to_string()),
268        k8s_openapi: std::env::var(ENV_K8S).unwrap_or_else(|_| "1.28".to_string()),
269        rust_log: std::env::var(ENV_RUST_LOG).unwrap_or_else(|_| "warn".to_string()),
270    })
271}

273fn write_stand_env(paths: &StandPaths) -> Result<(), String> {
274    let content = format!(
275        "POOLAI_HTTP_PORT={}\n\
276POOLAI_RAID_BASE_PATH={}/raid\n\
277POOLAI_DATA_PATH={}/data\n\
278POOLAI_JOB_STORE={}\n\
279POOLAI_JOB_LEASE_TTL_SECS={}\n\
280POOLAI_GALAXY_PRICING_FALLBACK_JSON={}\n\
281POOLAI_E2E_PROFILE={}\n\
282RUST_LOG={}\n\
283K8S_OPENAPI_ENABLED_VERSION={}\n\
284POOLAI_BIN={}\n\
285{}={}\n",
286        paths.port,
287        paths.stand_root.display(),
288        paths.stand_root.display(),
289        paths.job_store,
290        paths.lease_ttl,
291        paths.pricing_fallback,
292        paths.profile,
293        paths.rust_log,
294        paths.k8s_openapi,
295        paths.poolai_bin.display(),
296        ENV_STAND_BIN,
297        paths.stand_bin.display(),
298    );
299    fs::write(paths.stand_root.join("stand.env"), content).map_err(|e| e.to_string())?;
300    Ok(())
301}

303fn write_restart_script(paths: &StandPaths) -> Result<(), String> {
304    let script = r#"#!/usr/bin/env bash
305set -euo pipefail
306STAND_ROOT="$(cd "$(dirname "$0")" && pwd)"
307# shellcheck source=/dev/null
308source "${STAND_ROOT}/stand.env"
309exec "${POOLAI_E2E_STAND_BIN}" restart --stand-root "${STAND_ROOT}"
310"#;
311    let path = paths.stand_root.join("restart.sh");
312    fs::write(&path, script).map_err(|e| e.to_string())?;
313    #[cfg(unix)]
314    {
315        use std::os::unix::fs::PermissionsExt;
316        let mut perms = fs::metadata(&path)
317            .map_err(|e| e.to_string())?
318            .permissions();
319        perms.set_mode(0o755);
320        fs::set_permissions(&path, perms).map_err(|e| e.to_string())?;
321    }
322    Ok(())
323}

325fn spawn_poolai(paths: &StandPaths) -> Result<u32, String> {
326    let log_path = paths.stand_root.join("poolai.log");
327    let log = OpenOptions::new()
328        .create(true)
329        .append(true)
330        .open(&log_path)
331        .map_err(|e| format!("open log: {e}"))?;
332    let log_err = log.try_clone().map_err(|e| e.to_string())?;
333
334    let child = Command::new(&paths.poolai_bin)
335        .env(ENV_PORT, paths.port.to_string())
336        .env("POOLAI_RAID_BASE_PATH", paths.stand_root.join("raid"))
337        .env("POOLAI_DATA_PATH", paths.stand_root.join("data"))
338        .env(ENV_JOB_STORE, &paths.job_store)
339        .env(ENV_LEASE_TTL, &paths.lease_ttl)
340        .env(ENV_PRICING_FB, &paths.pricing_fallback)
341        .env(ENV_RUST_LOG, &paths.rust_log)
342        .env(ENV_K8S, &paths.k8s_openapi)
343        .stdout(Stdio::from(log))
344        .stderr(Stdio::from(log_err))
345        .spawn()
346        .map_err(|e| format!("spawn poolai: {e}"))?;
347
348    Ok(child.id())
349}

351fn prepare_stand_dirs(stand_root: &Path) -> Result<(), String> {
352    fs::create_dir_all(stand_root.join("raid")).map_err(|e| e.to_string())?;
353    fs::create_dir_all(stand_root.join("data")).map_err(|e| e.to_string())?;
354    Ok(())

357async fn start_stand(cli: &Cli) -> Result<StandReport, String> {
358    let stand_root = cli.stand_root.clone().unwrap_or_else(default_stand_root);
359    prepare_stand_dirs(&stand_root)?;
360    let paths = build_stand_paths(cli, stand_root)?;
361    write_stand_env(&paths)?;
362    write_restart_script(&paths)?;
363
364    eprintln!(
365        "Starting poolai ({}) on port {} (job store: {})...",
366        paths.poolai_bin.display(),
367        paths.port,
368        paths.job_store
369    );
370
371    let pid = spawn_poolai(&paths)?;
372    write_pid(&paths.stand_root, pid)?;
373    let base = base_url(paths.port);
374    wait_health(&base, cli.health_tries).await?;
375
376    Ok(StandReport {
377        ok: true,
378        command: "start",
379        stand_root: paths.stand_root.display().to_string(),
380        port: paths.port,
381        pid: Some(pid),
382        base_url: base,
383        poolai_bin: paths.poolai_bin.display().to_string(),
384        stand_bin: paths.stand_bin.display().to_string(),
385        tool: "poolai-e2e-stand",
386    })
387}

389async fn restart_stand(cli: &Cli) -> Result<StandReport, String> {
390    let stand_root = cli
391        .stand_root
392        .clone()
393        .ok_or(format!("restart requires --stand-root or {ENV_STAND_ROOT}"))?;
394    if !stand_root.is_dir() {
395        return Err(format!("stand root not found: {}", stand_root.display()));
396    }
397    stop_stand(&stand_root)?;
398    let paths = build_stand_paths(cli, stand_root)?;
399    if !paths.stand_root.join("stand.env").is_file() {
400        write_stand_env(&paths)?;
401        write_restart_script(&paths)?;
402    }
403    let pid = spawn_poolai(&paths)?;
404    write_pid(&paths.stand_root, pid)?;
405    let base = base_url(paths.port);
406    wait_health(&base, cli.health_tries).await?;
407    Ok(StandReport {
408        ok: true,
409        command: "restart",
410        stand_root: paths.stand_root.display().to_string(),
411        port: paths.port,
412        pid: Some(pid),
413        base_url: base,
414        poolai_bin: paths.poolai_bin.display().to_string(),
415        stand_bin: paths.stand_bin.display().to_string(),
416        tool: "poolai-e2e-stand",
417    })
418}

420fn stop_stand_report(cli: &Cli) -> Result<StandReport, String> {
421    let stand_root = cli
422        .stand_root
423        .clone()
424        .ok_or(format!("stop requires --stand-root or {ENV_STAND_ROOT}"))?;
425    let pid = stop_stand(&stand_root)?;
426    Ok(StandReport {
427        ok: true,
428        command: "stop",
429        stand_root: stand_root.display().to_string(),
430        port: cli.port,
431        pid,
432        base_url: base_url(cli.port),
433        poolai_bin: resolve_poolai_bin(&cli.profile)
434            .map(|p| p.display().to_string())
435            .unwrap_or_default(),
436        stand_bin: resolve_stand_bin().display().to_string(),
437        tool: "poolai-e2e-stand",
438    })
439}

441fn print_report(cli: &Cli, report: &StandReport) {
442    if cli.json_out {
443        println!(
444            "{}",
445            serde_json::to_string_pretty(report).unwrap_or_default()
446        );
447    } else if cli.print_stand_root {
448        println!("{}", report.stand_root);
449    } else {
450        eprintln!(
451            "poolai-e2e-stand {} OK stand={} pid={:?} base={}",
452            report.command, report.stand_root, report.pid, report.base_url
453        );
454    }
455}

457#[tokio::main]
458async fn main() -> Result<(), Box<dyn std::error::Error + Send + Sync>> {
459    let cli = match parse_cli() {
460        Ok(c) => c,
461        Err(e) => {
462            eprintln!("poolai-e2e-stand: {e}");
463            std::process::exit(1);
464        }
465    };
466
467    let report = match cli.command {
468        CommandKind::Start => start_stand(&cli).await?,
469        CommandKind::Restart => restart_stand(&cli).await?,
470        CommandKind::Stop => stop_stand_report(&cli)?,
471    };
472
473    print_report(&cli, &report);
474    Ok(())
475}
