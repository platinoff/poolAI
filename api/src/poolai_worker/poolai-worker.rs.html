47fn parse_args() -> Args {
48    let mut worker_id: Option<String> = None;
49    let mut coordinator_url = std::env::var("POOLAI_COORDINATOR_URL")
50        .unwrap_or_else(|_| "http://127.0.0.1:8080".to_string());
51    let mut advertise_address =
52        std::env::var("POOLAI_WORKER_ADDRESS").unwrap_or_else(|_| "127.0.0.1".to_string());
53    let mut advertise_port: u16 = std::env::var("POOLAI_WORKER_PORT")
54        .ok()
55        .and_then(|s| s.parse().ok())
56        .unwrap_or(0);
57    let mut max_memory_mb: Option<usize> = None;
58    let mut register_interval_secs: u64 = std::env::var("POOLAI_REGISTER_INTERVAL_SECS")
59        .ok()
60        .and_then(|s| s.parse().ok())
61        .unwrap_or(120);
62    let mut heartbeat_interval_secs: u64 = std::env::var("POOLAI_HEARTBEAT_INTERVAL_SECS")
63        .ok()
64        .and_then(|s| s.parse().ok())
65        .unwrap_or(10);
66    let mut channel =
67        std::env::var("POOLAI_WORKER_CHANNEL").unwrap_or_else(|_| "telegram".to_string());
68    let mut telegram_id = std::env::var("POOLAI_TELEGRAM_ID").ok();
69
70    let mut it = std::env::args().skip(1);
71    while let Some(arg) = it.next() {
72        match arg.as_str() {
73            "--worker-id" => worker_id = it.next(),
74            "--coordinator" => coordinator_url = it.next().unwrap_or(coordinator_url),
75            "--address" => advertise_address = it.next().unwrap_or(advertise_address),
76            "--port" => {
77                advertise_port = it
78                    .next()
79                    .and_then(|v| v.parse().ok())
80                    .unwrap_or(advertise_port);
81            }
82            "--max-memory" => {
83                max_memory_mb = it.next().and_then(|v| v.parse::<usize>().ok());
84            }
85            "--interval" => {
86                register_interval_secs = it
87                    .next()
88                    .and_then(|v| v.parse().ok())
89                    .unwrap_or(register_interval_secs);
90            }
91            "--heartbeat-interval" => {
92                heartbeat_interval_secs = it
93                    .next()
94                    .and_then(|v| v.parse().ok())
95                    .unwrap_or(heartbeat_interval_secs);
96            }
97            "--channel" => channel = it.next().unwrap_or(channel),
98            "--telegram-id" => telegram_id = it.next(),
99            other if other.starts_with('-') => warn!("Ignoring unknown argument: {}", other),
100            _ => {}
101        }
102    }
103
104    if advertise_port == 0 {
105        advertise_port = 9090;
106    }
107
108    Args {
109        worker_id: worker_id.unwrap_or_else(|| format!("worker-{}", uuid::Uuid::new_v4())),
110        coordinator_url: coordinator_url.trim_end_matches('/').to_string(),
111        advertise_address,
112        advertise_port,
113        max_memory_mb: max_memory_mb.unwrap_or(2048),
114        register_interval_secs,
115        heartbeat_interval_secs,
116        channel,
117        telegram_id,
118    }
119}

121fn http_client() -> Result<reqwest::Client, String> {
122    reqwest::Client::builder()
123        .timeout(Duration::from_secs(10))
124        .build()
125        .map_err(|e| e.to_string())
126}

128fn registration_body(args: &Args) -> serde_json::Value {
129    let mut metadata = HashMap::new();
130    metadata.insert("channel".to_string(), args.channel.clone());
131    metadata.insert("role".to_string(), "virtual_node".to_string());
132    if let Some(tg) = &args.telegram_id {
133        metadata.insert("telegram_id".to_string(), tg.clone());
134    }
135
136    serde_json::json!({
137        "peer_id": args.worker_id,
138        "address": args.advertise_address,
139        "port": args.advertise_port,
140        "capabilities": {
141            "cpu_cores": num_cpus::get(),
142            "memory_mb": args.max_memory_mb,
143            "gpu_devices": [],
144            "supports_tensor_parallelism": false,
145            "supports_pipeline_parallelism": false,
146        },
147        "metadata": metadata,
148    })
149}

151async fn register_remote(client: &reqwest::Client, args: &Args) -> Result<(), String> {
152    let url = format!("{}/api/v1/discovery/register-remote", args.coordinator_url);
153    let response = client
154        .post(&url)
155        .json(&registration_body(args))
156        .send()
157        .await
158        .map_err(|e| format!("register-remote request failed: {e}"))?;
159
160    if !response.status().is_success() {
161        let status = response.status();
162        let text = response.text().await.unwrap_or_default();
163        return Err(format!("register-remote HTTP {status}: {text}"));
164    }
165    Ok(())
166}

168async fn heartbeat_remote(client: &reqwest::Client, args: &Args) -> Result<(), String> {
169    let url = format!("{}/api/v1/discovery/heartbeat-remote", args.coordinator_url);
170    let body = serde_json::json!({
171        "peer_id": args.worker_id,
172        "capabilities": {
173            "cpu_cores": num_cpus::get(),
174            "memory_mb": args.max_memory_mb,
175            "gpu_devices": [],
176            "supports_tensor_parallelism": false,
177            "supports_pipeline_parallelism": false,
178            "active_requests": 0,
179            "capacity": 10,
180            "current_load": 0.0,
181        }
182    });
183    let response = client
184        .post(&url)
185        .json(&body)
186        .send()
187        .await
188        .map_err(|e| format!("heartbeat-remote failed: {e}"))?;
189
190    if !response.status().is_success() {
191        let status = response.status();
192        let text = response.text().await.unwrap_or_default();
193        return Err(format!("heartbeat-remote HTTP {status}: {text}"));
194    }
195    Ok(())
196}

198/// Phase 2: verify coordinator pool API is reachable (task path stub).
199async fn poll_coordinator_links(client: &reqwest::Client, rt: &WorkerRuntime) {
200    let health_url = format!("{}/api/v1/health", rt.args.coordinator_url);
201    let health_ok = client
202        .get(&health_url)
203        .send()
204        .await
205        .map(|r| r.status().is_success())
206        .unwrap_or(false);
207    *rt.coordinator_reachable.write().await = health_ok;
208
209    let pool_url = format!("{}/api/v1/workers", rt.args.coordinator_url);
210    let pool_ok = client
211        .get(&pool_url)
212        .send()
213        .await
214        .map(|r| r.status().is_success())
215        .unwrap_or(false);
216    *rt.pool_api_reachable.write().await = pool_ok;
217    if pool_ok {
218        rt.tasks_polled.fetch_add(1, Ordering::Relaxed);
219    }
220}

222async fn health_handler(State(rt): State<Arc<WorkerRuntime>>) -> Json<WorkerHealthResponse> {
223    Json(WorkerHealthResponse {
224        status: "healthy",
225        worker_id: rt.args.worker_id.clone(),
226        role: "virtual_node",
227        channel: rt.args.channel.clone(),
228        coordinator_reachable: *rt.coordinator_reachable.read().await,
229        pool_api_reachable: *rt.pool_api_reachable.read().await,
230        tasks_polled: rt.tasks_polled.load(Ordering::Relaxed),
231    })
232}

234async fn run_health_server(rt: Arc<WorkerRuntime>) -> Result<(), String> {
235    let app = Router::new()
236        .route("/health", get(health_handler))
237        .route("/api/v1/health", get(health_handler))
238        .with_state(rt.clone());
239
240    let addr = format!("{}:{}", rt.args.advertise_address, rt.args.advertise_port);
241    let listener = tokio::net::TcpListener::bind(&addr)
242        .await
243        .map_err(|e| format!("bind {addr}: {e}"))?;
244    info!("Worker health server listening on http://{}", addr);
245    axum::serve(listener, app)
246        .await
247        .map_err(|e| format!("health server: {e}"))?;
248    Ok(())
249}

251#[tokio::main]
252async fn main() -> Result<(), Box<dyn std::error::Error>> {
253    tracing_subscriber::fmt()
254        .with_env_filter(tracing_subscriber::EnvFilter::from_default_env())
255        .init();
256
257    let args = parse_args();
258    let rt = Arc::new(WorkerRuntime {
259        args: args.clone(),
260        tasks_polled: Arc::new(AtomicU64::new(0)),
261        coordinator_reachable: Arc::new(RwLock::new(false)),
262        pool_api_reachable: Arc::new(RwLock::new(false)),
263    });
264
265    info!(
266        "PoolAI virtual node: id={}, coordinator={}, health=http://{}:{}/health",
267        args.worker_id, args.coordinator_url, args.advertise_address, args.advertise_port
268    );
269
270    let rt_server = rt.clone();
271    tokio::spawn(async move {
272        if let Err(e) = run_health_server(rt_server).await {
273            error!("Health server exited: {}", e);
274        }
275    });
276
277    let client = http_client()?;
278    register_remote(&client, &args).await?;
279    info!("Registered with coordinator");
280
281    let mut ticks: u64 = 0;
282    loop {
283        poll_coordinator_links(&client, &rt).await;
284
285        match heartbeat_remote(&client, &args).await {
286            Ok(()) => tracing::debug!("Heartbeat OK"),
287            Err(e) => {
288                warn!("Heartbeat failed (will re-register on interval): {}", e);
289                if let Err(reg_err) = register_remote(&client, &args).await {
290                    error!("Re-register failed: {}", reg_err);
291                }
292            }
293        }
294
295        ticks = ticks.saturating_add(1);
296        let register_every = (args.register_interval_secs / args.heartbeat_interval_secs).max(1);
297        if ticks.is_multiple_of(register_every) {
298            if let Err(e) = register_remote(&client, &args).await {
299                error!("Periodic re-register failed: {}", e);
300            }
301        }
302
303        tokio::time::sleep(Duration::from_secs(args.heartbeat_interval_secs)).await;
304    }
305}
