82fn parse_args() -> Args {
83    let mut worker_id: Option<String> = None;
84    let mut coordinator_url = std::env::var("POOLAI_COORDINATOR_URL")
85        .unwrap_or_else(|_| "http://127.0.0.1:8080".to_string());
86    let mut advertise_address =
87        std::env::var("POOLAI_WORKER_ADDRESS").unwrap_or_else(|_| "127.0.0.1".to_string());
88    let mut advertise_port: u16 = std::env::var("POOLAI_WORKER_PORT")
89        .ok()
90        .and_then(|s| s.parse().ok())
91        .unwrap_or(0);
92    let mut max_memory_mb: Option<usize> = None;
93    let mut register_interval_secs: u64 = std::env::var("POOLAI_REGISTER_INTERVAL_SECS")
94        .ok()
95        .and_then(|s| s.parse().ok())
96        .unwrap_or(120);
97    let mut heartbeat_interval_secs: u64 = std::env::var("POOLAI_HEARTBEAT_INTERVAL_SECS")
98        .ok()
99        .and_then(|s| s.parse().ok())
100        .unwrap_or(10);
101    let mut channel =
102        std::env::var("POOLAI_WORKER_CHANNEL").unwrap_or_else(|_| "telegram".to_string());
103    let mut telegram_id = std::env::var("POOLAI_TELEGRAM_ID").ok();
104
105    let mut it = std::env::args().skip(1);
106    while let Some(arg) = it.next() {
107        match arg.as_str() {
108            "--worker-id" => worker_id = it.next(),
109            "--coordinator" => coordinator_url = it.next().unwrap_or(coordinator_url),
110            "--address" => advertise_address = it.next().unwrap_or(advertise_address),
111            "--port" => {
112                advertise_port = it
113                    .next()
114                    .and_then(|v| v.parse().ok())
115                    .unwrap_or(advertise_port);
116            }
117            "--max-memory" => {
118                max_memory_mb = it.next().and_then(|v| v.parse::<usize>().ok());
119            }
120            "--interval" => {
121                register_interval_secs = it
122                    .next()
123                    .and_then(|v| v.parse().ok())
124                    .unwrap_or(register_interval_secs);
125            }
126            "--heartbeat-interval" => {
127                heartbeat_interval_secs = it
128                    .next()
129                    .and_then(|v| v.parse().ok())
130                    .unwrap_or(heartbeat_interval_secs);
131            }
132            "--channel" => channel = it.next().unwrap_or(channel),
133            "--telegram-id" => telegram_id = it.next(),
134            other if other.starts_with('-') => warn!("Ignoring unknown argument: {}", other),
135            _ => {}
136        }
137    }
138
139    if advertise_port == 0 {
140        advertise_port = 9090;
141    }
142
143    Args {
144        worker_id: worker_id.unwrap_or_else(|| format!("worker-{}", uuid::Uuid::new_v4())),
145        coordinator_url: coordinator_url.trim_end_matches('/').to_string(),
146        advertise_address,
147        advertise_port,
148        max_memory_mb: max_memory_mb.unwrap_or(2048),
149        register_interval_secs,
150        heartbeat_interval_secs,
151        channel,
152        telegram_id,
153        cache_dir: resolve_cache_dir(),
154    }
155}

192            "supports_pipeline_parallelism": false,

198async fn register_remote(client: &reqwest::Client, args: &Args) -> Result<(), String> {
199    let url = format!("{}/api/v1/discovery/register-remote", args.coordinator_url);
200    let response = client
201        .post(&url)
202        .json(&registration_body(args))
203        .send()
204        .await
205        .map_err(|e| format!("register-remote request failed: {e}"))?;
206
207    if !response.status().is_success() {
208        let status = response.status();
209        let text = response.text().await.unwrap_or_default();
210        return Err(format!("register-remote HTTP {status}: {text}"));
211    }
212    if let Some(tg) = &args.telegram_id {
213        bind_telegram_remote(client, args, tg).await?;
214    }
215    join_worker_pool(client, args).await?;
216    Ok(())
217}

219async fn join_worker_pool(client: &reqwest::Client, args: &Args) -> Result<(), String> {
220    let url = format!(
221        "{}/api/v1/virtual-nodes/{}/pool/join",
222        args.coordinator_url, args.worker_id
223    );
224    let body = serde_json::json!({
225        "max_memory_mb": args.max_memory_mb,
226        "max_concurrent_requests": 10,
227    });
228    let response = client
229        .post(&url)
230        .json(&body)
231        .send()
232        .await
233        .map_err(|e| format!("pool join request failed: {e}"))?;
234    match response.status() {
235        reqwest::StatusCode::CREATED => {
236            info!("Joined coordinator worker pool as {}", args.worker_id);
237            Ok(())
238        }
239        reqwest::StatusCode::SERVICE_UNAVAILABLE => {
240            warn!("Coordinator pool not ready; worker stays discovery-only");
241            Ok(())
242        }
243        status => {
244            let text = response.text().await.unwrap_or_default();
245            Err(format!("pool join HTTP {status}: {text}"))
246        }
247    }
248}

311fn lease_renew_target(payload: &serde_json::Value) -> Option<(String, u64)> {
312    let job_id = payload
313        .get("job_id")
314        .and_then(|v| v.as_str())
315        .map(str::trim)
316        .filter(|s| !s.is_empty())?
317        .to_string();
318    let lease_epoch = payload
319        .get("lease_epoch")
320        .and_then(|v| v.as_u64())
321        .or_else(|| {
322            payload
323                .get("lease")
324                .and_then(|l| l.get("epoch"))
325                .and_then(|v| v.as_u64())
326        })?;
327    Some((job_id, lease_epoch))
328}

334fn lease_renew_error_stops_loop(err: &str) -> bool {
335    err.contains("409")
336        || err.contains("lease_epoch_rejected")
337        || err.contains("lease_already_active")
338        || err.contains("lease_not_found")

341/// Periodic renew while a leased task runs (PH-S116; interval from `JobLeaseConfig`).
342async fn run_lease_renew_ticker(
343    client: reqwest::Client,
344    args: Args,
345    job_id: String,
346    lease_epoch: u64,
347    interval_secs: u64,
348) {
349    if interval_secs == 0 {
350        return;
351    }
352    let mut ticker = tokio::time::interval(Duration::from_secs(interval_secs));
353    ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
354    ticker.tick().await;
355    loop {
356        ticker.tick().await;
357        match renew_job_lease(&client, &args, &job_id, lease_epoch).await {
358            Ok(()) => info!(
359                "Lease heartbeat: job_id={}, lease_epoch={}",
360                job_id, lease_epoch
361            ),
362            Err(e) => {
363                if lease_renew_error_stops_loop(&e) {
364                    warn!("Lease renew loop stopped for {}: {}", job_id, e);
365                    break;
366                }
367                warn!("Lease renew failed for {} (will retry): {}", job_id, e);
368            }
369        }
370    }
371}
372
373struct LeaseRenewGuard {
374    ticker: Option<tokio::task::JoinHandle<()>>,
375}
376
377impl LeaseRenewGuard {
378    fn start(client: &reqwest::Client, args: &Args, job_id: &str, lease_epoch: u64) -> Self {
379        let interval = lease_renew_interval_secs();
380        let ticker = if interval > 0 {
381            let client = client.clone();
382            let args = args.clone();
383            let job_id = job_id.to_string();
384            Some(tokio::spawn(run_lease_renew_ticker(
385                client,
386                args,
387                job_id,
388                lease_epoch,
389                interval,
390            )))
391        } else {
392            None
393        };
394        Self { ticker }
395    }
396}
397
398impl Drop for LeaseRenewGuard {
399    fn drop(&mut self) {
400        if let Some(handle) = self.ticker.take() {
401            handle.abort();
402        }
403    }
404}

467async fn run_raid_artifact_probe(
468    client: &reqwest::Client,
469    args: &Args,
470    payload: &serde_json::Value,
471) -> bool {
472    let bytes = match probe_bytes(payload) {
473        Ok(b) => b,
474        Err(e) => {
475            warn!("RAID artifact probe bytes failed: {}", e);
476            return false;
477        }
478    };
479    let message = match build_probe_message(&args.worker_id, payload) {
480        Ok(m) => m,
481        Err(e) => {
482            warn!("RAID artifact probe build failed: {}", e);
483            return false;
484        }
485    };
486    let url = format!(
487        "{}/api/v1/raid/distributed/artifacts/replicate",
488        args.coordinator_url
489    );
490    let wire_ok = match client.post(&url).json(&message).send().await {
491        Ok(response) => response.status().is_success(),
492        Err(e) => {
493            warn!("RAID wire put_artifact probe failed: {}", e);
494            false
495        }
496    };
497    if wire_ok {
498        if let Some(cache) = &args.cache_dir {
499            let name = probe_logical_name(payload);
500            match store_probe(cache, name, &bytes) {
501                Ok(path) => info!("Cached probe artifact at {}", path.display()),
502                Err(e) => warn!("Local artifact cache write failed: {}", e),
503            }
504        }
505    }
506    wire_ok
507}

532async fn execute_task(
533    client: &reqwest::Client,
534    rt: &WorkerRuntime,
535    task: VirtualNodeTaskDto,
536) -> (String, bool) {
537    let task_type = task.task_type.as_str();
538    let mut runtime = TaskRuntime::default();
539
540    let _lease_renew_guard = match lease_renew_target(&task.payload) {
541        Some((job_id, lease_epoch)) => {
542            match renew_job_lease(client, &rt.args, &job_id, lease_epoch).await {
543                Ok(()) => info!(
544                    "Lease renewed: job_id={}, lease_epoch={}",
545                    job_id, lease_epoch
546                ),
547                Err(e) => warn!("Initial lease renew failed for {}: {}", job_id, e),
548            }
549            Some(LeaseRenewGuard::start(
550                client,
551                &rt.args,
552                &job_id,
553                lease_epoch,
554            ))
555        }
556        None => None,
557    };
558
559    let run_raid_health = matches!(task_type, "raid_health_check")
560        || (task_type == "telegram_command" && telegram_command_is(&task.payload, "/raid"));
561    if run_raid_health {
562        let ok = run_raid_health_check(client, &rt.args).await;
563        *rt.raid_wire_ok.write().await = ok;
564        runtime.raid_wire_ok = Some(ok);
565    } else if task_type == "telegram_command" && telegram_command_is(&task.payload, "/status") {
566        runtime.raid_wire_ok = Some(*rt.raid_wire_ok.read().await);
567    }
568
569    if matches!(task_type, "raid_artifact_probe") {
570        let ok = run_raid_artifact_probe(client, &rt.args, &task.payload).await;
571        runtime.raid_artifact_ok = Some(ok);
572    }
573
574    if matches!(task_type, "pool_workers_probe" | "telegram_command") {
575        runtime.pool_worker_count = Some(fetch_pool_worker_count(client, &rt.args).await);
576    }
577
578    resolve_task_outcome(task_type, &task.payload, &runtime)
579}

581async fn poll_and_run_tasks(client: &reqwest::Client, rt: &WorkerRuntime) {
582    let url = format!(
583        "{}/api/v1/virtual-nodes/{}/tasks/poll",
584        rt.args.coordinator_url, rt.args.worker_id
585    );
586    let Ok(response) = client.get(&url).send().await else {
587        return;
588    };
589    if !response.status().is_success() {
590        return;
591    }
592    let Ok(body) = response.json::<PollTasksResponse>().await else {
593        return;
594    };
595    rt.tasks_polled.fetch_add(1, Ordering::Relaxed);
596
597    let Some(task) = body.task else {
598        return;
599    };
600
601    let task_id = task.id.clone();
602    let task_type = task.task_type.clone();
603    info!("Running task {} ({})", task_id, task_type);
604    let (status, _ok) = execute_task(client, rt, task).await;
605    if complete_task(client, &rt.args, &task_id, &status, Some(task_type.clone()))
606        .await
607        .is_ok()
608    {
609        rt.tasks_completed.fetch_add(1, Ordering::Relaxed);
610        *rt.last_task.write().await = Some(format!("{task_type}:{status}"));
611        info!("Task {} completed ({})", task_id, status);
612    }
613}

629        .await
630        .map(|r| r.status().is_success())
631        .unwrap_or(false);

633}

646        last_task: rt.last_task.read().await.clone(),
647        cache_dir: rt.args.cache_dir.as_ref().map(|p| p.display().to_string()),
648        cached_artifacts: rt
649            .args
650            .cache_dir
651            .as_ref()
652            .map(|p| count_cached_probes(p))
653            .unwrap_or(0),
654    })
655}

657async fn run_health_server(rt: Arc<WorkerRuntime>) -> Result<(), String> {
658    let app = Router::new()
659        .route("/health", get(health_handler))
660        .route("/api/v1/health", get(health_handler))
661        .with_state(rt.clone());
662
663    let addr = format!("{}:{}", rt.args.advertise_address, rt.args.advertise_port);
664    let listener = tokio::net::TcpListener::bind(&addr)
665        .await
666        .map_err(|e| format!("bind {addr}: {e}"))?;
667    info!("Worker health server listening on http://{}", addr);
668    axum::serve(listener, app)
669        .await
670        .map_err(|e| format!("health server: {e}"))?;
671    Ok(())
672}

674#[tokio::main]
675async fn main() -> Result<(), Box<dyn std::error::Error>> {
676    tracing_subscriber::fmt()
677        .with_env_filter(tracing_subscriber::EnvFilter::from_default_env())
678        .init();
679
680    let args = parse_args();
681    let rt = Arc::new(WorkerRuntime {
682        args: args.clone(),
683        tasks_polled: Arc::new(AtomicU64::new(0)),
684        tasks_completed: Arc::new(AtomicU64::new(0)),
685        coordinator_reachable: Arc::new(RwLock::new(false)),
686        pool_api_reachable: Arc::new(RwLock::new(false)),
687        raid_wire_ok: Arc::new(RwLock::new(false)),
688        last_task: Arc::new(RwLock::new(None)),
689    });
690
691    info!(
692        "PoolAI virtual node: id={}, coordinator={}, health=http://{}:{}/health",
693        args.worker_id, args.coordinator_url, args.advertise_address, args.advertise_port
694    );
695
696    let rt_server = rt.clone();
697    tokio::spawn(async move {
698        if let Err(e) = run_health_server(rt_server).await {
699            error!("Health server exited: {}", e);
700        }
701    });
702
703    let client = http_client()?;
704    register_remote(&client, &args).await?;
705    info!("Registered with coordinator (bootstrap tasks queued)");
706
707    let mut ticks: u64 = 0;
708    loop {
709        poll_coordinator_links(&client, &rt).await;
710        poll_and_run_tasks(&client, &rt).await;
711
712        match heartbeat_remote(&client, &args).await {
713            Ok(()) => tracing::debug!("Heartbeat OK"),
714            Err(e) => {
715                warn!("Heartbeat failed (will re-register): {}", e);
716                if let Err(reg_err) = register_remote(&client, &args).await {
717                    error!("Re-register failed: {}", reg_err);
718                }
719            }
720        }
721
722        ticks = ticks.saturating_add(1);
723        let register_every = (args.register_interval_secs / args.heartbeat_interval_secs).max(1);
724        if ticks.is_multiple_of(register_every) {
725            if let Err(e) = register_remote(&client, &args).await {
726                error!("Periodic re-register failed: {}", e);
727            }
728        }
729
730        tokio::time::sleep(Duration::from_secs(args.heartbeat_interval_secs)).await;
731    }
732}
